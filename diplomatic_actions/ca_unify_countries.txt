ca_unify_countries = {
	groups = { general }
	texture = "gfx/interface/icons/lens_toolbar_icons/incorporate_state.dds"
	requires_approval = yes
	uses_random_approval = no
	can_use_obligations = no
	should_notify_third_parties = yes

	unlocking_technologies = {
	}

	selectable = {
		NOT = { is_country_type = decentralized }
		is_revolutionary = no
		is_secessionist = no
		NOT = { is_subject = yes }
		NOT = { is_at_war = yes }
		only_latin_centralamericans_cultures = yes
	}

	potential = {
		
		scope:target_country = {
			NOT = { is_country_type = decentralized }
			is_ca_culture_country = yes
			country_rank <= rank_value:minor_power
			OR = {
				NOT = { is_subject = yes }
				is_subject_of = scope:actor
			}
			NOT = { is_at_war = yes }
		}
		ca_centralamerican_countries_can_unify_trigger = yes
	}

	possible = {		
		
		NOT = { has_war_with = scope:target_country }
		NOT = { is_diplomatic_play_enemy_of = scope:target_country }

		
		NOT = { infamy >= 25 }
	}


	accept_effect = {
		save_scope_as = ca_unify_actor
		scope:target_country = {
			save_scope_as = ca_unify_target
			custom_tooltip = ca_unify_countries_accept_effect_tt
		}

		annex = scope:target_country
		change_infamy = 5
	}

	decline_effect = {
		change_relations = {
			country = scope:target_country
			value = -25
		}
	}

	ai = {
		check_acceptance_for_will_propose = yes

		evaluation_chance = {
			value = 1
		}

		accept_score = {
			value = 0

			# Unification is a one-way door for the target, so the baseline is hostile.
			add = {
				desc = "DIPLOMATIC_ACCEPTANCE_BASE"
				value = -1000
			}
			add = {
				desc = "DIPLOMATIC_ACCEPTANCE_GDP"
				value = "scope:actor.ai_gdp_comparison(root)"
				min = -100
				max = 1000
			}			
			# Shared religion adds weight even when the cultures differ.
			if = {
				limit = {
					scope:actor.religion = scope:target_country.religion
				}
				add = {
					#desc = "CA_UNIFY_ACCEPT_SHARED_RELIGION"
					value = 10
				}
			}

			# Shared culture adds weight even when the cultures differ.
		    if = {
		    	limit = {
		    		any_primary_culture = {
		    			is_primary_culture_of = scope:actor
		    		}
		    	}
		    	add = {
		    		desc = "DIPLOMATIC_ACCEPTANCE_SAME_CULTURE"
		    		value = 500
		    	}
		    }
		    else_if = {
		    	### If you share a cultural discrimination trait
		    	limit = {
		    		any_primary_culture = {
		    			shares_language_trait_with_any_primary_culture = ROOT
		    		}
		    	}
		    	add = {
		    		desc = "DIPLOMATIC_ACCEPTANCE_CULTURAL_ACCEPTANCE"
		    		value = 200
		    	}
		    }
		    else_if = {
		    	### If you share a cultural discrimination trait
		    	limit = {
		    		any_primary_culture = {
		    			shares_language_trait_group_with_any_primary_culture = ROOT
		    		}
		    	}
		    	add = {
		    		desc = "DIPLOMATIC_ACCEPTANCE_CULTURAL_ACCEPTANCE"
		    		value = 250
		    	}
		    }
		    else_if = {
		    	### If you share a cultural discrimination trait
		    	limit = {
		    		any_primary_culture = {
		    			scope:actor = {
		    				"cultural_acceptance_base(prev)" >= 40
		    			}
		    		}
		    	}
		    	add = {
		    		desc = "DIPLOMATIC_ACCEPTANCE_DIFFERENT_CULTURE"
		    		value = -500
		    	}
		    }
			#DIPLOMATIC ACCEPTANCE RELATIONS
			add = {
				desc = "DIPLOMATIC_ACCEPTANCE_RELATIONS"
				# CAFM fix: "=<" no es un operador valido (es "<="); ademas los tramos se solapaban
				if = {
					limit = {
						"relations(scope:actor)" < relations_threshold:neutral
					}
					value = -1000
				}
				if = {
					limit = {
						"relations(scope:actor)" >= relations_threshold:neutral
						"relations(scope:actor)" < relations_threshold:cordial
					}
					value = -250
				}				
				if = {
					limit = {
						"relations(scope:actor)" >= relations_threshold:cordial
						"relations(scope:actor)" < relations_threshold:amicable
					}
					value = 100
				}
				if = {
					limit = {
						"relations(scope:actor)" >= relations_threshold:amicable
						"relations(scope:actor)" < relations_threshold:friendly
					}
					value = 500
				}
				if = {
					limit = {
						"relations(scope:actor)" >= relations_threshold:friendly
					}
					value = 750
				}
			}


            add = {
				desc = "DIPLOMATIC_ACCEPTANCE_ATTITUDE"
				if = {
					limit = {
						OR = {
							has_attitude = {
								who = scope:actor
								attitude = genial
							}
							has_attitude = {
							    who = scope:actor
							    attitude = loyal
						    }
							has_attitude = {
								who = scope:actor
								attitude = protective
							}
						}
					}
					value = 1000
				}
				else_if = {
					limit = {
						OR = {
							has_attitude = {
								who = scope:actor
								attitude = cooperative
							}
							has_attitude = {
								who = scope:actor
								attitude = conciliatory
							}
						}
					}
					value = 250
				}
				else_if = {
					limit = {
						has_attitude = {
							who = scope:actor
							attitude = domineering
						}
					}
					value = 25
				}
				else_if = {
					limit = {
						OR = {
							has_attitude = {
								who = scope:actor
								attitude = antagonistic
							}
							has_attitude = {
								who = scope:actor
								attitude = belligerent
							}
							has_attitude = {
							    who = scope:actor
							    attitude = rebellious
						}
						}
					}
					value = -1000
				}
				else = {
					value = -25
				}
			}
			
			if = {
				limit = {
					is_in_same_power_bloc = scope:actor
				}
				add = {
					desc = "CA_UNIFY_ACCEPT_SAME_POWER_BLOC"
					value = 500
				}
			}

			
			add = {
				desc = "DIPLOMATIC_ACCEPTANCE_RANK"
				value = scope:actor.country_rank
				subtract = country_rank
				multiply = 15
				min = 0
				max = 75
			}

			
			if = {
				limit = {
					"scope:actor.infamy" >= infamy_threshold:infamous
					"scope:actor.infamy" < infamy_threshold:notorious
				}
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_INFAMY"
					value = -40
				}
			}
			if = {
				limit = {
					"scope:actor.infamy" >= infamy_threshold:notorious
				}
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_INFAMY"
					value = -120
				}
			}
		}

		will_propose = {
			scope:target_country = { # CAFM fix: el que decide aceptar es el objetivo; el actor es root
				would_accept_diplomatic_action = {
					actor = root
					type = ca_unify_countries
				}
			}
		}

		propose_score = {
			value = 10000
		}
	}
}
