# Reconstruction
fsa_events.1 = {
	type = country_event

	event_image = {
		video = "europenorthamerica_political_extremism"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	title = fsa_events.1.t
	desc = fsa_events.1.d
	flavor = fsa_events.1.f

	duration = 3

	trigger = {
		# Fired via on_action
	}

	immediate = {
		save_scope_as = formation_country
		capital = {
			region = {
				save_scope_as = capital_region_scope
			}
		}
		every_country = {
			limit = {
				has_diplomatic_relevance = ROOT
			}
			post_notification = unification_notification_generic
		}
		add_modifier = {
			name = unification_prestige
			days = very_long_modifier_time
		}
		unification_claims_effect = yes
	}

	option = {
		name = fsa_events.1.a
		default_option = yes
		if = {
			limit = {
				NOT = {
					has_law = law_type:law_slavery_banned
				}
			}
			activate_law = law_type:law_slavery_banned
		}
		if = {
			limit = {
				NOT = {
					has_law = law_type:law_multicultural
				}
			}
			activate_law = law_type:law_multicultural
		}
	}

	option = {
		name = fsa_events.1.b
		if = {
			limit = {
				NOT = {
					has_law = law_type:law_slavery_banned
				}
			}
			activate_law = law_type:law_slavery_banned
		}
		if = {
			limit = {
				NOT = {
					has_law = law_type:law_multicultural
				}
			}
			activate_law = law_type:law_multicultural
		}
		add_primary_culture = cu:afro_american
		every_interest_group = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_reactionary
					has_ideology = ideology:ideology_paternalistic
					has_ideology = ideology:ideology_magnatial
					ig_has_carlist_ideology = yes
					has_ideology = ideology:ideology_moderantist
					leader = {
						has_reactionary_ideology = yes
					}
				}
			}
			prev = {
				add_radicals = {
					value = medium_radicals
					interest_group = prev
				}
			}
		}
		every_interest_group = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_anti_slavery
					has_ideology = ideology:ideology_liberal
					has_ideology = ideology:ideology_egalitarian
					has_ideology = ideology:ideology_anarchist # They approve of multiculturalism
				}
			}
			prev = {
				add_loyalists = {
					value = medium_radicals
					interest_group = prev
				}
			}
		}
		add_loyalists = {
			value = very_large_radicals
			culture = cu:afro_american
		}
		add_modifier = {
			name = fsa_reparation_cost
			days = normal_modifier_time
		}
		every_scope_state = {
			custom = every_yankee_state_in_root
			limit = {
				state_region = { is_homeland = cu:yankee }
			}
			state_region = { add_homeland = cu:afro_american }
			add_modifier = {
				name = fsa_reparation_reward
				days = normal_modifier_time
			}
		}
	}
}

# The Fugitive Slave
acw_events.2 = {
	type = country_event

	event_image = {
		video = "southamerica_slaves_night"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	title = acw_events.2.t
	desc = acw_events.2.d
	flavor = acw_events.2.f

	duration = 3

	trigger = {
		NOT = { has_law_or_variant = law_type:law_slavery_banned }
		NOT = { has_variable = american_civil_war }
		any_interest_group = {
			is_revolutionary = no
			is_insurrectionary = no
			ig_counts_as_marginal = no
			OR = {
				has_ideology = ideology:ideology_pro_slavery
				has_ideology = ideology:ideology_pro_slavery_colonial
				leader = {
					has_ideology = ideology:ideology_slaver
				}
			}
		}
	}

	immediate = {
		random_interest_group = {
			limit = {
				is_revolutionary = no
				is_insurrectionary = no
				ig_counts_as_marginal = no
				OR = {
					has_ideology = ideology:ideology_pro_slavery
					has_ideology = ideology:ideology_pro_slavery_colonial
					leader = {
						has_ideology = ideology:ideology_slaver
					}
				}
			}
			save_scope_as = slaver_ig
			leader = {
				save_scope_as = slaver_ig_leader
			}
		}
	}

	cooldown = { days = normal_modifier_time }

	cancellation_trigger = {
		has_global_variable = american_civil_war
	}

	option = {
		name = acw_events.2.a
        default_option = yes
		if = {
			limit = {
				active_law:lawgroup_slavery ?= {
					NOT = {
						has_amendment = amendment_type:amendment_american_fugitive_slaves_act
					}
				}
			}
			active_law:lawgroup_slavery ?= {
				add_amendment = {
					type = amendment_american_fugitive_slaves_act
					sponsor = PREV.ig:ig_landowners
				}
			}
		}
		every_interest_group = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_pro_slavery
					has_ideology = ideology:ideology_pro_slavery_colonial
					leader = { has_ideology = ideology:ideology_slaver }
				}
			}
			add_modifier = {
				name = ig_slavers_happy
				days = short_modifier_time
			}
		}
		every_interest_group = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_anti_slavery
					leader = { has_ideology = ideology:ideology_abolitionist }
				}
			}
			add_modifier = {
				name = ig_abolitionists_mad
				days = short_modifier_time
			}
		}
	}
	option = {
		name = acw_events.2.b
		if = {
			limit = {
				active_law:lawgroup_slavery ?= {
					has_amendment = amendment_type:amendment_american_fugitive_slaves_act
				}
			}
			active_law:lawgroup_slavery ?= {
				random_scope_amendment = {
					remove_amendment = yes # TODO make this more targeted once we have the requisite support
				}
			}
		}
		every_interest_group = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_pro_slavery
					has_ideology = ideology:ideology_pro_slavery_colonial
					leader = { has_ideology = ideology:ideology_slaver }
				}
				NOT = { has_modifier = ig_slavers_mad }
			}
			add_modifier = {
				name = ig_slavers_mad
				days = short_modifier_time
			}
		}
		every_interest_group = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_anti_slavery
					leader = { has_ideology = ideology:ideology_abolitionist }
				}
				NOT = { has_modifier = ig_abolitionists_happy }
			}
			add_modifier = {
				name = ig_abolitionists_happy
				days = short_modifier_time
			}
		}
	}
}

# Bleeding [State], a.k.a. Dynamic Bleeding Kansas
acw_events.5 = {
	type = country_event

	title = acw_events.5.t
	desc = acw_events.5.d
	flavor = acw_events.5.f
	placement = scope:bleeding_kansas

	event_image = {
		video = "europenorthamerica_american_civil_war"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_fire.dds"

	duration = 3

	trigger = {
		exists = c:USA
		c:USA = ROOT
		NOT = { has_law_or_variant = law_type:law_slavery_banned }
		NOT = { has_variable = bleeding_kansas_happened }
		any_scope_state = {
			is_slave_state = no
			is_incorporated = no
			any_neighbouring_state = {
				owner = ROOT
				is_slave_state = yes
			}
		}
	}

	immediate = {
		set_variable = {
			name = bleeding_kansas_happened
			value = yes
		}
		random_scope_state = {
			limit = {
				is_slave_state = no
				is_incorporated = no
				any_neighbouring_state = {
					owner = ROOT
					is_slave_state = yes
				}
			}
			save_scope_as = bleeding_kansas
		}
	}

	cancellation_trigger = {
		has_global_variable = american_civil_war
	}

	option = {
		name = acw_events.5.a
		default_option = yes
		hidden_effect = {
			scope:bleeding_kansas = {
				liberate_slaves = yes
			}
		}

		custom_tooltip = acw_events_5_free_state_tt
	}
	option = {
		name = acw_events.5.b

		scope:bleeding_kansas = {
			ordered_neighbouring_state = {
				limit = {
					owner = ROOT
					is_slave_state = yes
				}
				order_by = state_population
				position = 0
				check_range_bounds = no
				ordered_scope_pop = {
					limit = {
						is_pop_type = slaves
					}
					save_temporary_scope_as = relevant_pop
					order_by = total_size
					position = 0
					check_range_bounds = no
					move_partial_pop = {
						state = scope:bleeding_kansas
						population = {
							# move in at most 5% of the target state's population, max 5000, or the max size of the pop, whichever is smaller
							add = scope:bleeding_kansas.state_population
							multiply = 0.05
							if = {
								limit = {
									scope:relevant_pop.total_size >= 5000
								}
								max = 5000
							}
							else = {
								max = scope:relevant_pop.total_size
							}
						}
					}
				}
			}
		}
	}
}

				}

		}

			}

				type = je_acw_war
				target = c:FSA
			}

	}

	}
	option = {

			play_as = c:CSA
