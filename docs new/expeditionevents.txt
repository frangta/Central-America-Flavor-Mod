cold.102 = { #Choose commander
	type = country_event
    placement = ROOT

    title = cold.102.t
    desc = cold.102.d
    flavor = cold.102.f

    duration = 3

	event_image = {
		video = "europenorthamerica_antarctica"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		any_scope_character = {
			has_role = admiral
			is_busy = no
		}
	}

	immediate = {
		random_scope_character = {
			limit = {
				is_busy = no
				has_role = admiral
				has_variable = polar_explorer
				is_ruler = no
			}
			save_scope_as = expedition_leader_1
		}
		if = {
			limit = {
				any_scope_character = {
					is_busy = no
					is_ruler = no
					has_role = admiral
					has_trait = explorer
					NOT = {
						scope:expedition_leader_1 ?= this
					}
				}
			}
			random_scope_character = {
				limit = {
					is_busy = no
					is_ruler = no
					has_role = admiral
					has_trait = explorer
					NOT = {
						scope:expedition_leader_1 ?= this
					}
				}
				save_scope_as = expedition_leader_2
			}
		}
		else = {
			random_scope_character = {
				limit = {
					is_busy = no
					is_ruler = no
					has_role = admiral
					NOT = {
						scope:expedition_leader_1 ?= this
					}
				}
				save_scope_as = expedition_leader_2
			}
		}
		random_scope_character = {
			limit = {
				is_busy = no
				is_ruler = no
				has_role = admiral
				NOR = {
					scope:expedition_leader_1 ?= this
					scope:expedition_leader_2 ?= this
				}
			}
			save_scope_as = expedition_leader_3
		}
		random_scope_character = {
			limit = {
				is_busy = no
				is_ruler = no
				has_role = admiral
				NOR = {
					scope:expedition_leader_1 ?= this
					scope:expedition_leader_2 ?= this
					scope:expedition_leader_3 ?= this
				}
			}
			save_scope_as = expedition_leader_4
		}
	}

	option = {
		name = cold.102.a
		custom_tooltip = expedition_costs_tt
		trigger = {
			exists = scope:expedition_leader_1
		}
		default_option = yes
		highlighted_option = yes
		add_modifier = {
			name = modifier_large_expedition_cost
			multiplier = money_amount_multiplier_medium
		}
		scope:expedition_leader_1 ?= {
			set_variable = expedition_leader_var
			set_character_busy_and_immortal = yes
			ROOT = {
				set_variable = { name = expedition_leader_storage_var value = prev }
			}
		}
		add_journal_entry = {
			type = je_antarctica
		}
	}

	option = {
		name = cold.102.b
		custom_tooltip = expedition_costs_tt
		trigger = {
			exists = scope:expedition_leader_2
		}
		add_modifier = {
			name = modifier_large_expedition_cost
			multiplier = money_amount_multiplier_medium
		}
		scope:expedition_leader_2 ?= {
			set_variable = expedition_leader_var
			set_character_busy_and_immortal = yes
			ROOT = {
				set_variable = { name = expedition_leader_storage_var value = prev }
			}
		}
		add_journal_entry = {
			type = je_antarctica
		}
	}

	option = {
		name = cold.102.c
		custom_tooltip = expedition_costs_tt
		trigger = {
			exists = scope:expedition_leader_3
		}
		add_modifier = {
			name = modifier_large_expedition_cost
			multiplier = money_amount_multiplier_medium
		}
		scope:expedition_leader_3 ?= {
			set_variable = expedition_leader_var
			set_character_busy_and_immortal = yes
			ROOT = {
				set_variable = { name = expedition_leader_storage_var value = prev }
			}
		}
		add_journal_entry = {
			type = je_antarctica
		}

	option = {

		custom_tooltip = expedition_costs_tt
		trigger = {

			exists = scope:expedition_leader_4

cold.1 = {
	type = country_event
	placement = ROOT

	title = cold.1.t
	desc = cold.1.d
	flavor = cold.1.f

	duration = 3

	event_image = {
		video = "europenorthamerica_antarctica"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		je:je_antarctica ?= {
			"scripted_bar_progress(expedition_progress_bar)" >= 2
		}
		NOT = { has_variable = mountains_of_madness }
	}

	cancellation_trigger = {
		NOT = { has_variable = on_expedition }
	}

	immediate = {
		set_variable = mountains_of_madness
	}

	option = { # Move on
		name = cold.1.a
		default_option = yes
		expedition_minor_progress_effect = yes
	}

	option = { # Do some science
		name = cold.1.b
		random_list = {
			50 = {
				modifier = {
					if = {
						limit = {
							ROOT.modifier:country_expedition_events_explorer_mult > 0
						}
						add = 25
					}
					if = {
						limit = {
							scope:expedition_leader.modifier:character_expedition_events_explorer_mult > 0
						}
						add = 100
					}
				}
				scope:expedition_leader = {
					add_modifier = {
						name = antarctic_mountains_modifier
						days = normal_modifier_time
					}
				}
				add_modifier = {
					name = antarctic_research
					days = normal_modifier_time
				}
				expedition_minor_progress_effect = yes
			}
			50 = {
				add_small_peril_effect = yes
			}
		}
	}
	option = { # climb Erebus
		name = cold.1.c
		random_list = {
			25 = {
				modifier = {
					if = {
						limit = {
							ROOT.modifier:country_expedition_events_explorer_mult > 0
						}
						add = 12
					}
					if = {
						limit = {
							scope:expedition_leader.modifier:character_expedition_events_explorer_mult > 0
						}
						add = 100
					}
				}
				scope:expedition_leader = {
					add_modifier = {
						name = antarctic_mountains_modifier
						days = normal_modifier_time
						multiplier = 2
					}
				}
				add_modifier = {
					name = climbed_erebus
					days = normal_modifier_time
				}
				expedition_minor_progress_effect = yes
			}
			75 = {
				add_large_peril_effect = yes
			}
		}
	}
}

	}

		name = cold.5.a
		trigger = {
			NOT = { has_global_variable = south_pole_reached }
		}
		default_option = yes
		set_global_variable = { name = south_pole_reached }
		set_variable = {
			name = first_to_reach_south_pole

			}
		}
		remove_variable = on_expedition
		remove_expedition_events_effect = yes
	}
}


# The Great Ice Barrier
cold.11 = {
	type = country_event
	placement = ROOT

	title = cold.11.t
	desc = cold.11.d
	flavor = cold.11.f

	duration = 3

	event_image = {
		video = "unspecific_iceberg"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		je:je_antarctica ?= {
			"scripted_bar_progress(expedition_progress_bar)" < 2
		}
		NOT = { has_variable = great_ice_barrier_variable }
	}

	cancellation_trigger = {
		NOT = { has_variable = on_expedition }
	}

	immediate = {
		save_scope_as = expedition_country
		set_variable = great_ice_barrier_variable
	}

	option = { # beware of fissures
		name = cold.11.a
		default_option = yes
		random_list = {
			75 = {
				modifier = {
					if = {
						limit = {
							ROOT.modifier:country_expedition_events_explorer_mult > 0
						}
						add = 40
					}
					if = {
						limit = {
							scope:expedition_leader.modifier:character_expedition_events_explorer_mult > 0
						}
						add = 150
					}
				}
				expedition_minor_progress_effect = yes
			}
			25 = {
				add_small_peril_effect = yes
			}
		}
	}
	option = { # take a cautious approach
		name = cold.11.b
		custom_tooltip = expedition_event_ongoing_tt
	}
}

    placement = ROOT

    duration = 3

	trigger = {
		any_scope_character = {
			OR = {
				has_role = general
				has_role = admiral
			}
		}
	}

		random_scope_character = {
			limit = {
				is_ruler = no

					has_role = general
					has_role = admiral

			}

		}

			limit = {
				is_ruler = no

					has_role = general
					has_role = admiral
				}
				NOT = {
					this = scope:expedition_leader_1
				}
			}
			save_scope_as = expedition_leader_2
		}
		random_scope_character = {
			limit = {
				is_ruler = no
				OR = {
					has_role = general
					has_role = admiral
				}
				NOR = {
					this = scope:expedition_leader_1
					this = scope:expedition_leader_2
				}
			}
			save_scope_as = expedition_leader_3
		}
		set_variable = central_africa_expedition_var
		central_africa_location_effect = yes
	}

	option = {
		name = expedition_events.502.a
		default_option = yes
		custom_tooltip = expedition_costs_tt
		add_modifier = {
			name = modifier_large_expedition_cost
			multiplier = money_amount_multiplier_medium
		}
		scope:expedition_leader_1 = {
			set_variable = expedition_leader_var
			set_character_busy_and_immortal = yes
			save_scope_as = expedition_leader
			ROOT = {
				set_variable = { name = expedition_leader_storage_var value = prev }

			exists = scope:expedition_leader_2
		}

			name = modifier_large_expedition_cost

		}
		scope:expedition_leader_2 = {
			set_variable = expedition_leader_var
			set_character_busy_and_immortal = yes
			save_scope_as = expedition_leader
			ROOT = {
				set_variable = { name = expedition_leader_storage_var value = prev }
			}
		}
		trigger_event = { id = expedition_events.504 days = 30 }
	}

	option = {

    		limit = { exists = var:expedition_commander_vice }

				name = expedition_commander_vice

			}
		}
		else = {
			set_variable = {
				name = expedition_commander_vice
				value = 1
			}

			limit = { has_variable = congo_river_expedition_var }

			limit = { has_variable = niger_river_expedition_var }

		}
		else_if = {
			limit = { has_variable = central_africa_expedition_var }

		}
		else_if = {
			limit = { has_variable = west_america_expedition_var }
