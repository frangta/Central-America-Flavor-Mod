			}
		}
	}

		name = military_tech_events.402.c
		trigger = {
			scope:machine_gun_general = {
				OR = {
					NOT = { has_trait = trench_rat }
					NOT = { has_trait = stalwart_defender }
				}
			}
		}
		if = {
			limit ={
				scope:machine_gun_general = {
					has_trait = stalwart_defender
				}
			}
			scope:machine_gun_general = {
				add_trait = trench_rat
			}
		}
		else = {
			scope:machine_gun_general = {
				add_trait = stalwart_defender
			}
		}
	}
}
