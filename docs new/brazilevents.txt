# JE initiation event
amazonas.1 = {
	type = country_event
	placement = scope:amazon_rainforest_state_scope

	duration = 3

	title = amazonas.1.t
	desc = amazonas.1.d
	flavor = amazonas.1.f

	event_image = {
		video = "gfx/event_pictures/middleeast_jungle_expedition.bk2"
	}

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	cooldown = { days = normal_modifier_time }

	trigger = {
		is_at_war = no
		in_default = no
		any_scope_state = {
			has_state_trait = state_trait_amazon_rainforest
			has_potential_resource = building_rubber_plantation
			state_region = {
				remaining_undepleted = {
					type = building_rubber_plantation
					amount >= 15
				}
			}
		}
		has_technology_researched = rubber_mastication
		NOT = { has_journal_entry = je_amazonas }
	}

	immediate = {
		g:rubber = {
			save_scope_as = rubber_scope
		}
		if = {
			limit = {
				any_scope_state = {
					state_region = s:STATE_AMAZONAS
				}
			}
			random_scope_state = {
				limit = {
					state_region = s:STATE_AMAZONAS
				}
				save_scope_as = amazon_rainforest_state_scope
			}
		}
		else = {
			random_scope_state = {
				limit = {
					has_state_trait = state_trait_amazon_rainforest
					has_potential_resource = building_rubber_plantation
					state_region = {
						remaining_undepleted = {
							type = building_rubber_plantation
							amount >= 15
						}
					}
				}
				save_scope_as = amazon_rainforest_state_scope
			}
		}
	}

	option = {
		name = amazonas.1.a
		default_option = yes
		custom_tooltip = amazonas.1.tt
		add_journal_entry = { type = je_amazonas }
	}
}

# Treaty of Ayacucho Event / Bolivian Amazonas
amazonas.2 = {
	type = country_event
	placement = scope:amazonas_state_scope

	duration = 3

	title = amazonas.2.t
	desc = amazonas.2.d
	flavor = amazonas.2.f

	event_image = {
		video = "gfx/event_pictures/southamerica_aristocrats.bk2"
	}

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	cooldown = { days = normal_modifier_time }

	trigger = {
		# triggered by amazon_border_treaty_button_1
	}

	immediate = {
		random_country = {
			limit = {
				is_player = no
				capital = { region = sr:region_andes }
				any_scope_state = { state_region = s:STATE_AMAZONAS }
				OR = {
					relations:root >= relations_threshold:amicable
					army_size < { # your army is four times as large
    					value = root.army_size
    					multiply = 0.25
					}
					is_subject_of = root
				}
			}
			save_scope_as = bolivia_border_treaty_country_scope
		}
		ROOT = {
			save_scope_as = brazil_scope
		}
		random_scope_state = {
			limit = {
				state_region = s:STATE_AMAZONAS
			}
			save_scope_as = amazonas_state_scope
		}
	}

	option = {
		name = amazonas.2.a
		default_option = yes
		add_modifier = {
			name = amazonic_indemnities_brazil_1
			days = normal_modifier_time
		}
		random_country = {
			limit = {
				is_player = no
				capital = { region = sr:region_andes }
				any_scope_state = { state_region = s:STATE_AMAZONAS }
				OR = {
					relations:root >= relations_threshold:amicable
					army_size < { # your army is four times as large
    					value = root.army_size
    					multiply = 0.25
					}
					is_subject_of = root
					c:BRZ ?= root
				}
			}
			random_scope_state = {
				limit = {
					state_region = s:STATE_AMAZONAS
				}
				set_state_owner = root
			}
			if = {
				limit = {
					any_scope_state = {
						state_region = s:STATE_MATO_GROSSO
					}
				}
				random_scope_state = {
					limit = {
						state_region = s:STATE_MATO_GROSSO
					}
					set_state_owner = root
				}
			}
			add_modifier = {
				name = amazonic_indemnities_recipient_1
				days = normal_modifier_time
			}
		}

		every_country = {
			limit = {
				NOT = {
					THIS = ROOT
				}
				OR = {
					has_interest_marker_in_region = region_andes
					has_interest_marker_in_region = region_brazil
				}
			}
			post_notification = brz_amazon_border_treaty_bolivia
		}
	}
}

culture_brazil.1 = {
	type = country_event
	placement = scope:state

	title = culture_brazil.1.t
	desc = culture_brazil.1.d
	flavor = culture_brazil.1.f

	gui_window = event_window_1char_tabloid

	left_icon = scope:brazilian_national_hero

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/1Character_Banner"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	duration = 3

	dlc = dlc008

	cooldown = { days = normal_modifier_time }

	trigger = {
		# triggered by on_action on_battle_won pulse
		has_journal_entry = je_brazilian_nation_building
		OR = {
			AND = {
				NOT = { has_variable = brazilian_sulista_hero_var }
				NOT = { has_variable = sulistas_integrated }
			}
			AND = {
				NOT = { has_variable = brazilian_nordestino_hero_var }
				NOT = { has_variable = nordestinos_integrated }
			}
			AND = {
				NOT = { has_variable = brazilian_amazonic_hero_var }
				NOT = { has_variable = amazonics_integrated }
			}
			AND = {
				NOT = { has_variable = brazilian_paulista_hero_var }
				NOT = { has_variable = paulistas_integrated }
			}
		}
		# there exists a commander on the root side to be the national hero
		OR = {
			scope:attacker = {
				owner = ROOT
			}
			scope:defender ?= {
				owner = ROOT
			}
		}
		scope:enemy_country = {
			NOR = {
				country_has_primary_culture = cu:brazilian
				country_has_primary_culture = cu:sulista
				country_has_primary_culture = cu:nordestino
				country_has_primary_culture = cu:amazonic
				country_has_primary_culture = cu:paulista
			}
		}
	}

	immediate = {
		if = {
			limit = {
				scope:attacker = {
					owner = ROOT
				}
			}
			scope:attacker = {
				save_scope_as = brazilian_national_hero
			}
		}
		else = {
			scope:defender = {
				save_scope_as = brazilian_national_hero
			}
		}
		random_list = {
			1 = {
				trigger = {
				 	AND = {
				 		NOT = { has_variable = brazilian_sulista_hero_var }
				 		NOT = { has_variable = sulistas_integrated }
				 	}
				}
				create_character = {
					ig_leader = no
					female = no
					is_general = yes
					culture = cu:sulista
					on_created = {
						set_variable = {
							name = brazilian_hero_scope
						}
					}
				}
			}
			1 = {
				trigger = {
				 	AND = {
				 		NOT = { has_variable = brazilian_paulista_hero_var }
				 		NOT = { has_variable = paulistas_integrated }
				 	}
				}
				create_character = {
					ig_leader = no
					female = no
					is_general = yes
					culture = cu:paulista
					on_created = {
						set_variable = {
							name = brazilian_hero_scope
						}
					}
				}
			}
			1 = {
				trigger = {
				 	AND = {
				 		NOT = { has_variable = brazilian_nordestino_hero_var }
				 		NOT = { has_variable = nordestinos_integrated }
				 	}
				}
				create_character = {
					ig_leader = no
					female = no
					is_general = yes
					culture = cu:nordestino
					on_created = {
						set_variable = {
							name = brazilian_hero_scope
						}
					}
				}
			}
			1 = {
				trigger = {
				 	AND = {
				 		NOT = { has_variable = brazilian_amazonic_hero_var }
				 		NOT = { has_variable = amazonics_integrated }
				 	}
		 		}
				create_character = {
					ig_leader = no
					female = no
					is_general = yes
					culture = cu:amazonic
					on_created = {
						set_variable = {
							name = brazilian_hero_scope
						}
					}
				}
			}
		}

		random_scope_character = {
			limit = { has_variable = brazilian_hero_scope }
			save_scope_as = brazilian_national_hero
			culture = {
				save_scope_as = brazilian_national_hero_culture
			}
		}
	}

	option = {
		name = culture_brazil.1.a
		default_option = yes

		custom_tooltip = culture_brazil.1.hero_clarification

		if = {
			limit = {
				scope:brazilian_national_hero = {
					culture = {
						cu:sulista = this
					}
				}
			}
			set_variable = brazilian_sulista_hero_var
		}
		else_if = {
			limit = {
				scope:brazilian_national_hero = {
					culture = {
						cu:nordestino = this
					}
				}
			}
			set_variable = brazilian_nordestino_hero_var
		}
		else_if = {
			limit = {
				scope:brazilian_national_hero = {
					culture = {
						cu:amazonic = this
					}
				}
			}
			set_variable = brazilian_amazonic_hero_var
		}
		else_if = {
			limit = {
				scope:brazilian_national_hero = {
					culture = {
						cu:paulista = this
					}
				}
			}
			set_variable = brazilian_paulista_hero_var
		}

		scope:brazilian_national_hero = {
			if = {
				limit = {
					NOR = {
						has_trait = traditionalist_commander
						has_trait = popular_commander
						has_trait = celebrity_commander
					}
				}
				add_trait = traditionalist_commander
			}
			else_if = {
				limit = {
					has_trait = traditionalist_commander
				}
				remove_trait = traditionalist_commander
				add_trait = popular_commander
			}
			else_if = {
				limit = {
					has_trait = popular_commander
				}
				remove_trait = popular_commander
				add_trait = celebrity_commander
			}
		}
	}


	option = {
		name = culture_brazil.1.b

		scope:brazilian_national_hero = {
			kill_character = yes
		}
	}
}
