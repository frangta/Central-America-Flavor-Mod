# Probabilidad de exito de una intervencion federal (.35-.40), en %.
# Scope: UCA, con scope:ca_target = estado intervenido.
# Base 50 + 10 por nivel de lealtad, +10 si el ejercito esta en el gobierno,
# -10 si la UCA esta en otra guerra, ajuste por estado (grupo 7).
ca_intervention_success_chance = {
	add = { value = 50 desc = "CA_INTERVENTION_BASE" }
	add = { value = ca_target_loyalty multiply = 10 desc = "CA_INTERVENTION_LOYALTY" }
	if = {
		limit = { ig:ig_armed_forces ?= { is_in_government = yes } }
		add = { value = 10 desc = "CA_INTERVENTION_ARMY" }
	}
	if = {
		limit = { is_at_war = yes }
		add = { value = -10 desc = "CA_INTERVENTION_AT_WAR" }
	}
	if = {
		limit = { scope:ca_target ?= { c:ELS ?= this } }
		add = { value = 20 desc = "CA_INTERVENTION_ELS" }
	}
	# F7 (grupo 7): Costa Rica -15 %; Nicaragua +-10 % (Leon liberal en el gobierno o Granada)
	if = {
		limit = { scope:ca_target ?= { c:COS ?= this } }
		add = { value = -15 desc = "CA_INTERVENTION_COS" }
	}
	if = {
		limit = {
			scope:ca_target ?= {
				c:NIC ?= this
				ig:ig_intelligentsia ?= { is_in_government = yes }
			}
		}
		add = { value = 10 desc = "CA_INTERVENTION_NIC_LEON" }
	}
	else_if = {
		limit = { scope:ca_target ?= { c:NIC ?= this } }
		add = { value = -10 desc = "CA_INTERVENTION_NIC_GRANADA" }
	}
	if = {
		limit = {
			scope:ca_target ?= { c:GUA ?= this }
			game_date >= 1839.1.1
		}
		add = { value = -10 desc = "CA_INTERVENTION_GUA_CARRERA" }
	}
	if = {
		limit = {
			scope:ca_target ?= { c:ALT ?= this }
			c:GUA ?= { has_variable = ca_role_separatist }
		}
		add = { value = -10 desc = "CA_INTERVENTION_ALT_GUATEMALA" }
	}
	min = 5
	max = 95
}

# Probabilidad de victoria federal en una guerra civil de estado (.63-.73), en %.
# Scope: UCA, con scope:ca_target = estado en guerra civil.
# Base 40 + 5 por nivel de lealtad + 10 por cada ayuda de los capitulos A/B
# (variable ca_cw_federal_help en el estado).
ca_civil_war_federal_chance = {
	add = { value = 40 desc = "CA_CIVIL_WAR_BASE" }
	add = { value = ca_target_loyalty multiply = 5 desc = "CA_CIVIL_WAR_LOYALTY" }
	if = {
		limit = { scope:ca_target ?= { has_variable = ca_cw_federal_help } }
		add = { value = scope:ca_target.var:ca_cw_federal_help multiply = 10 desc = "CA_CIVIL_WAR_CHAPTERS" }
	}
	min = 5
	max = 95
}
