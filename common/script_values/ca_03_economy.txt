# Dinero de eventos y botones escalado al PIB, con minimos bajos para las
# economias pequenas de Centroamerica en 1836. Redondeado a 10.
# Uso:  add_treasury = { value = ca_money_small multiply = -1 }
# Texto: [ScriptValue('ca_money_small')|0]
# Scope: pais.
############################################################################

ca_money_small = {
	value = gdp
	multiply = 0.0003		# 0,03 % del PIB
	divide = 10
	round = yes
	multiply = 10
	min = 30
}

ca_money_medium = {
	value = gdp
	multiply = 0.0008		# 0,08 % del PIB
	divide = 10
	round = yes
	multiply = 10
	min = 80
}

ca_money_large = {
	value = gdp
	multiply = 0.002		# 0,2 % del PIB
	divide = 10
	round = yes
	multiply = 10
	min = 200
}
