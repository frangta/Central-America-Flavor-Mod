# Generales con ideologia unitaria en la UCA.
ca_cen_unitarian_generals = {
	value = 0
	every_scope_character = {
		limit = {
			has_role = character_role_general
			has_ideology = ideology:ideology_ca_unitarian
		}
		add = 1
	}
}

ca_cen_army_general_factor = {
	value = ca_cen_unitarian_generals
	multiply = ca_cen_army_general
}

ca_cen_monthly_loyal_factor = {
	value = ca_core_states_loyal_count
	multiply = ca_cen_monthly_loyal
}
ca_cen_monthly_rebel_factor = {
	value = ca_core_states_rebel_count
	multiply = ca_cen_monthly_rebel
}
ca_cen_monthly_separated_factor = {
	value = ca_core_states_separated_count
	multiply = ca_cen_monthly_separated
}

# Estados nucleo en lealtad minima (Rebelde o Separado): fracaso por revuelta separatista.
ca_cen_core_low_loyalty_count = {
	value = ca_core_states_rebel_count
	add = ca_core_states_separated_count
}

# Presion separatista que se suma cada mes.
ca_cen_sep_pressure_monthly = {
	value = 0
	add = { value = ca_core_states_rebel_count multiply = ca_cen_sep_rebel }
	add = { value = ca_core_states_wavering_count multiply = ca_cen_sep_wavering }
	if = {
		limit = {
			OR = {
				has_law_or_variant = law_type:law_ca_central_american_unitary
				has_variable = ca_cen_law_unitary_started
			}
		}
		add = ca_cen_sep_law
	}
	if = { limit = { has_variable = frca_concordat } add = ca_cen_sep_concordat }
	if = { limit = { has_variable = frca_capital_guatemala } add = ca_cen_sep_capital }
	if = {
		limit = { any_scope_state = { turmoil >= 0.2 } }
		add = ca_cen_sep_turmoil
	}
	if = { limit = { game_date >= 1838.1.1 } add = ca_cen_sep_date }
	if = { limit = { has_variable = ca_cen_propaganda_active } add = ca_cen_sep_propaganda }
}

# Avance mensual del golpe.
ca_cen_coup_monthly = {
	value = 3
	if = { limit = { ig:ig_armed_forces ?= { is_powerful = yes } } add = 3 }
	if = { limit = { ig:ig_armed_forces ?= { ig_approval >= 5 } } add = 2 }
	if = { limit = { ig:ig_intelligentsia ?= { is_powerful = yes } } add = -3 }
	if = { limit = { ig:ig_petty_bourgeoisie ?= { is_in_government = yes } } add = -2 }
	add = ca_cen_unitarian_generals
	min = -5
	max = 12
}

# Puntos de los centralistas en la eleccion de 1838.
ca_cen_election_centralist = {
	value = ca_bloc_power_centralist
	if = {
		limit = { has_variable = ca_cen_campaign }
		add = { value = var:ca_cen_campaign multiply = ca_cen_campaign_points }
	}
}
ca_cen_election_federal = {
	value = ca_bloc_power_federal
}
