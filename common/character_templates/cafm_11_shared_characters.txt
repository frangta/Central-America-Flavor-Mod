# =====================================================================
# CAFM — PERSONAJES COMPARTIDOS ENTRE CASTS
# Mismo esquema por bloques que cafm_01 ... cafm_99. Un personaje
# compartido lleva las variables de todos sus casts; el registro
# global cafm_hc_<clave> impide que exista dos veces.
# Certeza: V verificado · P parcial · M de memoria (confirmar fechas)
# =====================================================================

cafm_manuel_quijano = {
	# Manuel Quijano y Garcia — casts COS + NIC · nivel B · certeza V
	# FUENTE: R. Fernandez Guardia, "La guerra de la Liga y la invasion de
	#         Quijano", Revista del Archivo Nacional 3 (1939), pp. 373-382.
	# NOTAS: Nacio hacia 1801 en Cartago (padre nicaraguense). Artillero
	#        espanol (Cadiz, Cartagena, La Habana, Panama), deserto en 1821;
	#        imperialista en Ochomogo (1823); capitan federal (1826). Complot
	#        liguero de 1835 -> desterrado por Carrillo. Invade Guanacaste
	#        en 1836 y Carrillo lo declara fuera de la ley (4 jul 1836).
	#        Coronel al servicio de Nicaragua: El Potrero (31 ene 1840),
	#        San Juan del Norte (1841, incidente de la HMS Tweed), deserta a
	#        Malespin (1844), toma de Leon (ene 1845), Monterredondo (14 ago 1845).
	#        En Costa Rica es un proscrito: no tiene bloque de uso para COS.
	#        Lo crean el evento ca_nicaragua_events.8 (opcion b) y
	#        ca_costa_rica_events.99 (Guanacaste independiente).

	# --- DATOS BÁSICOS ---
	first_name = "cafm_Manuel"
	last_name = "cafm_Quijano"
	historical = yes
	female = no
	birth_date = 1801.1.1			# P: "hacia 1801" segun la fuente
	culture = cu:costa_rican
	religion = rel:catholic
	home_region = STATE_COSTA_RICA

	# --- ROLES E IDENTIDAD ---
	role = general

	# --- POLÍTICA Y RIQUEZA ---
	interest_group = ig_armed_forces
	ideology = ideology_traditionalist	# ligero en 1835; en Nicaragua, del partido "servil"

	# --- ATRIBUTOS MILITARES ---
	commander_rank = commander_rank_1
	hq = region_central_america

	# --- RASGOS (TRAITS) ---
	traits = {
		ambitious
		bandit
	}

	# --- GENERACIÓN DINÁMICA DE RASGOS ---
	trait_generation = { }			# vacío: personaje histórico, sin rasgos aleatorios

	# --- EFECTOS AL CREARSE ---
	on_created = {
		set_variable = cafm_cast_cos
		set_variable = cafm_cast_nic
		set_global_variable = { name = cafm_hc_quijano value = this }
	}

	# --- BLOQUES DE APARICIÓN DINÁMICA (USAGE BLOCKS) ---
	# Solo en Nicaragua: coronel desde 1840, general hasta 1845
	commander_usage = {
		country_trigger = {
			c:NIC ?= THIS
			NOT = { has_global_variable = cafm_hc_quijano }
		}
		role = general
		earliest_usage_date = 1839.6.1
		latest_usage_date = 1846.1.1
	}
}
