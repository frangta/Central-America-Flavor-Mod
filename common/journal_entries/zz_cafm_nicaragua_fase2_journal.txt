je_ca_nic_division = {
	icon = "gfx/interface/icons/event_icons/event_skull.dds"
	group = je_group_historical_content

	activation_description = {
		always = yes
	}

	current_value = {
		value = root.var:nic_division
	}
	goal_add_value = {
		add = 70
	}
	progressbar = yes

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_ca_nic_division_status_guerra
				trigger = { var:nic_division >= ca_nic_division_guerra }
			}
			triggered_desc = {
				desc = je_ca_nic_division_status_ruptura
				trigger = { var:nic_division >= ca_nic_division_ruptura }
			}
			triggered_desc = {
				desc = je_ca_nic_division_status_tension
				trigger = { var:nic_division >= ca_nic_division_tension }
			}
			triggered_desc = {
				desc = je_ca_nic_division_status_baja
				trigger = { always = yes }
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_ca_nic_division_conclusion_managua
		color = "gold_positive"
		trigger = {
			has_variable = nic_pacto_managua
		}
		effect = {
			remove_variable = nic_division
			add_modifier = { name = nic_capital_managua years = 20 }
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		conclusion_desc = je_ca_nic_division_conclusion_guerra
		color = "blue_negative"
		trigger = {
			var:nic_division >= ca_nic_division_guerra
		}
		effect = {
			remove_variable = nic_division
			add_modifier = { name = nic_guerra_de_bandos years = 10 }
		}
	}

	on_weekly_pulse = {
		effect = {
			ca_nic_division_clamp = yes
			ca_nic_division_thresholds = yes
		}
	}

	on_monthly_pulse = {
		effect = {
			ca_nic_division_drift = yes
		}
	}

	weight = 200
	transferable = no
	can_revolution_inherit = no
}
