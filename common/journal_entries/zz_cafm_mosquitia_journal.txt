	activation_description = {
		always = yes
	}

	current_value = {
		value = root.var:mkt_protectorado
	}
	goal_add_value = {
		add = 60
	}
	progressbar = yes

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_ca_mkt_protectorado_status_oficina
				trigger = { var:mkt_protectorado >= ca_mkt_protectorado_oficina }
			}
			triggered_desc = {
				desc = je_ca_mkt_protectorado_status_tutela
				trigger = { var:mkt_protectorado >= ca_mkt_protectorado_tutela }
			}
			triggered_desc = {
				desc = je_ca_mkt_protectorado_status_consul
				trigger = { var:mkt_protectorado >= ca_mkt_protectorado_consul }
			}
			triggered_desc = {
				desc = je_ca_mkt_protectorado_status_corona
				trigger = { always = yes }
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_ca_mkt_protectorado_conclusion_corona
		color = "gold_positive"
		trigger = {
			var:mkt_protectorado <= 20
		}
		effect = {
			remove_variable = mkt_protectorado
			add_modifier = { name = mkt_corona_propia years = 20 }
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		conclusion_desc = je_ca_mkt_protectorado_conclusion_oficina
		color = "blue_negative"
		trigger = {
			var:mkt_protectorado >= ca_mkt_protectorado_oficina
		}
		effect = {
			remove_variable = mkt_protectorado
			add_modifier = { name = mkt_oficina_en_belice years = 20 }
		}
	}

	on_weekly_pulse = {
		effect = {
			ca_mkt_protectorado_clamp = yes
			ca_mkt_protectorado_thresholds = yes
		}
	}

	on_monthly_pulse = {
		effect = {
			ca_mkt_protectorado_drift = yes
		}
	}

	weight = 200
	transferable = no
	can_revolution_inherit = no
}
