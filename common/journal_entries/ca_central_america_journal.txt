	timeout = 3292

		custom_tooltip = je_frca_timeout_d2_tt

		ca_frca_journal_cleanup = yes
		trigger_event = { id = ca_central_america.122 popup = yes } # La Federacion se desvanece
		ca_frca_fire_hook = { HOOK = 5 }
	}

	on_fail = {
		set_variable = frca_outcome_d1
		ca_frca_journal_cleanup = yes
		activate_law = law_type:law_ca_caudillismo # F13 (A3): D1 trae la era de los caudillos
		trigger_event = { id = ca_central_america.13 popup = yes }
		custom_tooltip = central_america_capital_tt

		hidden_effect = {
			ig:ig_landowners ?= {
				remove_ideology = ideology_central_american_reactionary
				add_ideology = ideology_caudillismo
			}
			ig:ig_armed_forces ?= {
				remove_ideology = ideology_central_american_reactionary
			}
			ig:ig_devout ?= {
				add_ideology = ideology_paternalistic
				remove_ideology = ideology_central_american_reactionary
			}
			remove_variable = is_frca_var
		}
	}

	on_complete = {
		ca_frca_journal_cleanup = yes
		if = {
			limit = { has_variable = frca_diet_signed }
			set_variable = frca_outcome_c
			ca_frca_outcome_c = yes
		}
		else_if = {
			limit = { has_variable = frca_won_war }
			set_variable = frca_outcome_b
			ca_frca_outcome_b = yes
		}
		else = {
			set_variable = frca_outcome_a
			ca_frca_outcome_a = yes
		}
		ca_frca_fire_hook = { HOOK = 5 }
	}

	transferable = no

	invalid = {
		NOT = { exists = c:UCA }
	}
}

# CAFM F6 - Diario 2 del estado: la lucha interna (R1, opcion b).

# (Leal 75 / Vacilante 50 / Rebelde 25). La mueven los botones del diario 1, las elecciones,
# los golpes, las guerras civiles, los eventos y +-1 al mes segun quien gobierna.
# 100 -> Beats, 0 -> Falls (ca_fvs_resolve, grupo 9). Sin pulso de eventos.

	icon = "gfx/interface/icons/event_icons/event_flags/event_uca_flag.dds"
	group = je_group_historical_content
	should_be_pinned_by_default_involved = yes

	immediate = {
		if = {
			limit = { NOT = { has_variable = ca_fvs_bar } }
			set_variable = { name = ca_fvs_bar value = ca_fvs_start_value }
		}
		ca_fvs_bar_sync = yes
	}

	# F28: barra con escala fija 0-100 (antes current_value/goal_add_value, que mostraba
	# el avance desde el valor inicial, por ejemplo "-10%").
	scripted_progress_bar = ca_fvs_bar_display
	weight = 100

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_ca_fvs_status_federalists
				trigger = { var:ca_fvs_bar >= 60 }
			}
			triggered_desc = {
				desc = je_ca_fvs_status_separatists
				trigger = { var:ca_fvs_bar <= 40 }
			}
			triggered_desc = {
				desc = je_ca_fvs_status_balanced
				trigger = { always = yes }
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			ca_fvs_monthly = yes
			ca_fvs_bar_sync = yes
		}
	}

	complete = {
		OR = {
			custom_tooltip = {
				text = je_ca_fvs_complete_tt
				var:ca_fvs_bar >= 100
			}
			custom_tooltip = {
				text = je_ca_fvs_result_beats_tt
				has_variable = ca_fvs_result_beats
			}
		}
	}

	fail = {
		OR = {
			custom_tooltip = {
				text = je_ca_fvs_fail_tt
				var:ca_fvs_bar <= 0
			}
			custom_tooltip = {
				text = je_ca_fvs_result_falls_tt
				has_variable = ca_fvs_result_falls
			}
		}
	}

	on_complete = {
		ca_fvs_resolve_self = { RESULT = beats CAUSE = journal }
	}

	on_fail = {
		ca_fvs_resolve_self = { RESULT = falls CAUSE = journal }
	}

	transferable = no

	invalid = {
		OR = {
			NOT = { exists = c:UCA }
			NOT = { is_subject_of = c:UCA }
		}
	}
}

je_ca_military_reform = {
	icon = "gfx/interface/icons/law_icons/professional_army.dds"
	group = je_group_internal_affairs
	should_be_pinned_by_default_involved = yes

	# F13 (A3): lo anade el pulso anual desde 1845 con la ley de caudillismo.
	possible = {
		ca_is_ca_national_country = yes
		NOT = { has_variable = ca_military_reform_completed }
	}

	immediate = {
		if = {
			limit = { NOT = { has_variable = ca_military_bar } }
			set_variable = { name = ca_military_bar value = 0 }
		}
		set_variable = { name = ca_caudillo_coups value = 0 }
		set_variable = ca_military_reform_active_var
	}

	current_value = {
		value = root.var:ca_military_bar
	}
	goal_add_value = {
		value = 100
	}
	progressbar = yes
	weight = 100

	on_monthly_pulse = {
		effect = {
			ca_military_coup_count = yes
			ca_military_bar_change = { VALUE = ca_military_monthly_change }
		}
		events = {
			ca_military.3 # La academia militar
		}
		random_events = {
			95 = 0
			5 = ca_military.4 # El caudillo se niega a licenciar sus tropas
		}
	}

	complete = {
		custom_tooltip = {
			text = je_ca_military_reform_bar_tt
			var:ca_military_bar >= 100
		}
		OR = {
			has_law_or_variant = law_type:law_professional_army
			has_law_or_variant = law_type:law_national_militia
			has_law_or_variant = law_type:law_mass_conscription
			has_law = law_type:law_ca_guardia_nacional
		}
		has_technology_researched = general_staff
		# Ningun caudillo como gobernante
		custom_tooltip = {
			text = je_ca_military_reform_no_caudillo_ruler_tt
			NOT = {
				ruler ?= {
					OR = {
						has_ideology = ideology:ideology_central_america_caudillo
						has_ideology = ideology:ideology_ca_peasant_caudillismo
						has_ideology = ideology:ideology_ca_tropical_bonapartism
						has_ideology = ideology:ideology_ca_tropical_utopian
					}
				}
			}
		}
		# Ningun grupo caudillista en el gobierno
		custom_tooltip = {
			text = je_ca_military_reform_no_caudillo_gov_tt
			NOT = {
				any_interest_group = {
					is_in_government = yes
					OR = {
						has_ideology = ideology:ideology_ca_caudillismo_jingoist
						has_ideology = ideology:ideology_ca_autocratic_autonomist
						has_ideology = ideology:ideology_caudillismo
					}
				}
			}
		}
		ig:ig_armed_forces ?= { is_powerful = no }
	}

	on_complete = {
		set_variable = ca_military_reform_completed
		remove_variable = ca_military_reform_active_var
		trigger_event = { id = ca_military.1 popup = yes }
		add_modifier = {
			name = ca_professional_army_bonus_modifier
		}
	}

	# Fracaso: dos golpes de caudillo, o 30 anos
	fail = {
		custom_tooltip = {
			text = je_ca_military_reform_fail_tt
			var:ca_caudillo_coups >= 2
		}
	}

	timeout = 10950

	on_fail = {
		ca_military_reform_failure = yes
	}

	on_timeout = {
		ca_military_reform_failure = yes
	}

	invalid = {
		is_subject = yes
	}
}
