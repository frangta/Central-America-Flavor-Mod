je_ca_central_american_fragile_unity = { #events 1836 - 1842
    icon = "gfx/interface/icons/event_icons/waving_flag.dds"
    group = je_group_historical_content
    should_be_pinned_by_default_involved = yes
    # Enlace a los botones interactivos
    scripted_button = ca_button_centralist_discipline
    scripted_button = ca_button_federalist_support

    # Conexión con la barra de progreso definida en el diseño
    progressbar = yes
    scripted_progress_bar = ca_core_progress_bar

    is_shown_when_inactive = {
        c:UCA ?= this
    }

    possible = {
        c:UCA ?= this
    }

    immediate = {
        # Dispara el evento de selección de facción al iniciar la partida
        trigger_event = { id = ca_central_america.80 popup = yes }
    }

    complete = {
        # El diario se completa si la barra de unidad o soporte llega al máximo (100)
        scope:journal_entry = { "scripted_bar_progress(ca_core_progress_bar)" >= 100 }
    }

    on_complete = {
        # 1. FACCIÓN FEDERAL: Jugando con la UCA
        if = {
            limit = {
                c:UCA ?= this
                has_variable = ca_faction_federal
            }
            custom_tooltip = ca_federal_victory_tt
            # Impide la guerra civil y aumenta el federalismo
            add_modifier = {
                name = ca_federal_triumph_modifier
                days = 3650
            }
            trigger_event = { id = ca_central_america_victory.1 popup = yes }
        }
        
        # 2. FACCIÓN SEPARATISTA: Jugando con un estado
        if = {
            limit = {
                NOT = { c:UCA ?= this }
                has_variable = ca_faction_separatist
            }
            # Logra su independencia, sobrevive contra federalistas/centralistas
            make_independent = yes
            set_variable = recognized_state_independence
            trigger_event = { id = ca_separatist_victory.1 popup = yes }
        }

        # 3. FACCIÓN CENTRALISTA: Jugando con la UCA
        if = {
            limit = {
                c:UCA ?= this
                has_variable = ca_faction_centralist
            }
            # Gana apoyo al centralismo, baja separatismo y gana legitimidad
            add_legitimacy = 25
            change_variable = { name = ca_separatist_support add = -50 }
            trigger_event = { id = ca_centralist_victory.1 popup = yes }
        }

        # 4. FACCIÓN CENTRALISTA: Jugando con un estado
        if = {
            limit = {
                NOT = { c:UCA ?= this }
                has_variable = ca_faction_centralist
            }
            # El estado se convierte en la nueva capital/líder de la federación
            c:UCA = {
                annex = root
            }
            change_tag = UCA
            trigger_event = { id = ca_centralist_victory.2 popup = yes }
        }
    }

    fail = {
        # El diario fracasa si la barra de unidad o soporte cae a 0
        scope:journal_entry = { "scripted_bar_progress(ca_core_progress_bar)" <= 0 }
    }

    on_fail = {
        # 1. FACCIÓN FEDERAL: Jugando con la UCA (Ruta Principal Histórica)
        if = {
            limit = {
                c:UCA ?= this
                has_variable = ca_faction_federal
            }
            # Dispara la Segunda Guerra Civil Histórica
            trigger_event = { id = ca_central_america.42 popup = yes }
        }

        # 2. FACCIÓN SEPARATISTA: Jugando con un estado
        if = {
            limit = {
                NOT = { c:UCA ?= this }
                has_variable = ca_faction_separatist
            }
            # Es anexionado forzosamente por los federalistas o centralistas victoriosos
            trigger_event = { id = ca_separatist_defeat.1 popup = yes }
        }

        # 3. FACCIÓN CENTRALISTA: Jugando con la UCA
        if = {
            limit = {
                c:UCA ?= this
                has_variable = ca_faction_centralist
            }
            # Consecuencia: Se pierde apoyo centralista, aumenta separatismo, guerra de 3 bandos
            change_variable = { name = ca_separatist_support add = 50 }
            trigger_event = { id = ca_centralist_defeat.1 popup = yes } # Evento de guerra de 3 bandos
        }

        # 4. FACCIÓN CENTRALISTA: Jugando con un estado
        if = {
            limit = {
                NOT = { c:UCA ?= this }
                has_variable = ca_faction_centralist
            }
            # Consecuencia: Balcanización del estado en microestados
            trigger_event = { id = ca_state_balkanization.1 popup = yes }
        }
    }

    invalid = {
        # Si la nación es destruida o anexada por un poder externo
        NOT = { exists = c:UCA }
    }

    on_invalid = {
        # Limpieza de variables para optimizar el rendimiento del juego
        remove_variable = ca_federalist_support
        remove_variable = ca_centralist_support
        remove_variable = ca_separatist_support
    }

    timeout = 0

    modifiers_while_active = {
        # Se puede añadir un modificador de inestabilidad global aquí si se desea
    }

    on_weekly_pulse = {
        # Espacio reservado para rastrear guerras civiles a tres bandos
    }

    on_monthly_pulse = {
        # Sistema de probabilidad para la IA y eventos dinámicos
        random_events = {
            85 = 0 # Vía histórica federalista (Sin evento adverso)
            15 = ca_central_america.centralist_coup # Intento de golpe centralista
        }
    }

    on_yearly_pulse = {
        # Espacio para advertencias anuales sobre pérdida de territorios
    }

    weight = 100
    transferable = no
    can_revolution_inherit = yes
    can_deactivate = no
    should_be_pinned_by_default = yes
}

je_ca_military_reform = {
    icon = "gfx/interface/icons/law_icons/professional_army.dds"
    group = je_group_internal_affairs #je_group_military
    should_be_pinned_by_default_involved = yes
    possible = {
        OR = {
            country_has_primary_culture = cu:central_american
            country_has_primary_culture = cu:costa_rican
            country_has_primary_culture = cu:guatemalan
            country_has_primary_culture = cu:el_salvadorian
            country_has_primary_culture = cu:honduran
            country_has_primary_culture = cu:nicaraguan
        }
        has_law_or_variant = law_type:law_ca_caudillismo
        NOT = { has_variable = ca_military_reform_completed }
    }

    complete = { 
        # Leyes e hitos militares requeridos
        OR = {
            has_law_or_variant = law_type:law_professional_army
            has_law_or_variant = law_type:law_national_militia
            has_law_or_variant = law_type:law_mass_conscription
        }
        has_technology_researched = general_staff
        num_regular_battalions >= 10
        
        # 1. ningun personaje (generales/gobernantes/agitadores) tiene ideologías caudillistas
        NOT = {
            any_scope_character = {
                OR = {
                    has_ideology = ideology:ideology_central_america_caudillo
                    has_ideology = ideology:ideology_ca_peasant_caudillismo
                    has_ideology = ideology:ideology_ca_tropical_bonapartism
                    has_ideology = ideology:ideology_ca_tropical_utopian
                }
            }
        }

        # 2. ningun ig tiene estas ideologias centralistas/autonomistas
        NOT = {
            any_interest_group = {
                OR = {
                    has_ideology = ideology:ideology_ca_caudillismo_jingoist
                    has_ideology = ideology:ideology_ca_autocratic_autonomist
					has_ideology = ideology:ideology_caudillismo
                }
            }
        }

        # 3. Las Fuerzas Armadas están fuera del gobierno, no son poderosas y aprueban al gobierno
        ig:ig_armed_forces = {
            is_in_government = no
            is_powerful = no
            ig_approval >= 15 
        }
    }

    on_complete = {
        set_variable = ca_military_reform_completed
        trigger_event = { id = ca_military.1 }
        
        # Modificador permanente al lograr la profesionalización
        add_modifier = {
            name = ca_professional_army_bonus_modifier
        }
    }
    immediate = {
    # Guardamos a las Fuerzas Armadas en un "scope" para referenciarlas en futuros eventos
    ig:ig_armed_forces = {
        save_scope_as = ca_armed_forces_ig
    }
    # Establecemos una variable para indicar que el proceso de reforma está activo
    set_variable = ca_military_reform_active_var
     }

    fail = {
    # El diario fracasa catastróficamente si no se logra eliminar a todos los caudillos del pais 
    
    }

    on_fail = {
    # Limpieza de variables para mantener el código optimizado
    remove_variable = ca_military_reform_active_var

    # Disparamos el evento de fracaso (deberás definir ca_military.2 en tus eventos)
    trigger_event = { id = ca_military.2 popup = yes }

    # Consecuencia: Las Fuerzas Armadas ganan muchísimo poder y se atrincheran
    ig:ig_armed_forces = {
        add_modifier = {

        }
