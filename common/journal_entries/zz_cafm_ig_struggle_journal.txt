je_ca_ig_struggle_armed_forces = {
	icon = "gfx/interface/icons/event_icons/event_protest.dds"
	group = je_group_internal_affairs

	activation_description = {
		always = yes
	}

	should_be_pinned_by_default_involved = yes

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_ca_ig_struggle_status_to_federal
				trigger = { has_variable = ca_struggle_armed_forces_to var:ca_struggle_armed_forces_to = flag:federal }
			}
			triggered_desc = {
				desc = je_ca_ig_struggle_status_to_centralist
				trigger = { has_variable = ca_struggle_armed_forces_to var:ca_struggle_armed_forces_to = flag:centralist }
			}
			triggered_desc = {
				desc = je_ca_ig_struggle_status_to_separatist
				trigger = { always = yes }
			}
		}
	}

	scripted_progress_bar = ca_ig_struggle_armed_forces_bar
	weight = 50

	on_monthly_pulse = {
		effect = {
			ca_struggle_monthly = { IG = armed_forces }
		}
	}

	# El IG ya no existe, o el diario de la UCA termino.
	outcome = {
		conclusion_desc = je_standard_failure_conclusion_desc
		flags = { hidden invalid_outcome }
		trigger = {
			OR = {
				NOT = { exists = ig:ig_armed_forces }
				NOT = { has_journal_entry = je_historic_central_america_falls_apart }
			}
		}
		effect = {
			ca_struggle_end_quiet = { IG = armed_forces }
		}
	}

	# El IG se pasa al bando rival.
	outcome = {
		header = je_ca_ig_struggle_change_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_ca_ig_struggle_change_conclusion
		color = "gold_positive"
		trigger = {
			custom_tooltip = {
				text = je_ca_ig_struggle_min_time_tt
				NOT = { has_variable = ca_struggle_armed_forces_lock }
			}
			custom_tooltip = {
				text = je_ca_ig_struggle_change_tt
				scope:journal_entry = { "scripted_bar_progress(ca_ig_struggle_armed_forces_bar)" >= 100 }
			}
		}
		effect = {
			ca_struggle_end_change = { IG = armed_forces }
		}
	}

	# El IG conserva su bando.
	outcome = {
		header = je_ca_ig_struggle_keep_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_ca_ig_struggle_keep_conclusion
		color = "blue_negative"
		trigger = {
			custom_tooltip = {
				text = je_ca_ig_struggle_min_time_tt
				NOT = { has_variable = ca_struggle_armed_forces_lock }
			}
			custom_tooltip = {
				text = je_ca_ig_struggle_keep_tt
				scope:journal_entry = { "scripted_bar_progress(ca_ig_struggle_armed_forces_bar)" <= 0 }
			}
		}
		effect = {
			ca_struggle_end_keep = { IG = armed_forces }
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		conclusion_desc = je_ca_ig_struggle_timeout_conclusion
		flags = { timeout }
		effect = {
			custom_tooltip = je_ca_ig_struggle_timeout_tt
			ca_struggle_end_quiet = { IG = armed_forces }
		}
	}

	timeout = 730

	transferable = no
}
