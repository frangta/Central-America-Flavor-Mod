je_ca_agro_export_boom = {
    icon = "gfx/interface/icons/goods_icons/coffee.dds"
    group = je_group_economic_development
    should_be_pinned_by_default = yes

    # ACTIVACIÓN
    possible = {
        # Cualquier país de cultura centroamericana
        OR = {
            country_has_primary_culture = cu:central_american
            country_has_primary_culture = cu:costa_rican
            country_has_primary_culture = cu:guatemalan
            country_has_primary_culture = cu:el_salvadorian
            country_has_primary_culture = cu:honduran
            country_has_primary_culture = cu:nicaraguan
        }
        # Que no sea una gran potencia aún
        rank_value < 50
    }

    # CONDICIONES DE COMPLETADO (Objetivos Claros)
    complete = {
        # 1. Tener una base productiva fuerte
        # (Suma de niveles de café + fruta + azúcar > 30)
        calc_building_level_count:building_coffee_plantation >= 15
        calc_building_level_count:building_fruit_plantation >= 10
        
        # 2. Infraestructura para exportar
        calc_building_level_count:building_port >= 5
        
        # 3. Política Comercial: NO Aislacionismo
        NOT = { has_law = law_type:law_isolationism }
    }

    # RECOMPENSA
    on_complete = {
        # Evento de celebración
        trigger_event = { id = ca_economy.1 }
        
        # Modificador permanente de eficiencia comercial
        add_modifier = {
            name = ca_agro_export_miracle_modifier
            months = 180 # 15 años de bonanza
        }
    }
}

# 2. MILITAR: EL FIN DE LOS CAUDILLOS
# Objetivo: Crear un ejército nacional organizado.
# Referencia: Similar a las reformas militares europeas.
# =========================================================
je_ca_military_reform = {
    icon = "gfx/interface/icons/law_icons/professional_army.dds"
    group = je_group_military
    
    possible = {
        OR = {
            country_has_primary_culture = cu:central_american
            country_has_primary_culture = cu:costa_rican
            country_has_primary_culture = cu:guatemalan
            country_has_primary_culture = cu:el_salvadorian
            country_has_primary_culture = cu:honduran
            country_has_primary_culture = cu:nicaraguan
        }
        # Tienen leyes arcaicas
        has_law = law_type:law_peasant_levies
    }

    complete = {
        # 1. Eliminar las levas campesinas
        OR = {
            has_law = law_type:law_professional_army
            has_law = law_type:law_national_militia
            has_law = law_type:law_mass_conscription
        }
        
        # 2. Tener una academia militar
        # (Asumiendo que tienes un edificio de universidad renombrado o usas la tech)
        has_technology_researched = general_staff
        
        # 3. Tener un ejército decente (para el tamaño de la región)
        # > 10 batallones regulares
        calc_regular_battalion_count >= 10
        
        # 4. Los militares deben ser leales (no golpistas)
        ig:ig_armed_forces = {
            approval >= 5
        }
    }

    on_complete = {
        trigger_event = { id = ca_military.1 }
        add_modifier = {
            name = ca_professional_army_bonus_modifier
        }
    }
}

je_ca_enlightenment = {
    icon = "gfx/interface/icons/jomini/literacy_icon.dds"
    group = je_group_internal_affairs
    
    possible = {
        OR = {
            country_has_primary_culture = cu:central_american
            country_has_primary_culture = cu:costa_rican
            country_has_primary_culture = cu:guatemalan
            country_has_primary_culture = cu:el_salvadorian
            country_has_primary_culture = cu:honduran
            country_has_primary_culture = cu:nicaraguan
        }
        literacy_rate < 0.40 # Solo visible si la alfabetización es baja
    }

    complete = {
        # 1. Meta de Alfabetización (Alta para la región en 1836)
        literacy_rate >= 0.40
        
        # 2. Institución Educativa activa
        has_institution = institution_schools
        institution_investment_level:institution_schools >= 3
        
        # 3. Tener al menos una Universidad
        calc_building_level_count:building_university >= 1
        
        # 4. Intelectuales influyentes
        ig:ig_intelligentsia = {
            is_powerful = yes
            is_in_government = yes
        }
    }

    on_complete = {
        trigger_event = { id = ca_culture.1 }
        add_modifier = {
            name = ca_enlightenment_era_modifier
        }
    }
}
