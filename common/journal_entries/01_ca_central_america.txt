je_ca_agro_export_boom = {
    icon = "gfx/interface/icons/goods_icons/coffee.dds"
    group = je_group_economic_development
    should_be_pinned_by_default = yes

    possible = {
        OR = {
            country_has_primary_culture = cu:central_american
            country_has_primary_culture = cu:costa_rican
            country_has_primary_culture = cu:guatemalan
            country_has_primary_culture = cu:el_salvadorian
            country_has_primary_culture = cu:honduran
            country_has_primary_culture = cu:nicaraguan
        }
        # CORRECCIÓN: En Vic3, el rango se evalúa con 'country_rank' (mayor, menor, etc.)
        # Si quieres que no sea una Gran Potencia, usamos esto:
        NOT = { country_rank = rank_value:great_power }
        # Evitamos que se dispare si ya se completó
        NOT = { has_variable = ca_agro_export_miracle_completed }
    }

    complete = {
        # CORRECCIÓN: La sintaxis correcta para sumar niveles de edificios en todo el país
        # es usar 'building_levels'. calc_building_level_count no es nativo.
        building_levels = {
            type = building_coffee_plantation
            value >= 15
        }
        building_levels = {
            type = building_fruit_plantation
            value >= 10
        }
        building_levels = {
            type = building_port
            value >= 5
        }
        NOT = { has_law = law_type:law_isolationism }
    }

    on_complete = {
        set_variable = ca_agro_export_miracle_completed
        trigger_event = { id = ca_economy.1 }
        add_modifier = {
            name = ca_agro_export_miracle_modifier
            months = 180 # 15 años de bonanza
        }
    }
}
