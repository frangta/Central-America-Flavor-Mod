		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = "gold_positive"
		trigger = {
			OR = {
				# La Republica unitaria
				AND = {
					custom_tooltip = { text = je_frca_outcome_unitary_tt always = yes }
					custom_tooltip = {
						text = je_ca_centralist_unity_full_tt

					}

						text = je_frca_unitary_phase_tt

						ca_bloc_power_centralist >= ca_bloc_unitary_power_min

					custom_tooltip = {

				}

					custom_tooltip = { text = je_frca_outcome_b_tt always = yes }

					custom_tooltip = {
						text = uca_won_war_tt
						has_variable = frca_won_war
					}
					custom_tooltip = {
						text = je_frca_core_regions_tt
						ca_frca_holds_core_regions = yes
					}

						text = je_ca_centralist_unity_b_tt
						scope:journal_entry = { "scripted_bar_progress(ca_frca_unity_bar)" >= ca_cen_unity_threshold_b }

						text = je_frca_fronts_lost_max_2_tt

					}
				}
			}
		}

			ca_frca_journal_cleanup = yes
			if = {

					ca_cen_separatist_revolt = yes
				}
			}
		}
		effect = {
			if = {
				limit = { ca_fvc_liberal_revolution = yes }
				set_variable = frca_liberal_revolution
				trigger_event = { id = cafm_uca_route.6 popup = yes }

				limit = { ca_cen_separatist_revolt = yes }
				set_variable = frca_separatist_revolt

				}
				ig:ig_armed_forces ?= {
					remove_ideology = ideology_central_american_reactionary
					remove_ideology = ideology_central_american_reactionary_2
				}
				ig:ig_devout ?= {
					add_ideology = ideology_paternalistic
					remove_ideology = ideology_central_american_reactionary
				}
				remove_variable = is_frca_var
			}
		}
	}

		header = JOURNAL_ENTRY_TIMEOUT

			custom_tooltip = { text = je_ca_centralist_deadline_tt game_date >= 1845.1.8 }

			custom_tooltip = je_frca_timeout_d2_tt

		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC

		effect = {
			set_variable = frca_outcome_d2
			ca_frca_journal_cleanup = yes
			trigger_event = { id = ca_central_america.122 popup = yes }
		}
	}
