# =========================================================================
# DIARIO MADRE: El Estado Libre de Costa Rica
# =========================================================================
je_ca_historic_situation_in_costarica = {
	icon = "gfx/interface/icons/event_icons/event_flags/event_uca_flag.dds"
	group = je_group_historical_content

	activation_description = {
		always = yes
	}

	scripted_progress_bar = ca_cos_soberania_bar

	# Mapa del diario: el istmo segun su inestabilidad (gris = fuera del istmo)
	map_color = {
		lerp = {
			min_color = "grey"
			max_color = "red"
			factor = ca_jemap_isthmus_unrest
		}
	}
	map_tooltip = {
		if = {
			limit = { region = sr:region_central_america }
			custom_tooltip_no_bullet = CA_JEMAP_ISTHMUS_TT
		}
		else = {
			custom_tooltip_no_bullet = CA_JEMAP_OUTSIDE_TT
		}
	}

	#CAFM FIX

	immediate = {
		set_variable = ca_cos_madre_activa
	}

	# Planificador: abre cada capitulo en su ventana de fechas
	# Hitos sueltos del diario madre (p. ej. la Constituyente del 14 nov 1838)
	on_weekly_pulse = {
		effect = {
			ca_cos_madre_weekly_effect = yes
		}
	}

	on_monthly_pulse = {
		effect = {
			ca_cos_chapter_scheduler_effect = yes
			ca_cos_consequences_monthly = yes # F15 (E3)
		}
		# CAFM fix: antes estaban en on_yearly_pulse_country (solo el 1 de enero)
		events = {
			ca_costa_rica_events.2
			ca_costa_rica_events.3
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_ca_historic_situation_in_costarica_status_federal
				trigger = { has_variable = ca_cos_ruta_federal }
			}
			triggered_desc = {
				desc = je_ca_historic_situation_in_costarica_status_separatista
				trigger = { has_variable = ca_cos_ruta_separatista }
			}
			triggered_desc = {
				desc = je_ca_historic_situation_in_costarica_status_inicio
				trigger = { always = yes }
			}
		}
	}

	outcome = {
		conclusion_desc = je_standard_failure_conclusion_desc
		flags = { hidden invalid_outcome }
		trigger = {
			NOT = { c:COS ?= this }
		}
		effect = {
			ca_cos_close_mother_effect = yes
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = "gold_positive"
		trigger = {
			OR = {
				# Ruta separatista (Carrillo): la amenaza de Morazan quedo resuelta
				custom_tooltip = {
					text = je_ca_historic_situation_in_costarica_goal_carrillo
					has_variable = ca_cos_cap_d_exito
				}
				# Ruta federal: Costa Rica sigue en la Federacion con un gobierno estable
				custom_tooltip = {
					text = je_ca_historic_situation_in_costarica_goal_federal
					has_variable = ca_cos_ruta_federal
					game_date >= 1840.1.1
					scope:journal_entry = {
						scripted_bar_progress = { name = ca_cos_soberania_bar value >= 60 }
					}
				}
			}
		}
		effect = {
			ca_cos_close_mother_effect = yes
			trigger_event = { id = ca_costa_rica_events.300 popup = yes }
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		conclusion_desc = je_standard_failure_conclusion_desc
		color = "blue_negative"
		trigger = {
			OR = {
				custom_tooltip = {
					text = je_ca_historic_situation_in_costarica_fail_colapso
					scope:journal_entry = {
						scripted_bar_progress = { name = ca_cos_soberania_bar value <= 0 }
					}
				}
				custom_tooltip = {
					text = je_ca_historic_situation_in_costarica_fail_morazan
					has_variable = ca_cos_morazan_victorioso
				}
			}
		}
		effect = {
			ca_cos_close_mother_effect = yes
			# Fracaso gradual: la barra decide cuan grave es la caida
			if = {
				limit = {
					scope:journal_entry = {
						scripted_bar_progress = { name = ca_cos_soberania_bar value <= 25 }
					}
				}
				# Fracaso grave: la Liga renace
				trigger_event = { id = ca_costa_rica_events.310 popup = yes }
			}
			else = {
				# Fracaso leve: Morazan gobierna Costa Rica
				trigger_event = { id = ca_costa_rica_events.320 popup = yes }
			}
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		flags = { timeout }
		effect = {
			ca_cos_close_mother_effect = yes
			# La era de Carrillo termina: empieza la Republica de los cafetaleros
			if = {
				limit = {
					game_date < 1850.6.1
					NOT = { has_journal_entry = je_ca_cos_republica }
				}
				add_journal_entry = { type = je_ca_cos_republica }
			}
		}
	}




	# Sobrevive a la caida de la UCA (antes se invalidaba al desaparecer la UCA)

	# Fin de la era de Carrillo: ~7 anos desde 1836
	timeout = 2555

	weight = 200
	transferable = no
	can_revolution_inherit = no
}


# =========================================================================
# CAPITULO A: La Pacificacion del Norte (junio - diciembre 1836)
# Usa la variable ca_pacificacion_progress, igual que los eventos 5-8 y 81,
# asi que esos eventos funcionan sin cambios.
# =========================================================================
je_ca_pacificacion_norte = {
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	group = je_group_historical_content

	activation_description = {
		always = yes
	}

	# Mapa de la frontera norte (gui/ca_pacificacion_map.gui)
	widget = {
		gui = "gui/ca_pacificacion_map.gui"
		name = "ca_pacificacion_map_widget"
		container = "custom_widget_container_1"
	}

	progressbar = yes
	current_value = {
		value = 0
		if = {
			limit = { has_variable = ca_pacificacion_progress }
			add = var:ca_pacificacion_progress
		}
	}
	goal_add_value = {
		add = 100
	}

	immediate = {
		set_variable = ca_crisis_guanacaste_active
		if = {
			limit = { NOT = { has_variable = ca_pacificacion_progress } }
			set_variable = { name = ca_pacificacion_progress value = 0 }
		}
	}

	# Las etapas historicas salen una por semana, en orden (ver ca_cos_pacificacion_beats_effect)
	on_weekly_pulse = {
		effect = {
			ca_cos_pacificacion_beats_effect = yes
		}
	}

	outcome = {
		conclusion_desc = je_standard_failure_conclusion_desc
		flags = { hidden invalid_outcome }
		trigger = {
			NOT = { ca_cos_capitulo_valido_trigger = yes }
		}
		effect = {
			remove_variable = ca_crisis_guanacaste_active
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = "gold_positive"
		trigger = {
			scope:journal_entry = { is_goal_complete = yes }
		}
		effect = {
			remove_variable = ca_crisis_guanacaste_active
			set_variable = ca_cos_cap_a_exito
			ca_cos_change_soberania_effect = { VALUE = 20 }
			# Fase 3 en orden historico: ciudad heroica (3 sep) -> Informe Jimenez (10 sep) -> Maniobra maestra
			trigger_event = { id = ca_costa_rica_events.10 days = 20 }
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		conclusion_desc = je_standard_failure_conclusion_desc
		color = "blue_negative"
		trigger = {
			game_date >= 1837.1.1
		}
		effect = {
			remove_variable = ca_crisis_guanacaste_active
			set_variable = ca_cos_guanacaste_perdido
			ca_cos_change_soberania_effect = { VALUE = -30 }
			trigger_event = { id = ca_costa_rica_events.99 popup = yes }
		}
	}





	weight = 150
	transferable = no
	can_revolution_inherit = no
}

# CAPITULO B: La cuestion de Bocas del Toro (1836 - 1837)
# Sin barra: solo cuenta el desenlace. Puede correr en paralelo al A.
# =========================================================================
je_ca_cos_bocas_del_toro = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	group = je_group_historical_content

	activation_description = {
		always = yes
	}

	# Las etapas historicas salen por semana y en orden (ver ca_cos_bocas_beats_effect)
	on_weekly_pulse = {
		effect = {
			ca_cos_bocas_beats_effect = yes
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_ca_cos_bocas_del_toro_status_perdido
				trigger = { has_variable = ca_bocas_del_toro_lost }
			}
			triggered_desc = {
				desc = je_ca_cos_bocas_del_toro_status_defendida
				trigger = { has_variable = ca_cos_bocas_defendida }
			}
			triggered_desc = {
				desc = je_ca_cos_bocas_del_toro_status_reclamo
				trigger = { has_variable = ca_bocas_del_toro_claimed }
			}
			triggered_desc = {
				desc = je_ca_cos_bocas_del_toro_status_inicio
				trigger = { always = yes }
			}
		}
	}

	outcome = {
		conclusion_desc = je_standard_failure_conclusion_desc
		flags = { hidden invalid_outcome }
		trigger = {
			NOT = { ca_cos_capitulo_valido_trigger = yes }
		}
		effect = { }
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = "gold_positive"
		trigger = {
			OR = {
				has_variable = ca_epifania_tragica          # 18.a: la Federacion calla
				has_variable = ca_cos_bocas_apoyo_federal   # 18.b: la Federacion va a la guerra
				has_variable = ca_cos_bocas_rechazo_temprano # 14.b: la UCA no financia a Galindo
				has_variable = ca_cos_bocas_renuncia        # 13.b / 162.c: Costa Rica renuncia
				has_variable = ca_cos_bocas_reclamo_propio  # 162.b: Carrillo responde solo
				has_variable = ca_cos_bocas_defendida       # 161.b: Paredes se retira
			}
		}
		effect = {
			if = {
				limit = { has_variable = ca_epifania_tragica }
				# La humillacion alimenta el orgullo de San Jose frente a la Federacion
				ca_cos_change_soberania_effect = { VALUE = 10 }
			}
			else_if = {
				limit = { has_variable = ca_cos_bocas_apoyo_federal }
				# San Jose queda en deuda con la Federacion
				ca_cos_change_soberania_effect = { VALUE = -10 }
			}
			else_if = {
				limit = { has_variable = ca_cos_bocas_defendida }
				# El distrito resistio: San Jose sale reforzada
				ca_cos_change_soberania_effect = { VALUE = 15 }
			}
			else_if = {
				limit = { has_variable = ca_cos_bocas_reclamo_propio }
				# Carrillo actua sin la Federacion
				ca_cos_change_soberania_effect = { VALUE = 10 }
			}
			else_if = {
				limit = { has_variable = ca_cos_bocas_renuncia }
				ca_cos_change_soberania_effect = { VALUE = -5 }
			}
			else = {
				ca_cos_change_soberania_effect = { VALUE = 5 }
			}
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		flags = { timeout }
		effect = { }
	}


	# Si la cadena se corta, el capitulo se cierra solo a inicios de 1838
	timeout = 700


	weight = 120
	transferable = no
	can_revolution_inherit = no
}

# =========================================================================
# CAPITULO E: La Republica de los cafetaleros (1842 - 1850)
# Lo abren .351 (Alfaro, tras la caida de Morazan), .300 (Estado
# consolidado), .361 (unidad por las armas) o el fin del diario madre.
# Fuentes: TSE (2008); Corella Ovares, Revista de Historia (UNA) 80 (2019).
# =========================================================================
je_ca_cos_republica = {
	icon = "gfx/interface/icons/event_icons/event_election.dds"
	group = je_group_historical_content

	activation_description = {
		always = yes
	}

	#CAFM FIX

	# Las etapas historicas salen por semana y en orden (ver ca_cos_republica_beats_effect)
	on_weekly_pulse = {
		effect = {
			ca_cos_republica_beats_effect = yes
		}
	}

	outcome = {
		conclusion_desc = je_standard_failure_conclusion_desc
		flags = { hidden invalid_outcome }
		trigger = {
			NOT = { c:COS ?= this }
		}
		effect = { }
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_standard_completion_conclusion_desc
		color = "gold_positive"
		trigger = {
			has_variable = ca_cos_quiros_1850_fired
		}
		effect = { }
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		flags = { timeout }
		effect = {
			set_variable = ca_cos_republica_cerrada
		}
	}

	timeout = 3650


	weight = 150
	transferable = no
	can_revolution_inherit = no
}
