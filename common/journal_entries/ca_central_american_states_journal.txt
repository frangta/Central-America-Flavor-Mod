			4 = ca_talamanca_events.8
			4 = ca_talamanca_events.9
			4 = ca_talamanca_events.10
		}

		custom_tooltip = {
			text = je_ca_talamanca_complete_tt

		}
	}

	# CAFM Fase 1 (Honduras): fechas exactas del motín y del fin del mandato
	on_weekly_pulse = {
		effect = {
			if = {
				limit = {
					has_variable = hon_e2_fired
					NOT = { has_variable = hon_e6_fired }
					game_date < 1837.6.1
					OR = {
						game_date >= 1836.12.24
						AND = {
							NOT = { has_variable = hon_historic_mode }
							game_date >= 1836.6.1
							has_variable = hon_caudillo
							var:hon_caudillo >= 60
						}
					}
				}
				trigger_event = { id = ca_honduras_events.6 popup = yes }
			}
			else_if = {
				limit = {
					has_variable = hon_e2_fired
					NOT = { has_variable = hon_e7_fired }
					game_date < 1837.6.1
					OR = {
						game_date >= 1836.12.31
						AND = {
							NOT = { has_variable = hon_historic_mode }
							has_variable = hon_e6_fired
						}
					}
				}
				trigger_event = { id = ca_honduras_events.7 popup = yes }
			}
		}

je_ca_soconusco = {
	# CAFM F14 (D3): "El Soconusco". Diario de CIP (jugador o IA) desde el primer dia.
	# Barra "Centroamerica <-> Mexico" 0-100 (ca_soconusco_bar), movida por las culturas y la lealtad.
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	group = je_group_historical_content
	should_be_pinned_by_default_involved = yes

	immediate = {
		if = {
			limit = { NOT = { has_variable = ca_soconusco_bar } }
			set_variable = { name = ca_soconusco_bar value = 50 }
		}
	}

	current_value = {
		value = root.var:ca_soconusco_bar
	}
	goal_add_value = {
		value = 100
	}
	progressbar = yes
	weight = 100

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_ca_soconusco_status_ca
				trigger = { var:ca_soconusco_bar >= 60 }
			}
			triggered_desc = {
				desc = je_ca_soconusco_status_mex
				trigger = { var:ca_soconusco_bar <= 40 }
			}
			triggered_desc = {
				desc = je_ca_soconusco_status_neutral
				trigger = { always = yes }
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			ca_soconusco_change = { VALUE = ca_soconusco_monthly_change }
		}
		events = {
			ca_los_altos_soconusco_events.1
			ca_los_altos_soconusco_events.2
			ca_los_altos_soconusco_events.3
			ca_los_altos_soconusco_events.4
		}
		random_events = {
			100 = 0
			5 = ca_los_altos_soconusco_events.6
			5 = ca_los_altos_soconusco_events.7
			5 = ca_los_altos_soconusco_events.8
			5 = ca_los_altos_soconusco_events.9
			5 = ca_los_altos_soconusco_events.10
		}
	}

	complete = {
		custom_tooltip = {
			text = je_ca_soconusco_complete_tt
			var:ca_soconusco_bar >= 100
		}
		custom_tooltip = {
			text = je_ca_soconusco_culture_tt
			"culture_percent_country(cu:central_american)" >= "culture_percent_country(cu:mexican)"
		}
	}

	fail = {
		OR = {
			custom_tooltip = {
				text = je_ca_soconusco_fail_tt
				var:ca_soconusco_bar <= 0
			}
			custom_tooltip = {
				text = je_ca_soconusco_mexican_tt
				"culture_percent_country(cu:mexican)" >= 0.45
			}
			has_variable = ca_soconusco_irrevocable
		}
	}

	on_complete = {
		set_variable = ca_soconusco_result_central
		trigger_event = { id = ca_central_america.5 popup = yes }
	}

	on_fail = {
		set_variable = ca_soconusco_result_mexico
		trigger_event = { id = ca_central_america.5 popup = yes }
	}

	invalid = {
		NOT = { c:CIP ?= this }
	}

	transferable = no
}
