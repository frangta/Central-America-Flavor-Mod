# ============================================================
# CAFM - La Deuda Inglesa (empréstito de Londres de la década de 1820)
# Basado en je_haitian_debt (vanilla)
#
# Lógica:
#  - UCA arranca con la deuda completa (tramo 4).
#  - Si UCA deja de existir, cada sucesor hereda su parte:
#      GUA = tramo 3 | ELS, HON, NIC = tramo 2 | COS = tramo 1
#    (reparto aproximado por población; ajústalo a tu documento de diseño)
#  - Los tramos son modificadores sobre el propio diario (se propagan al país),
#    igual que modifier_french_independence_payments_X en el diario de Haití.
#  - Se completa al llegar a 0 tramos; falla si el país repudia la deuda.
#  - Acreedor: GBR (bonistas de la City). No recibe ingreso continuo porque
#    históricamente los acreedores eran privados, no el Tesoro británico.
# ============================================================

je_ca_british_debt = {
	icon = "gfx/interface/icons/event_icons/bicorne_hat.dds"

	group = je_group_historical_content

	is_shown_in_lobby = {
		c:UCA ?= THIS
	}

	is_shown_when_inactive = {
		NOT = { has_variable = ca_british_debt_closed }
		OR = {
			# La Federación, mientras la deuda no se haya repartido
			AND = {
				c:UCA ?= THIS
				NOT = { has_global_variable = ca_british_debt_divided }
			}
			# Los sucesores, solo si la Federación ya no existe
			AND = {
				NOT = { exists = c:UCA }
				OR = {
					c:GUA ?= THIS
					c:ELS ?= THIS
					c:HON ?= THIS
					c:NIC ?= THIS
					c:COS ?= THIS
				}
			}
		}
	}

	possible = {
		exists = c:GBR
		NOT = { is_subject_of = c:GBR }
	}

	scripted_button = je_ca_british_debt_negotiation_button
	scripted_button = je_ca_british_debt_payment_button
	scripted_button = je_ca_british_debt_repudiate_button

	immediate = {
		c:GBR ?= {
			save_scope_as = britain_scope
		}
		set_variable = ca_british_debt_initialized

		# Tramo inicial según quién hereda la deuda
		if = {
			limit = { c:UCA ?= THIS }
			scope:journal_entry = {
				add_modifier = { name = modifier_ca_british_debt_4 months = -1 }
			}
		}
		else_if = {
			limit = { c:GUA ?= THIS }
			scope:journal_entry = {
				add_modifier = { name = modifier_ca_british_debt_3 months = -1 }
			}
		}
		else_if = {
			limit = {
				OR = {
					c:ELS ?= THIS
					c:HON ?= THIS
					c:NIC ?= THIS
				}
			}
			scope:journal_entry = {
				add_modifier = { name = modifier_ca_british_debt_2 months = -1 }
			}
		}
		else = {
			# COS
			scope:journal_entry = {
				add_modifier = { name = modifier_ca_british_debt_1 months = -1 }
			}
		}

		# Una vez repartida, la Federación no vuelve a cargar la deuda completa si se reunifica
		if = {
			limit = { NOT = { c:UCA ?= THIS } }
			set_global_variable = ca_british_debt_divided
		}
	}

	complete = {
		# Guarda: evita que se complete antes de que immediate ponga el primer tramo
		has_variable = ca_british_debt_initialized
		NOT = { has_modifier = modifier_ca_british_debt_repudiated }
		custom_tooltip = {
			text = je_ca_british_debt_paid_tt
			scope:journal_entry = {
				NOR = {
					has_modifier = modifier_ca_british_debt_4
					has_modifier = modifier_ca_british_debt_3
					has_modifier = modifier_ca_british_debt_2
					has_modifier = modifier_ca_british_debt_1
				}
			}
		}
	}

	on_complete = {
		set_variable = ca_british_debt_closed
		save_scope_as = ca_debtor_scope
		trigger_event = { id = ca_british_debt.1 popup = yes }
		c:GBR ?= { trigger_event = { id = ca_british_debt.2 popup = yes } }
	}

	fail = {
		OR = {
			custom_tooltip = {
				text = je_ca_british_debt_repudiated_tt
				has_modifier = modifier_ca_british_debt_repudiated
			}
			# CAFM P3: impago (ca_british_debt.14)
			custom_tooltip = {
				text = je_ca_british_debt_default_tt
				has_variable = ca_british_debt_default
			}
		}
	}

	# CAFM P3 (punto 2): presion de los bonistas y del consul. Cada evento tiene su propio trigger.
	on_monthly_pulse = {
		random_events = {
			100 = 0
			6 = ca_british_debt.7
			4 = ca_british_debt.8
			3 = ca_british_debt.9
			3 = ca_british_debt.10
			3 = ca_british_debt.11
			3 = ca_british_debt.12
			2 = ca_british_debt.13
			5 = ca_british_debt.14
			4 = ca_british_debt.15
			4 = ca_british_debt.16
			3 = ca_british_debt.17
			3 = ca_british_debt.18
			6 = ca_british_debt.19
			4 = ca_british_debt.20
		}
	}

	on_fail = {
		set_variable = ca_british_debt_closed
		save_scope_as = ca_debtor_scope
		c:GBR ?= { trigger_event = { id = ca_british_debt.6 popup = yes } }
	}

	invalid = {
		OR = {
			NOT = { exists = c:GBR }
			# Si GBR absorbe al deudor como súbdito, la deuda deja de tener sentido
			is_subject_of = c:GBR
			# Un sucesor pierde su parte si la Federación se reconstituye
			AND = {
				exists = c:UCA
				NOT = { c:UCA ?= THIS }
			}
		}
	}

	on_invalid = {
		remove_variable = ca_british_debt_initialized
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_ca_british_debt_status_4
				trigger = { je:je_ca_british_debt ?= { has_modifier = modifier_ca_british_debt_4 } }
			}
			triggered_desc = {
				desc = je_ca_british_debt_status_3
				trigger = { je:je_ca_british_debt ?= { has_modifier = modifier_ca_british_debt_3 } }
			}
			triggered_desc = {
				desc = je_ca_british_debt_status_2
				trigger = { je:je_ca_british_debt ?= { has_modifier = modifier_ca_british_debt_2 } }
			}
			triggered_desc = {
				desc = je_ca_british_debt_status_1
				trigger = { je:je_ca_british_debt ?= { has_modifier = modifier_ca_british_debt_1 } }
			}
		}
	}

	weight = 1000

	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = yes
}
