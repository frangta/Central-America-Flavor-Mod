# ==============================================
# MOVIMIENTO LIBERAL (Reformistas / Modernizadores)
# ==============================================
movement_ca_liberal = {
    category = movement_category_ideological
    ideology = ideology_ca_liberal_movement

    character_ideologies = {
        ideology_market_liberal
        ideology_liberal_leader
        ideology_republican_leader
        ideology_radical
        ideology_humanitarian
        ideology_ca_liberal_monarchist
        ideology_ca_secularist

    }

    creation_trigger = {
        NOT = {
            any_political_movement = {
                is_political_movement_type = movement_ca_liberal
            }
        }
        only_centralamericans_cultures = yes
    }

    creation_weight = {
        value = 100
        if = {
            limit = {
                has_law_or_variant = law_type:law_monarchy
                country_has_voting_franchise = no
            }
            multiply = 0.1
        }
    }

    on_created = {
        add_movement_enthusiasm_modifier = yes
    }
    disband_trigger = {
        
        OR = {
            country_has_primary_culture = cu:central_american
            country_has_primary_culture = cu:guatemalan
            country_has_primary_culture = cu:el_salvadorian
            country_has_primary_culture = cu:honduran
            country_has_primary_culture = cu:nicaraguan
            country_has_primary_culture = cu:costa_rican
            country_has_primary_culture = cu:panamanian
        }
        
        NOT = { has_variable = ca_organization_apro }
    }
	
    #on_disbanded = {
    #    
    #    #hidden_effect = { remove_modifier = fervor_ideologico }
    #}

    character_support_trigger = {
        OR = {
            has_ideology = ideology:ideology_ca_liberal_monarchist 
            has_ideology = ideology:ideology_ca_secularist 
            has_ideology = ideology:ideology_ca_criollo_liberal 
            has_ideology = ideology:ideology_ca_criollo_elite 
            scope:interest_group ?= {
                OR = {
                    has_ideology = ideology:ideology_liberal
                    has_ideology = ideology:ideology_liberal_modern
                }
            }
        }
    }

    character_support_weight = {
        value = 200
        if = {
            limit = {
                OR = {
                    has_ideology = ideology:ideology_republican_leader
                    has_ideology = ideology:ideology_liberal_leader
                    has_ideology = ideology:ideology_market_liberal
                    has_ideology = ideology:ideology_radical
                }
            }
            multiply = 5
        }
        else_if = {
            limit = { has_ideology = ideology:ideology_modernizer_leader }
            multiply = 2
        }
    }

    # -- REGLAS DE APOYO DE LA POBLACIÓN --
    pop_support_trigger = {
        NOT = { is_pop_type = slaves } # Los esclavos no participan en este movimiento
    }

    pop_support_factors = {
        movement_support_high_urbanization
        movement_support_high_literacy
        movement_support_capitalists
        movement_support_academics
        movement_support_bureaucrats
        movement_support_clerks
    }

    pop_support_weight = {
        # 1. Lógica de Supresión/Fomento (Estándar de Victoria 3)
        if = {
            limit = {
                exists = scope:political_movement
                scope:political_movement ?= {
                    OR = {
                        is_being_suppressed = yes
                        is_being_bolstered = yes
                    }
                }
            }
            if = {
                limit = { scope:political_movement = { is_being_suppressed = yes } }
                add = {
                    value = suppression_base_value
                    multiply = { value = -0.5 add = owner.modifier:country_suppression_attraction_factor }
                }
            }
            else = {
                add = {
                    value = bolstering_base_value
                    multiply = { value = 0.5 add = owner.modifier:country_bolster_attraction_factor }
                }
            }
        }

        # 2. Bono por Urbanización (Movimiento urbano)
        add = {
            value = 10
            multiply = state.state_urbanization_rate
        }

        # 3. Preferencias por Profesión
        if = {
            limit = { is_pop_type = capitalists }
            add = 30
        }
        else_if = {
            limit = { is_pop_type = academics }
            add = 40
        }
        else_if = {
            limit = { is_pop_type = bureaucrats }
            add = 20
        }
        else_if = {
            limit = { is_pop_type = clerks }
            add = 15
        }
        else_if = {
            limit = { is_pop_type = shopkeepers }
            add = 10
        }

        # 4. Multiplicador por Alfabetización (Pops más educados se unen más)
        multiply = {
            value = literacy_rate
            add = 0.10
        }

        # 5. Multiplicador por Radicalismo
        multiply = {
            value = pop_radical_fraction
            add = 1
        }
    }

    revolution = {
        possible = { political_movement_support >= 0.05 }
        weight = { value = 100 }
        state_weight = {
            value = 1
            if = {
                limit = { owner = { ig:ig_intelligentsia = { is_powerful = yes } } }
                add = 50
            }
        }
        target_fraction_of_states = { value = 0.35 }
    }
}

# MOVIMIENTO CARRILLISTA (Autocracia Aislacionista de Costa Rica)
# Objetivo: Autocracia, Burocracia Designada y Aislamiento del caos federal
# ==============================================
movement_ca_carrillista = {
    category = movement_category_ideological
    ideology = ideology_carrillismo 

    character_ideologies = {
        ideology_carrillismo
        ideology_authoritarian
        ideology_ca_criollo_paternalistic
        ideology_central_america_caudillo
    }

    creation_trigger = {
        # Restringido estrictamente a Costa Rica
        c:COS ?= this
        NOT = {
            any_political_movement = { is_political_movement_type = movement_ca_carrillista }
        }
    }

    creation_weight = { 
        value = 100 
        
        # Surge con mucha más fuerza si Costa Rica todavía está atrapada en la caótica Federación
        if = {
            limit = { is_subject_of = c:UCA }
            multiply = 2 
        }
    }
    
    on_created = { 
        add_movement_enthusiasm_modifier = yes 
    }

    character_support_trigger = {
        OR = {
            has_ideology = ideology:ideology_carrillismo
            has_ideology = ideology:ideology_authoritarian
            has_ideology = ideology:ideology_ca_criollo_paternalistic
            
            # Grupos pragmáticos y oligarcas que buscan estabilidad por encima de todo
            scope:interest_group ?= {
                OR = {
                    is_interest_group_type = ig_armed_forces
                    is_interest_group_type = ig_landowners
                    is_interest_group_type = ig_intelligentsia
                }
            }
        }
    }

    character_support_weight = {
        value = 150
        if = {
            limit = { has_ideology = ideology:ideology_carrillismo }
            multiply = 5
        }
        else_if = {
            limit = { has_ideology = ideology:ideology_ca_criollo_paternalistic }
            multiply = 2
        }
    }

    pop_support_trigger = {
        NOT = { is_pop_type = slaves }
        culture = cu:costa_rican
    }

    # Atraen a las élites y a la burocracia estatal, ignoran a las bases campesinas
    pop_support_factors = {
        movement_support_aristocrats
        movement_support_bureaucrats
        movement_support_officers
        movement_support_capitalists
    }

    # Si Costa Rica es independiente, buscarán dar un Golpe de Estado Autoritario
    revolution = {
        possible = { political_movement_support >= 0.05 }
        weight = { value = 150 }
        state_weight = {
            value = 1
            if = {
                limit = { owner = { ig:ig_armed_forces = { is_powerful = yes } } }
                add = 50
            }
        }
    }

    # Si Costa Rica es súbdito de la UCA, buscarán la Independencia Absoluta
    secession = {
        possible = {
            political_movement_support >= 0.05
            owner ?= { is_subject_of = c:UCA }
        }
        weight = { value = 1000 }
        state_weight = {
            value = 1
        }
    }
}
