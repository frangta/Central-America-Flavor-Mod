movement_ca_autocratic_autonomist = {
    category = movement_category_ideological
    ideology = ideology_ca_autocratic_autonomist_movement

    character_ideologies = {
        ideology_ca_leader_autocratic_autonomist
        ideology_central_america_caudillo
        ideology_authoritarian
    }

    creation_trigger = {
        NOT = {
            any_political_movement = { is_political_movement_type = movement_ca_autocratic_autonomist }
        }
        only_centralamericans_cultures = yes
        has_variable = ca_organization_apro
    }

    creation_weight = {
        value = 50
        if = {
            limit = { has_law_or_variant = law_type:law_ca_central_american_unitary }
            multiply = 2
        }
        if = {
            limit = { has_law_or_variant = law_type:law_ca_caudillismo }
            multiply = 1.5
        }
        if = {
            limit = { is_subject = yes }
            multiply = 1.5
        }
    }

    on_created = {
        add_movement_enthusiasm_modifier = yes
    }

    
	
    disband_trigger = {
                has_variable = ca_organization_apro
    }

    character_support_trigger = {
        OR = {
            has_ideology = ideology:ideology_ca_leader_autocratic_autonomist
            has_ideology = ideology:ideology_central_america_caudillo
            has_ideology = ideology:ideology_authoritarian
            scope:interest_group ?= {
                OR = {
                    is_interest_group_type = ig_armed_forces
                    is_interest_group_type = ig_landowners
                }
            }
        }
    }

    character_support_weight = {
        value = 100
        if = {
            limit = {
                OR = {
                    has_ideology = ideology:ideology_ca_leader_autocratic_autonomist
                    has_ideology = ideology:ideology_central_america_caudillo
                    has_ideology = ideology:ideology_authoritarian
                }
            }
            multiply = 4
        }
    }

    pop_support_trigger = {
        NOT = { is_pop_type = slaves }
    }

    pop_support_factors = {
        movement_support_aristocrats
        movement_support_officers
    }

    pop_support_weight = {
        value = 10
        if = {
            limit = { owner ?= { has_law_or_variant = law_type:law_ca_central_american_unitary } }
            multiply = { value = 2 desc = "CA_MOV_UNITARY_LAW" }
        }
        if = {
            limit = { owner ?= { has_law_or_variant = law_type:law_ca_caudillismo } }
            multiply = { value = 1.5 desc = "CA_MOV_CAUDILLOS" }
        }
        if = {
            limit = { owner ?= { is_subject = yes } }
            multiply = { value = 1.5 desc = "CA_MOV_MEMBER_STATE" }
        }
        multiply = {
            value = pop_radical_fraction
            add = 1
        }
    }

    additional_radicalism_factors = {
        value = 0
        if = {
            limit = { owner ?= { has_law_or_variant = law_type:law_ca_central_american_federal } }
            add = { value = 0.25 desc = "CA_RAD_AUTOCRATIC_AUTONOMIST" }
        }
    }

    revolution = {
        possible = { political_movement_support >= 0.1 }
        weight = { value = 50 }
    }
}
