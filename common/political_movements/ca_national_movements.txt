# 1. Movimiento Nacional Guatemalteco (GUA)
movement_ca_guatemala_national = {
	category = movement_category_pan_national
	ideology = ideology_national_liberal_movement
	character_ideologies = { ideology_ethno_nationalist ideology_radical }

	can_cause_obstinance = yes

	creation_trigger = {
		any_scope_state = { state_region = s:STATE_GUATEMALA }
		NOT = { country_has_primary_culture = cu:guatemalan }
	}
	
	creation_weight = { value = 100 }
	on_created = { add_movement_enthusiasm_modifier = yes }

    disband_trigger = {
        OR = {
            NOT = { any_scope_state = { state_region = s:STATE_GUATEMALA } }
            
            cu:guatemalan = { culture_current_fervor < 20 }
        }
    }

    on_disbanded = {
        
    }

	character_support_trigger = { culture = cu:guatemalan }
	character_support_weight = { value = 100 }

	pop_support_trigger = {
		culture = cu:guatemalan
		state = { state_region = s:STATE_GUATEMALA }
	}

	pop_support_factors = {
		movement_support_national_awakening
		movement_support_cultural_fervor
		movement_support_academics
		movement_support_shopkeepers
		movement_support_officers
		movement_support_farmers
		movement_support_clerks
		movement_support_soldiers
		movement_support_peasants
		movement_support_colonial_settlers
		movement_support_nationalism_tech
		movement_support_pan_nationalism_tech
	}

	pop_support_weight = {
		add = { value = 10 desc = "POP_BASE_SUPPORT" }
		multiply = { value = culture.culture_current_fervor subtract = 15 multiply = 0.015 add = 1 }
	}

	secession = {
		possible = { political_movement_support >= 0.05 }
		weight = { value = 1000 }
		state_weight = {
			value = 1
			if = { limit = { NOT = { state_region = s:STATE_GUATEMALA } } multiply = 0 }
		}
		forced_tags = {
			GUA = { trigger = { always = yes } weight = 1000 }
		}
	}
}

# 8. Movimiento Nacional de Los Altos (ALT)
movement_ca_los_altos_national = {
	category = movement_category_pan_national
	ideology = ideology_national_liberal_movement
	character_ideologies = { ideology_ethno_nationalist ideology_radical }

	can_cause_obstinance = yes

	creation_trigger = {
		# Surge principalmente cuando Guatemala (GUA) los controla, o hay alto tumulto
		any_scope_state = { state_region = s:STATE_GUATEMALA }
		c:GUA ?= this 
		NOT = { exists = c:ALT }
	}
	
	creation_weight = { value = 100 }
	on_created = { add_movement_enthusiasm_modifier = yes }

	
	disband_trigger = {
        OR = {
            exists = c:ALT
            cu:central_american = { culture_current_fervor < 20 }
        }
    }

    on_disbanded = { }
	
	# Los Altos se basaba mucho en los criollos que rechazaban el gobierno central de Guatemala
	character_support_trigger = { 
		has_ideology = ideology:ideology_reformer 
		culture = cu:central_american
	}
	character_support_weight = { value = 100 }

	pop_support_trigger = {
		culture = cu:central_american
		state = { state_region = s:STATE_GUATEMALA }
	}
	pop_support_factors = {
		movement_support_national_awakening
