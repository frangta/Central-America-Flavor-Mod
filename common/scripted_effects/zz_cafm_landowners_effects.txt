# =====================================================================
# CAFM - Terratenientes (hacendados) sin ideologias vanilla
# ---------------------------------------------------------------------
# El mod les da ideology_ca_criollo_paternalistic, ideology_ca_criollo_stratocratic
# e ideology_central_american_reactionary_2, pero vanilla les deja
# patriarchal (nadie la quitaba) y el on_enable vanilla puede volver a
# poner republican_paternalistic en paises hispanohablantes.
# Este efecto quita las vanilla. Se llama al empezar la partida y cada mes
# (zz_cafm_landowners_on_actions.txt), solo en paises cuyos terratenientes
# ya tienen la ideologia criolla del mod.
# Alcance: pais.
# =====================================================================
cafm_clean_landowner_ideologies = {
	ig:ig_landowners ?= {
		if = {
			limit = { has_ideology = ideology:ideology_paternalistic }
			remove_ideology = ideology_paternalistic
		}
		if = {
			limit = { has_ideology = ideology:ideology_hierarchic }
			remove_ideology = ideology_hierarchic
		}
		if = {
			limit = { has_ideology = ideology:ideology_patriarchal }
			remove_ideology = ideology_patriarchal
		}
		if = {
			limit = { has_ideology = ideology:ideology_republican_paternalistic }
			remove_ideology = ideology_republican_paternalistic
		}
		if = {
			limit = { has_ideology = ideology:ideology_oligarchic }
			remove_ideology = ideology_oligarchic
		}
	}
}
