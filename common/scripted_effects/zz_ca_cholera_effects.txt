# Derogacion historica de los Codigos Livingston (13 mar 1838), solo IA historica. Scope: UCA.
ca_livingston_historic_repeal = {
	if = {
		limit = {
			ca_frca_ai_historic = yes
			has_variable = ca_livingston_applied
			NOT = { has_variable = ca_livingston_repealed }
			game_date >= 1838.3.13
		}
		set_variable = ca_livingston_repealed
		if = {
			limit = { has_modifier = uca_new_penal_code }
			remove_modifier = uca_new_penal_code
		}
		add_modifier = { name = ca_livingston_repealed months = 60 }
	}
}
