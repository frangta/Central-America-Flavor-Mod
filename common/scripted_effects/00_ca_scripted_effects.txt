
calculate_federalists_vs_separatists_progress = {
	set_variable = { name = federalists_vs_separatists_next value = 0 }
	if = {
		limit = {
			ig:ig_intelligentsia = {
				is_powerful = yes
			}
		}
		change_variable = {
			name = federalists_vs_separatists_next
			add = 1
		}
	}
	if = {
		limit = {
			ig:ig_intelligentsia = {
				is_in_government = yes
			}
		}
		change_variable = {
			name = federalists_vs_separatists_next
			add = 1
		}
	}
	if = {
		limit = {
			ig:ig_intelligentsia  = {
				is_marginal = yes
			}
		}
		change_variable = {
			name = federalists_vs_separatists_next
			add = -5
		}
	}
	if = {
		limit = {
			ig:ig_intelligentsia = {
				is_in_government = no
			}
		}
		change_variable = {
			name = federalists_vs_separatists_next
			add = -2
		}
	}	
	if = {
		limit = {
			OR = {
				has_law = law_type:law_ca_central_american_federal
			
			}
		}
		change_variable = {
			name = federalists_vs_separatists_next
			add = 10
		}
	}
	if = {
		limit = {
			OR = {
				has_law = law_type:law_ca_central_american_unitary
		}
		change_variable = {
			name = federalists_vs_separatists_next
			add = -10
		}
	}
	}
	if = {
		limit = {
			ig:ig_landowners = {
				is_powerful = yes
			}
		}
		change_variable = {
			name = federalists_vs_separatists_next
			add = 2
		}
	}
	if = {
		limit = {
			ig:ig_landowners = {
				is_in_government = yes
			}
		}
		change_variable = {
			name = federalists_vs_separatists_next
			add = 1
		}
	}	
	if = {
		limit = {
			ig:ig_landowners  = {
				is_marginal = yes
			}
		}
		change_variable = {
			name = federalists_vs_separatists_next
			add = 2
		}
	}
	if = {
		limit = {
			ig:ig_landowners = {
				is_in_government = no
			}
		}
		change_variable = {
			name = federalists_vs_separatists_next
			add = 1
		}
	}

	


}

        every_scope_state = { # Transición de State Region a State

            add_modifier = { name = liberal_stronghold }
        }
    }

    # GUATEMALA: La Vieja Élite y el Caudillismo

            add_modifier = { name = unitary_stronghold }
            add_modifier = { name = conservative_stronghold }
            add_modifier = { name = reactionary_stronghold }
        }
    }

            add_modifier = { name = conservative_stronghold_small_nation }
        }
    }
