
calculate_federalists_vs_separatists_progress = {
	set_variable = { name = federalists_vs_separatists_next value = 0 }
	if = {
		limit = {
			ig:ig_intelligentsia = {
				is_powerful = yes
			}
		}
		change_variable = {
			name = federalists_vs_separatists_next
			add = 1
		}
	}
	if = {
		limit = {
			ig:ig_intelligentsia = {
				is_in_government = yes
			}
		}
		change_variable = {
			name = federalists_vs_separatists_next
			add = 1
		}
	}
	if = {
		limit = {
			ig:ig_intelligentsia  = {
				is_marginal = yes
			}
		}
		change_variable = {
			name = federalists_vs_separatists_next
			add = -5
		}
	}
	if = {
		limit = {
			ig:ig_intelligentsia = {
				is_in_government = no
			}
		}
		change_variable = {
			name = federalists_vs_separatists_next
			add = -2
		}
	}	
	if = {
		limit = {
			OR = {
				has_law = law_type:law_federal
			
			}
		}
		change_variable = {
			name = federalists_vs_separatists_next
			add = 10
		}
	}
	if = {
		limit = {
			OR = {
				has_law = law_type:law_devolved
		}
		change_variable = {
			name = federalists_vs_separatists_next
			add = -10
		}
	}
	}
	if = {
		limit = {
			ig:ig_landowners = {
				is_powerful = yes
			}
		}
		change_variable = {
			name = federalists_vs_separatists_next
			add = 2
		}
	}
	if = {
		limit = {
			ig:ig_landowners = {
				is_in_government = yes
			}
		}
		change_variable = {
			name = federalists_vs_separatists_next
			add = 1
		}
	}	
	if = {
		limit = {
			ig:ig_landowners  = {
				is_marginal = yes
			}
		}
		change_variable = {
			name = federalists_vs_separatists_next
			add = 2
		}
	}
	if = {
		limit = {
			ig:ig_landowners = {
				is_in_government = no
			}
		}
		change_variable = {
			name = federalists_vs_separatists_next
			add = 1
		}
	}

	


}
calculate_frca_support_progress = {
	
		if = {
				limit = {
					has_variable = gua_support_var
					NOT = { exists = c:GUA }
					NOT = { has_variable = je_gua_support_var }
				}
				change_variable = { name = frca_support add = 1 }
			}
		if = {
				limit = {
					OR = {
						NOT = { has_variable = gua_support_var }
						exists = c:GUA
					}
					has_variable = je_gua_support_var
				}
				change_variable = { name = frca_support add = -1 }
			}
		if = {
				limit = {
					has_variable = els_support_var
					NOT = { exists = c:ELS }
					NOT = { has_variable = je_els_support_var }
				}
				change_variable = { name = frca_support add = 1 }
			}
		if = {
				limit = {
					OR = {
						NOT = { has_variable = els_support_var }
						exists = c:ELS
					}
					has_variable = je_els_support_var
				}
				change_variable = { name = frca_support add = -1 }
			}
		if = {
				limit = {
					has_variable = hon_support_var
					NOT = { exists = c:HON }
					NOT = { has_variable = je_hon_support_var }
				}
				change_variable = { name = frca_support add = 1 }
			}
		if = {
				limit = {
					OR = {
						NOT = { has_variable = hon_support_var }
						exists = c:HON
					}
					has_variable = je_hon_support_var
				}
				change_variable = { name = frca_support add = -1 }
			}
		if = {
				limit = {
					has_variable = nic_support_var
					NOT = { exists = c:NIC }
					NOT = { has_variable = je_nic_support_var }
				}
				change_variable = { name = frca_support add = 1 }
			}
		if = {
				limit = {
					OR = {
						NOT = { has_variable = nic_support_var }
						exists = c:NIC
					}
					has_variable = je_nic_support_var
				}
				change_variable = { name = frca_support add = -1 }
			}
		if = {
				limit = {
					has_variable = cos_support_var
					NOT = { exists = c:COS }
					NOT = { has_variable = je_cos_support_var }
				}
				change_variable = { name = frca_support add = 1 }
			}
		if = {
				limit = {
					OR = {
						NOT = { has_variable = cos_support_var }
						exists = c:COS
					}
					has_variable = je_cos_support_var
				}
				change_variable = { name = frca_support add = -1 }
			}


	    if = {
	     	limit = {
	     		exists = c:ELS
	     		has_variable = frca_elsalvador_peace_var
	     	}
	     	change_variable = { name = frca_support add = -5 }
	     }
	    if = {
	     	limit = {
	     		exists = c:HON
	     		has_variable = frca_honduras_peace_var
	     	}
	     	change_variable = { name = frca_support add = -5 }
	     }
	    if = {
	     	limit = {
	     		exists = c:NIC
	     		has_variable = frca_nicaragua_peace_var
	     	}
			change_variable = { name = frca_support add = -5 }
	     }
	    if = {
	     	limit = {
	     		exists = c:COS
	     		has_variable = frca_costarica_peace_var
	     	}
	     	change_variable = { name = frca_support add = -5 }
	     }



}
