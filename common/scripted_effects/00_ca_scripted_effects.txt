ca_setup_ideological_strongholds_effect = {
    # ---------------------------------------------------------
    # EL SALVADOR: La Capital Federal y Liberal
    # ---------------------------------------------------------
    s:STATE_SAN_SALVADOR = { 
        every_scope_state = { # Transición de State Region a State
            add_modifier = { name = federal_stronghold }
            add_modifier = { name = liberal_stronghold }
        }
    }

    # ---------------------------------------------------------
    # GUATEMALA: La Vieja Élite y el Caudillismo
    # ---------------------------------------------------------
    s:STATE_GUATEMALA = {
        every_scope_state = {
            add_modifier = { name = unitary_stronghold }
            add_modifier = { name = conservative_stronghold }
            add_modifier = { name = reactionary_stronghold }
        }
    }

    # ---------------------------------------------------------
    # NICARAGUA: División y Separatismo
    # ---------------------------------------------------------
    s:STATE_NICARAGUA = {
        every_scope_state = {
            add_modifier = { name = separatism_stronghold }
            add_modifier = { name = liberal_stronghold_small_nation }
            add_modifier = { name = conservative_stronghold_small_nation }
        }
    }

    # ---------------------------------------------------------
    # COSTA RICA: Aislacionismo y Orden
    # ---------------------------------------------------------
    s:STATE_COSTA_RICA = {
        every_scope_state = {
            add_modifier = { name = separatism_stronghold }
            add_modifier = { name = unitary_stronghold_small_nation }
        }
    }

    # ---------------------------------------------------------
    # HONDURAS: Federalismo en Decadencia
    # ---------------------------------------------------------
    s:STATE_HONDURAS = {
        every_scope_state = {
            add_modifier = { name = waning_federal_stronghold }
            add_modifier = { name = conservative_stronghold_small_nation }
        }
    }
}
