# --- CREAR GUATEMALA ---
ca_create_guatemala_effect = {
    if = {
        limit = {
            exists = s:STATE_GUATEMALA.region_state:UCA
            NOT = { exists = c:GUA }
        }
        # Ajuste de Tierras Natales (Homelands)
        s:STATE_GUATEMALA = {
            add_homeland = cu:guatemalan
            remove_homeland = cu:central_american
        }
        # Creación del País
        create_country = {
            tag = GUA
            origin = ROOT
            state = s:STATE_GUATEMALA.region_state:UCA
            on_created = {
                set_variable = { name = ca_secession_country }
                save_scope_as = scope_GUA
                
                # Ajuste Cultural
                add_primary_culture = cu:guatemalan
                remove_primary_culture = cu:central_american
                
                # Llamada al setup político y de personajes
                ca_liberate_guatemala_effect = yes 
            }
        }
        # Convertir en Dependencia
		c:UCA ?= {
			create_diplomatic_pact = {
			country = scope:scope_GUA
			type = ca_dependency
			}
		}
 
    }
}
