# CAFM P11: efectos de bastiones, identidad post-UCA y golpes de Estado.

# Scope: pais. Pone o quita los companeros de pais de los bastiones segun lo que
# tengan sus estados (las claves country_/interest_group_ no funcionan en un estado).
ca_sh_refresh_country = {
	if = {
		limit = {
			NOT = { has_modifier = ca_sh_federal_country }
			any_scope_state = {
				OR = {
					has_modifier = federal_stronghold
					has_modifier = age_of_federal_stronghold
				}
			}
		}
		add_modifier = { name = ca_sh_federal_country }
	}
	else_if = {
		limit = {
			has_modifier = ca_sh_federal_country
			NOT = {
				any_scope_state = {
				OR = {
					has_modifier = federal_stronghold
					has_modifier = age_of_federal_stronghold
				}
				}
			}
		}
		remove_modifier = ca_sh_federal_country
	}

	# unitary_stronghold ya da autoridad cuando se aplica al pais (eventos .79 y .119):
	# no se suma el companero en ese caso.
	if = {
		limit = {
			NOT = { has_modifier = ca_sh_unitary_country }
			NOT = { has_modifier = unitary_stronghold }
			any_scope_state = {
				OR = {
					has_modifier = unitary_stronghold
					has_modifier = age_of_unitary_stronghold
				}
			}
		}
		add_modifier = { name = ca_sh_unitary_country }
	}
	else_if = {
		limit = {
			has_modifier = ca_sh_unitary_country
			OR = {
				has_modifier = unitary_stronghold
				NOT = {
					any_scope_state = {
				OR = {
					has_modifier = unitary_stronghold
					has_modifier = age_of_unitary_stronghold
				}
					}
				}
			}
		}
		remove_modifier = ca_sh_unitary_country
	}
	if = {
		limit = {
			NOT = { has_modifier = ca_sh_waning_country }
			any_scope_state = {
				OR = {
					has_modifier = waning_federal_stronghold
					has_modifier = waning_unitary_stronghold
				}
			}
		}
		add_modifier = { name = ca_sh_waning_country }
	}
	else_if = {
		limit = {
			has_modifier = ca_sh_waning_country
			NOT = {
				any_scope_state = {
				OR = {
					has_modifier = waning_federal_stronghold
					has_modifier = waning_unitary_stronghold
				}
				}
			}
		}
		remove_modifier = ca_sh_waning_country
	}

	if = {
		limit = {
			NOT = { has_modifier = ca_sh_liberal_country }
			any_scope_state = {
				OR = {
					has_modifier = liberal_stronghold
					has_modifier = age_of_liberal_stronghold
				}
			}
		}
		add_modifier = { name = ca_sh_liberal_country }
	}
	else_if = {
		limit = {
			has_modifier = ca_sh_liberal_country
			NOT = {
				any_scope_state = {
				OR = {
					has_modifier = liberal_stronghold
					has_modifier = age_of_liberal_stronghold
				}
				}
			}
		}
		remove_modifier = ca_sh_liberal_country
	}

	if = {
		limit = {
			NOT = { has_modifier = ca_sh_reactionary_country }
			any_scope_state = {
				OR = {
					has_modifier = reactionary_stronghold
					has_modifier = reactionary_stronghold_small_nation
					has_modifier = age_of_reactionary_stronghold
				}
			}
		}
		add_modifier = { name = ca_sh_reactionary_country }
	}
	else_if = {
		limit = {
			has_modifier = ca_sh_reactionary_country
			NOT = {
				any_scope_state = {
				OR = {
					has_modifier = reactionary_stronghold
					has_modifier = reactionary_stronghold_small_nation
					has_modifier = age_of_reactionary_stronghold
				}
				}
			}
		}
		remove_modifier = ca_sh_reactionary_country
	}

	if = {
		limit = {
			NOT = { has_modifier = ca_sh_conservative_country }
			any_scope_state = {
				OR = {
					has_modifier = conservative_stronghold
					has_modifier = age_of_conservative_stronghold
				}
			}
		}
		add_modifier = { name = ca_sh_conservative_country }
	}
	else_if = {
		limit = {
			has_modifier = ca_sh_conservative_country
			NOT = {
				any_scope_state = {
				OR = {
					has_modifier = conservative_stronghold
					has_modifier = age_of_conservative_stronghold
				}
				}
			}
		}
		remove_modifier = ca_sh_conservative_country
	}
}

# Identidad politica post-UCA. Antes nunca se aplicaba (ca_post_uca_game_setup no lo
# llama nadie). Una sola vez, cuando la UCA ya no existe o se disolvio: quita el
# modificador ca_uca_* de cada region y pone su version post-UCA.
ca_p11_post_uca_setup = {
	if = {
		limit = {
			NOT = { has_global_variable = ca_p11_post_uca_done }
			OR = {
				NOT = { exists = c:UCA }
				has_global_variable = ca_dissolution_fired
			}
		}
		set_global_variable = ca_p11_post_uca_done
		ca_p11_post_uca_one = { REGION = STATE_SAN_SALVADOR OLD = ca_uca_liberal_stronghold NEW = ca_liberal_stronghold }
		ca_p11_post_uca_one = { REGION = STATE_GUATEMALA OLD = ca_uca_conservative_capital NEW = ca_conservative_capital }
		ca_p11_post_uca_one = { REGION = STATE_COSTA_RICA OLD = ca_uca_isolated_autonomists NEW = ca_isolated_autonomists }
		ca_p11_post_uca_one = { REGION = STATE_HONDURAS OLD = ca_uca_federalist_heartland NEW = ca_federalist_heartland }
		ca_p11_post_uca_one = { REGION = STATE_PANAMA OLD = ca_uca_isthmian_liberalism NEW = ca_isthmian_liberalism }
		ca_p11_post_uca_one = { REGION = STATE_NICARAGUA OLD = ca_uca_leones_vs_granadinos NEW = ca_leones_vs_granadinos }
		ca_p11_post_uca_one = { REGION = STATE_CHIAPAS OLD = ca_uca_frontier_tensions NEW = ca_frontier_tensions }
	}
}
# Uso: ca_p11_post_uca_one = { REGION = STATE_X OLD = ca_uca_x NEW = ca_x }
ca_p11_post_uca_one = {
	s:$REGION$ = {
		every_scope_state = {
			if = {
				limit = { has_modifier = $OLD$ }
				remove_modifier = $OLD$
			}
			if = {
				limit = { NOT = { has_modifier = $NEW$ } }
				add_modifier = { name = $NEW$ }
			}
		}
	}
}

# Scope: pais. Avance mensual del complot (ca_coup_plot, 0-100) y desenlace.
# ca_coup_side: 1 unitario, 2 federalista.
ca_p11_coup_tick = {
	if = {
		limit = { has_variable = ca_coup_plot }
		change_variable = { name = ca_coup_plot add = 3 }
		if = {
			limit = {
				OR = {
					AND = {
						var:ca_coup_side = 1
						ca_p11_movement_ge = { TYPE = movement_ca_unitarian VALUE = 0.15 }
					}
					AND = {
						var:ca_coup_side = 2
						ca_p11_movement_ge = { TYPE = movement_ca_federalist VALUE = 0.15 }
					}
				}
			}
			change_variable = { name = ca_coup_plot add = 3 }
		}
		if = {
			limit = { government_legitimacy >= 75 }
			change_variable = { name = ca_coup_plot add = -4 }
		}
		if = {
			limit = { has_modifier = ca_p11_plot_paranoia }
			change_variable = { name = ca_coup_plot add = -4 }
		}
		if = {
			limit = { has_modifier = ca_p11_civil_peace }
			change_variable = { name = ca_coup_plot add = -3 }
		}
		if = {
			limit = { var:ca_coup_plot >= 100 }
			trigger_event = { id = ca_coup_events.8 popup = yes }
		}
		else_if = {
			limit = { var:ca_coup_plot <= 0 }
			trigger_event = { id = ca_coup_events.9 popup = yes }
		}
	}
}
