				limit = {
					has_variable = gua_support_var

				change_variable = { name = frca_support add = 1 }

# Impacto: Reducción moderada
ca_reduce_support_separatism_effect = {
	if = {
		limit = { has_variable = frca_support }
		change_variable = {
			name = frca_support
			subtract = 10
		}
	}
}
