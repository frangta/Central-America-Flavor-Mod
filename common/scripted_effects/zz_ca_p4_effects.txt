# CAFM P4 - Bocas del Toro y acciones regionales del minimapa (puntos 4a y 5)
# Variables globales:
#   ca_bocas_control           0-100, control efectivo de Bocas (empieza en 20 con .6)
#   ca_bocas_native_pact       pacto con guaymies y miskitos (quita a GBR el pretexto)
#   ca_bocas_protested         protesta en Londres o acudir a EE. UU.
#   ca_bocas_border_pact       limites pactados con Nueva Granada
#   ca_bocas_federal_district  Bocas es distrito federal
# Scope: UCA salvo que se diga otra cosa.
############################################################################

# Cambiar el control de Bocas (0-100). Uso: ca_bocas_control_change = { VALUE = 10 }
ca_bocas_control_change = {
	if = {
		limit = { NOT = { has_global_variable = ca_bocas_control } }
		set_global_variable = { name = ca_bocas_control value = 20 }
	}
	change_global_variable = { name = ca_bocas_control add = $VALUE$ }
	if = {
		limit = {
			has_global_variable = ca_bocas_control
			global_var:ca_bocas_control > 100
		}
		set_global_variable = { name = ca_bocas_control value = 100 }
	}
	if = {
		limit = {
			has_global_variable = ca_bocas_control
			global_var:ca_bocas_control < 0
		}
		set_global_variable = { name = ca_bocas_control value = 0 }
	}
}

# Pulso mensual (lo llama ca_frca_monthly_update). Sin guarnicion el control se pierde poco a poco;
# con el control bajo, los britanicos pueden desembarcar desde la Mosquitia (.176).
ca_bocas_monthly = {
	if = {
		limit = {
			has_global_variable = ca_bocas_control
			ca_bocas_held_by_uca = yes
		}
		if = {
			limit = {
				NOT = { has_variable = ca_bocas_garrison }
				NOT = { has_global_variable = ca_bocas_federal_district }
			}
			random = {
				chance = 50
				ca_bocas_control_change = { VALUE = -1 }
			}
		}
		if = {
			limit = {
				NOT = { ca_bocas_control_ge = { VALUE = 30 } }
				NOT = { has_global_variable = ca_bocas_native_pact }
			}
			random = {
				chance = 5
				trigger_event = { id = ca_central_america.176 popup = yes }
			}
		}
	}
}

# Estado de una region (para modificadores de las acciones). Uso:
# ca_mm_region_modifier = { CTAG = NIC STATE = STATE_NICARAGUA NAME = ca_coast_colony MONTHS = 120 }
ca_mm_region_modifier = {
	ca_frca_state_modifier = { CTAG = $CTAG$ STATE = $STATE$ NAME = $NAME$ MONTHS = $MONTHS$ }
}

# Espera de una accion regional. Uso: ca_mm_region_cooldown = { KEY = mos_colony DAYS = 730 }
ca_mm_region_cooldown = {
	set_variable = { name = ca_mm_cd_$KEY$ days = $DAYS$ }
}

# Z2 (F): acciones regionales de defensa. Las usan los botones del minimapa
# (ca_p4_minimap_sgui.txt) y la IA (events/ca_frca_ai_events.txt).
ca_mm_act_mos_commissioner = {
	ca_cohesion_spend = { VALUE = 10 }
	add_treasury = { value = ca_money_small multiply = -1 }
	ca_mm_region_cooldown = { KEY = mos_commissioner DAYS = 365 }
	set_global_variable = { name = ca_mos_commissioner years = 5 }
	ca_frca_front_change = { FRONT = british VALUE = -1 }
}

ca_mm_act_mos_king = {
	ca_cohesion_spend = { VALUE = 5 }
	ca_mm_region_cooldown = { KEY = mos_king DAYS = 730 }
	c:MKT = { save_scope_as = ca_mkt }
	change_relations = { country = scope:ca_mkt value = 30 }
	ca_frca_front_change = { FRONT = british VALUE = -1 }
}

ca_mm_act_mos_protest = {
	ca_mm_region_cooldown = { KEY = mos_protest DAYS = 730 }
	c:GBR = { save_scope_as = ca_gbr }
	change_relations = { country = scope:ca_gbr value = -20 }
	ca_frca_front_change = { FRONT = british VALUE = -1 }
}

ca_mm_act_boc_garrison = {
	add_treasury = { value = ca_money_medium multiply = -1 }
	ca_mm_region_cooldown = { KEY = boc_garrison DAYS = 365 }
	set_variable = { name = ca_bocas_garrison days = 730 }
	ca_bocas_control_change = { VALUE = 15 }
}

ca_mm_act_boc_native_pact = {
	ca_cohesion_spend = { VALUE = 5 }
	ca_mm_region_cooldown = { KEY = boc_native_pact DAYS = 1825 }
	set_global_variable = ca_bocas_native_pact
	ca_bocas_control_change = { VALUE = 10 }
}

ca_mm_act_boc_protest = {
	ca_mm_region_cooldown = { KEY = boc_protest DAYS = 1825 }
	set_global_variable = { name = ca_bocas_protested years = 5 }
	c:GBR = { save_scope_as = ca_gbr }
	change_relations = { country = scope:ca_gbr value = -10 }
}

# Z2 (F): accion de la IA sobre un estado (eventos ca_frca_ai.1). Usa los efectos de los botones.
ca_ai_do_state_action = {
	if = {
		limit = {
			ca_frca_route_federal = yes
			ca_frca_action_possible_campaign = { TAG = $TAG$ }
			gold_reserves >= ca_federal_campaign_cost
		}
		ca_frca_action_campaign = { TAG = $TAG$ CTAG = $CTAG$ STATE = $STATE$ }
	}
	else_if = {
		limit = {
			ca_frca_route_federal = yes
			ca_frca_action_possible_negotiate = { TAG = $TAG$ }
		}
		ca_frca_action_negotiate = { TAG = $TAG$ }
	}
	else_if = {
		limit = {
			ca_frca_route_centralist = yes
			ca_frca_action_possible_intendant = { TAG = $TAG$ }
		}
		ca_frca_action_intendant = { TAG = $TAG$ CTAG = $CTAG$ STATE = $STATE$ }
	}
	else_if = {
		limit = {
			ca_frca_route_centralist = yes
			ca_frca_action_possible_coopt = { TAG = $TAG$ }
			gold_reserves >= ca_bribe_caudillo_cost
		}
		ca_frca_action_coopt = { TAG = $TAG$ }
	}
}
