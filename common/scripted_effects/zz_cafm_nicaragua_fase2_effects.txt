# Crea un personaje una sola vez (registro global anti-duplicados).
ca_nic_create_once = {
	if = {
		limit = { NOT = { has_global_variable = $KEY$ } }
		create_character = { template = $TEMPLATE$ }
	}
}

ca_nic_division_clamp = {
	if = {
		limit = { var:nic_division > 100 }
		set_variable = { name = nic_division value = 100 }
	}
	if = {
		limit = { var:nic_division < 0 }
		set_variable = { name = nic_division value = 0 }
	}
}

ca_nic_division_thresholds = {
	if = {
		limit = {
			NOT = { has_variable = nic_div_guerra_fired }
			var:nic_division >= ca_nic_division_guerra
		}
		set_variable = nic_div_guerra_fired
		trigger_event = { id = ca_nicaragua_events.52 popup = yes }
	}
	else_if = {
		limit = {
			NOT = { has_variable = nic_div_ruptura_fired }
			var:nic_division >= ca_nic_division_ruptura
		}
		set_variable = nic_div_ruptura_fired
		trigger_event = { id = ca_nicaragua_events.51 popup = yes }
	}
	else_if = {
		limit = {
			NOT = { has_variable = nic_div_tension_fired }
			var:nic_division >= ca_nic_division_tension
		}
		set_variable = nic_div_tension_fired
		trigger_event = { id = ca_nicaragua_events.50 popup = yes }
	}
}

ca_nic_division_drift = {
	# La rivalidad entre Leon y Granada no se apaga sola.
	change_variable = { name = nic_division add = 1 }
	# Granada fuera del reparto.
	if = {
		limit = { ig:ig_landowners ?= { ig_approval < 0 } }
		change_variable = { name = nic_division add = 2 }
	}
	# Leon fuera del reparto.
	if = {
		limit = { ig:ig_intelligentsia ?= { ig_approval < 0 } }
		change_variable = { name = nic_division add = 2 }
	}
	# La capital alterna calma a las dos ciudades.
	if = {
		limit = { has_variable = nic_capital_alterna }
		change_variable = { name = nic_division subtract = 2 }
	}
	# La guerra arma a los bandos.
	if = {
		limit = { is_at_war = yes }
		change_variable = { name = nic_division add = 1 }
	}
	# Managua ya es la sede.
	if = {
		limit = { has_variable = nic_pacto_managua }
		change_variable = { name = nic_division subtract = 5 }
	}
}

ca_nic_fonseca_clamp = {
	if = {
		limit = { var:nic_fonseca > 100 }
		set_variable = { name = nic_fonseca value = 100 }
	}
	if = {
		limit = { var:nic_fonseca < 0 }
		set_variable = { name = nic_fonseca value = 0 }
	}
}

ca_nic_fonseca_thresholds = {
	if = {
		limit = {
			NOT = { has_variable = nic_fon_dictadura_fired }
			var:nic_fonseca >= ca_nic_fonseca_dictadura
		}
		set_variable = nic_fon_dictadura_fired
		add_modifier = { name = nic_levas_del_mariscal months = 60 }
	}
	else_if = {
		limit = {
			NOT = { has_variable = nic_fon_mariscal_fired }
			var:nic_fonseca >= ca_nic_fonseca_mariscal
		}
		set_variable = nic_fon_mariscal_fired
		add_modifier = { name = nic_gran_mariscal_mando months = 60 }
	}
	else_if = {
		limit = {
			NOT = { has_variable = nic_fon_mando_fired }
			var:nic_fonseca >= ca_nic_fonseca_mando
		}
		set_variable = nic_fon_mando_fired
		ig:ig_armed_forces ?= { add_modifier = { name = ca_ig_approval_up months = 60 } }
	}
}

ca_nic_fonseca_drift = {
	# El mando militar se va comiendo al gobierno.
	change_variable = { name = nic_fonseca add = 1 }
	# Las Fuerzas Armadas en el gobierno.
	if = {
		limit = { ig:ig_armed_forces ?= { is_in_government = yes } }
		change_variable = { name = nic_fonseca add = 1 }
	}
	# En guerra decide el cuartel.
	if = {
		limit = { is_at_war = yes }
		change_variable = { name = nic_fonseca add = 2 }
	}
	# La Intelligentsia en el gobierno.
	if = {
		limit = { ig:ig_intelligentsia ?= { is_in_government = yes } }
		change_variable = { name = nic_fonseca subtract = 1 }
	}
	# Fonseca ya cayo.
	if = {
		limit = { has_variable = nic_fonseca_caido }
		change_variable = { name = nic_fonseca subtract = 10 }
	}
}

		limit = { var:nic_transito < 0 }

ca_nic_transito_thresholds = {
	if = {
		limit = {
			NOT = { has_variable = nic_tra_ruta_fired }
			var:nic_transito >= ca_nic_transito_ruta
		}
		set_variable = nic_tra_ruta_fired
		add_modifier = { name = nic_peaje_del_san_juan months = 240 }
	}
	else_if = {
		limit = {
			NOT = { has_variable = nic_tra_contrato_fired }
			var:nic_transito >= ca_nic_transito_contrato
		}
		set_variable = nic_tra_contrato_fired
		add_modifier = { name = nic_contrato_del_transito months = 180 }
	}
	else_if = {
		limit = {
			NOT = { has_variable = nic_tra_interes_fired }
			var:nic_transito >= ca_nic_transito_interes
		}
		set_variable = nic_tra_interes_fired
	}
}

ca_nic_transito_drift = {
	# El rio San Juan interesa cada ano un poco mas.
	change_variable = { name = nic_transito add = 1 }
	# La fiebre del oro de California.
	if = {
		limit = { game_date >= 1849.1.1 }
		change_variable = { name = nic_transito add = 1 }
	}
	# La boca del rio no es nuestra.
	if = {
		limit = { has_variable = nic_san_juan_perdido }
		change_variable = { name = nic_transito subtract = 2 }
	}
	# Hay guarnicion en San Juan del Norte.
	if = {
		limit = { has_variable = nic_san_juan_guarnecido }
		change_variable = { name = nic_transito add = 2 }
	}
}
