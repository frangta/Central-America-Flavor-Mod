# =====================================================================
# CAFM — Crear el cast de cada país centroamericano
#
# Un efecto por país: cafm_create_cast_<país> = { ALL = no }
#   ALL = no  -> crea solo a los personajes vivos y activos en la fecha
#                (dentro de su ventana "desde–hasta").
#   ALL = yes -> crea de golpe a todos los que ya son adultos (18 años) y no
#                han pasado su fecha "hasta" (no revive a los que ya salieron).
#   La variable global cafm_cast_create_all hace lo mismo que ALL = yes
#   para todos los países (set_global_variable = cafm_cast_create_all).
#
# Protección contra duplicados: no crea a quien ya existe (cafm_hc_<clave>)
# ni a quien murió (cafm_dead_<clave>).
#
# Se llama al nacer el país, desde cafm_cast_activate_<país>
# (zz_cafm_cast_effects.txt), después de traer al cast que vivía en la UCA
# y antes de poner al gobernante de la fecha.
#
# Ventanas: los bloques de uso de cada plantilla o, si no tiene, las fechas
# de claude/CAFM_NATIONAL_CAST_1836_1850.md.
# =====================================================================

# ---------------------------------------------------------------------
# cafm_create_cast_member
#   KEY      = clave del registro (cafm_hc_<KEY>)
#   TEMPLATE = plantilla
#   FROM, TO = ventana de actividad (desde, hasta sin incluir)
#   ADULT    = fecha en que cumple 18 años (para ALL = yes)
#   ALL      = yes / no
# ---------------------------------------------------------------------
cafm_create_cast_member = {
	if = {
		limit = {
			NOT = { has_global_variable = cafm_dead_$KEY$ }
			NOT = { global_var:cafm_hc_$KEY$ ?= { always = yes } }
			game_date < $TO$
			OR = {
				game_date >= $FROM$
				AND = {
					OR = {
						always = $ALL$
						has_global_variable = cafm_cast_create_all
					}
					game_date >= $ADULT$
				}
			}
		}
		create_character = { template = $TEMPLATE$ }
	}
}
