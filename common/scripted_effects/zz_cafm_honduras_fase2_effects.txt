# Crea un personaje una sola vez (registro global anti-duplicados).
ca_hon_create_once = {
	if = {
		limit = { NOT = { has_global_variable = $KEY$ } }
		create_character = { template = $TEMPLATE$ }
	}
}

ca_hon_republica_clamp = {
	if = {
		limit = { var:hon_republica > 100 }
		set_variable = { name = hon_republica value = 100 }
	}
	if = {
		limit = { var:hon_republica < 0 }
		set_variable = { name = hon_republica value = 0 }
	}
}

ca_hon_republica_thresholds = {
	if = {
		limit = {
			NOT = { has_variable = hon_rep_nacion_fired }
			var:hon_republica >= ca_hon_republica_nacion
		}
		set_variable = hon_rep_nacion_fired
		add_modifier = { name = hon_estado_en_pie months = 240 }
	}
	else_if = {
		limit = {
			NOT = { has_variable = hon_rep_estado_fired }
			var:hon_republica >= ca_hon_republica_estado
		}
		set_variable = hon_rep_estado_fired
		add_modifier = { name = hon_constitucion_vigente months = 180 }
	}
}

ca_hon_republica_drift = {
	# La Iglesia en paz con el Estado.
	if = {
		limit = { ig:ig_devout ?= { ig_approval > 0 } }
		change_variable = { name = hon_republica add = 1 }
	}
	# Los letrados en paz con el Estado.
	if = {
		limit = { ig:ig_intelligentsia ?= { ig_approval > 0 } }
		change_variable = { name = hon_republica add = 1 }
	}
	# Gobierno civil tras Pespire.
	if = {
		limit = { has_variable = hon_pespire }
		change_variable = { name = hon_republica add = 2 }
	}
	# La guerra se come la hacienda.
	if = {
		limit = { is_at_war = yes }
		change_variable = { name = hon_republica subtract = 2 }
	}
	# El ejercito descontento.
	if = {
		limit = { ig:ig_armed_forces ?= { ig_approval < 0 } }
		change_variable = { name = hon_republica subtract = 1 }
	}
}

ca_hon_costa_clamp = {
	if = {
		limit = { var:hon_costa > 100 }
		set_variable = { name = hon_costa value = 100 }
	}
	if = {
		limit = { var:hon_costa < 0 }
		set_variable = { name = hon_costa value = 0 }
	}
}

ca_hon_costa_thresholds = {
	if = {
		limit = {
			NOT = { has_variable = hon_cos_perdida_fired }
			var:hon_costa >= ca_hon_costa_perdida
		}
		set_variable = hon_cos_perdida_fired
		add_modifier = { name = hon_costa_sin_aduana months = 240 }
	}
	else_if = {
		limit = {
			NOT = { has_variable = hon_cos_ocupacion_fired }
			var:hon_costa >= ca_hon_costa_ocupacion
		}
		set_variable = hon_cos_ocupacion_fired
		add_modifier = { name = hon_puerto_ocupado months = 120 }
	}
	else_if = {
		limit = {
			NOT = { has_variable = hon_cos_presion_fired }
			var:hon_costa >= ca_hon_costa_presion
		}
		set_variable = hon_cos_presion_fired
	}
}

ca_hon_costa_drift = {
	# Los cortes de Belice avanzan sin permiso de nadie.
	change_variable = { name = hon_costa add = 1 }
	# El reino misquito reclama la costa.
	if = {
		limit = { exists = c:MKT }
		change_variable = { name = hon_costa add = 1 }
	}
	# La protesta diplomatica surte algun efecto.
	if = {
		limit = { has_variable = hon_protesta_en_londres }
		change_variable = { name = hon_costa subtract = 2 }
	}
	# Hay guarnicion en Trujillo.
	if = {
		limit = { has_variable = hon_guarnicion_trujillo }
		change_variable = { name = hon_costa subtract = 2 }
	}
	# Clayton-Bulwer ata las manos de Londres.
	if = {
		limit = { has_variable = hon_clayton_bulwer }
		change_variable = { name = hon_costa subtract = 3 }
	}
}
