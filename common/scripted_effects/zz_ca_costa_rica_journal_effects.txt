# -------------------------------------------------------------------------
# Sumar o restar Soberania de San Jose
# Uso: ca_cos_change_soberania_effect = { VALUE = 20 }   (o VALUE = -30)
# -------------------------------------------------------------------------
ca_cos_change_soberania_effect = {
	je:je_ca_historic_situation_in_costarica ?= {
		add_progress = {
			name = ca_cos_soberania_bar
			value = $VALUE$
		}
	}
}

# -------------------------------------------------------------------------
# Planificador de capitulos (se llama cada mes desde el diario madre)
# Regla: solo un capitulo CON BARRA a la vez. Bocas del Toro no tiene barra
# y puede correr en paralelo.
# Cada capitulo tiene ventana de fechas: si Costa Rica aparece tarde
# (p. ej. por la disolucion de 1838), los capitulos pasados se saltan.
# -------------------------------------------------------------------------
ca_cos_chapter_scheduler_effect = {
	# Capitulo B: La cuestion de Bocas del Toro (feb - jun 1836)
	if = {
		limit = {
			game_date >= 1836.2.1
			game_date <= 1836.6.1
			NOT = { has_variable = ca_cos_cap_b_iniciado }
		}
		set_variable = ca_cos_cap_b_iniciado
		add_journal_entry = { type = je_ca_cos_bocas_del_toro }
	}

	# Capitulo A: La Pacificacion del Norte (may - sep 1836)
	if = {
		limit = {
			game_date >= 1836.5.1
			game_date <= 1836.9.1
			NOT = { has_variable = ca_cos_cap_a_iniciado }
			NOT = { ca_cos_capitulo_con_barra_activo_trigger = yes }
		}
		set_variable = ca_cos_cap_a_iniciado
		add_journal_entry = { type = je_ca_pacificacion_norte }
	}

	# Capitulo C: Las urnas y el cuartel (ene 1837 - mar 1838)
	if = {
		limit = {
			game_date >= 1837.1.1
			game_date <= 1838.3.1
			NOT = { has_variable = ca_cos_cap_c_iniciado }
			NOT = { ca_cos_capitulo_con_barra_activo_trigger = yes }
		}
		set_variable = ca_cos_cap_c_iniciado
		add_journal_entry = { type = je_ca_cos_urnas_y_cuartel }
	}

	# Capitulo D: El regreso del Aguila (1839 - 1841), solo en la ruta separatista
	if = {
		limit = {
			game_date >= 1839.1.1
			game_date <= 1841.12.31
			OR = {
				has_variable = ca_cos_ruta_separatista
				has_variable = ca_cos_asilo_morazan # F15 (E3): el asilo de Morazan permite su regreso
			}
			NOT = { has_variable = ca_cos_cap_d_iniciado }
			NOT = { ca_cos_capitulo_con_barra_activo_trigger = yes }
		}
		set_variable = ca_cos_cap_d_iniciado
		add_journal_entry = { type = je_ca_cos_regreso_aguila }
	}
}

# -------------------------------------------------------------------------
# Balcanizacion: "La Liga renace"
# Se separan Cartago (PC8), Heredia (PC5), Alajuela (PC6) y Guanacaste (PC2).
# Limon y Osa quedan con San Jose.
# Provincias tomadas del bloque "Costarica" de common/decisions/Warcivil.txt.
# -------------------------------------------------------------------------
ca_cos_balkanize_effect = {
	ca_cos_release_city_effect = { TAG = PC8 PROV = x32AA32 }	# Cartago
	ca_cos_release_city_effect = { TAG = PC5 PROV = x535AE8 }	# Heredia
	ca_cos_release_city_effect = { TAG = PC6 PROV = xB0243F }	# Alajuela
	# Guanacaste (2 provincias) solo si no se perdio ya en la Pacificacion del Norte
	if = {
		limit = {
			NOT = { has_variable = ca_cos_guanacaste_perdido }
			NOT = { exists = c:PC2 }
		}
		create_country = {
			tag = PC2
			origin = ROOT
			province = p:x478109
			province = p:xD88369
			on_created = {
				set_variable = ca_cos_ciudad_liga
			}
		}
		change_relations = {
			country = c:PC2
			value = -50
		}
	}
	set_variable = ca_cos_balcanizada
}

# Libera una ciudad de la Liga a partir de una provincia de Costa Rica.
# Mismo patron que ca_create_losaltoswithoutsoconusco_effect del mod.
ca_cos_release_city_effect = {
	if = {
		limit = {
			NOT = { exists = c:$TAG$ }
		}
		create_country = {
			tag = $TAG$
			origin = ROOT
			province = p:$PROV$
			on_created = {
				set_variable = ca_cos_ciudad_liga
			}
		}
		change_relations = {
			country = c:$TAG$
			value = -50
		}
	}
}

# -------------------------------------------------------------------------
# Reunificacion por pacto: absorbe las ciudades que queden
# -------------------------------------------------------------------------
ca_cos_annex_liga_effect = {
	if = { limit = { exists = c:PC8 } annex = c:PC8 }
	if = { limit = { exists = c:PC5 } annex = c:PC5 }
	if = { limit = { exists = c:PC6 } annex = c:PC6 }
	if = {
		limit = {
			exists = c:PC2
			c:PC2 = { has_variable = ca_cos_ciudad_liga }
		}
		annex = c:PC2
	}
}

# -------------------------------------------------------------------------
# Cierre del diario madre (exito o fracaso): apaga los capitulos

ca_cos_close_mother_effect = {

	remove_variable = ca_crisis_guanacaste_active
}

# -------------------------------------------------------------------------
# Capitulo A: etapas de la Pacificacion del Norte (pulso semanal del diario)
# Una etapa por semana, en orden historico. Cada una exige la anterior, asi
# que si el diario abre tarde las etapas atrasadas salen en semanas seguidas.
# Fuente: Fernandez Guardia, Revista del Archivo Nacional 3 (1939).
# -------------------------------------------------------------------------
ca_cos_pacificacion_beats_effect = {
	# May 1836: Joaquin Rivas, insultado en la villa de Nicaragua
	if = {
		limit = {
			game_date >= 1836.5.10
			NOT = { has_variable = ca_cos_pn_comisionado_fired }
		}
		set_variable = ca_cos_pn_comisionado_fired
		trigger_event = { id = ca_costa_rica_events.40 popup = yes }
	}
	# 24-27 jun: llega el correo de la invasion; facultades extraordinarias
	else_if = {
		limit = {
			game_date >= 1836.6.25
			NOT = { has_variable = ca_cos_pn_ligueros_fired }
		}
		set_variable = ca_cos_pn_ligueros_fired
		trigger_event = { id = ca_costa_rica_events.41 popup = yes }
	}
	# 29 jun: el motin de la Villa (el .4 dispara el .5)
	else_if = {
		limit = {
			game_date >= 1836.6.29
			NOT = { has_variable = ca_motin_villa_fired }
		}
		set_variable = ca_motin_villa_fired
		trigger_event = { id = ca_costa_rica_events.4 popup = yes }
	}
	# 7 jul: el saqueo de Bagaces
	else_if = {
		limit = {
			game_date >= 1836.7.7
			NOT = { has_variable = ca_saqueo_bagaces_fired }
		}
		set_variable = ca_saqueo_bagaces_fired
		trigger_event = { id = ca_costa_rica_events.6 popup = yes }
	}
	# 8 jul: el segundo asalto
	else_if = {
		limit = {
			game_date >= 1836.7.8
			NOT = { has_variable = ca_cos_pn_segundo_asalto_fired }
		}
		set_variable = ca_cos_pn_segundo_asalto_fired
		trigger_event = { id = ca_costa_rica_events.61 popup = yes }
	}
	# Julio: El Bolson, solo si la campana va bien (progreso >= 40)
	else_if = {
		limit = {
			game_date >= 1836.7.15
			NOT = { has_variable = ca_desembarco_bolson_fired }
			var:ca_pacificacion_progress >= 40
		}
		set_variable = ca_desembarco_bolson_fired
		trigger_event = { id = ca_costa_rica_events.7 popup = yes }
	}
	# 24 jul: Sapoa, solo con la expedicion en campana
	else_if = {
		limit = {
			game_date >= 1836.7.24
			NOT = { has_variable = ca_cos_pn_sapoa_fired }
			has_variable = ca_cos_expedicion_en_campo
		}
		set_variable = ca_cos_pn_sapoa_fired
		trigger_event = { id = ca_costa_rica_events.71 popup = yes }
	}
	# 4 ago: Morazan envia a Mora Fernandez y Bermudez (evento de la UCA)
	else_if = {
		limit = {
			game_date >= 1836.8.4
			NOT = { has_variable = ca_cos_pn_mediacion_pedida }
			exists = c:UCA
		}
		set_variable = ca_cos_pn_mediacion_pedida
		c:UCA ?= { trigger_event = { id = ca_costa_rica_events.8 popup = yes } }
	}

	# Avance semanal: la campana pacifica el norte poco a poco
	if = {
		limit = { has_variable = ca_cos_quijano_expulsado }
		change_variable = { name = ca_pacificacion_progress add = 3 }
	}
	else_if = {
		limit = { has_variable = ca_cos_expedicion_en_campo }
		change_variable = { name = ca_pacificacion_progress add = 1 }
	}
}

# -------------------------------------------------------------------------
# Capitulo B: etapas de Bocas del Toro (pulso semanal del diario)
# Una etapa por semana, en orden historico; cada una exige la anterior.
# Fuente: carta de Paredes (21 ene 1837), Revista del Archivo Nacional 3 (1938).
# -------------------------------------------------------------------------

	# Feb 1836: jamaicanos bajo bandera britanica en las lagunas
	if = {
		limit = {
			game_date >= 1836.2.1
			NOT = { has_variable = ca_subditos_corona_fired }
		}
		set_variable = ca_subditos_corona_fired
		trigger_event = { id = ca_costa_rica_events.12 popup = yes }
	}
	# Mar 1836: el titulo de 1803 y el uti possidetis de 1825
	else_if = {
		limit = {
			game_date >= 1836.3.1
			NOT = { has_variable = ca_cos_bocas_titulo_fired }
		}
		set_variable = ca_cos_bocas_titulo_fired
		trigger_event = { id = ca_costa_rica_events.13 popup = yes }
	}
	# 30 may 1836: decreto federal y expedicion de Galindo (evento de la UCA)
	else_if = {

			game_date >= 1836.5.30

			NOT = { has_variable = ca_cos_bocas_uca_consultada }
		}
		set_variable = ca_cos_bocas_uca_consultada
		if = {
			limit = { exists = c:UCA }
			c:UCA = { trigger_event = { id = ca_costa_rica_events.14 popup = yes } }

			# Sin Federacion no hay expedicion

	else_if = {

			game_date >= 1836.12.1

	}
	# 18 dic 1836: el desembarco de Paredes

		limit = {
			game_date >= 1836.12.18

		}
		set_variable = ca_cos_bocas_paredes_fired
		trigger_event = { id = ca_costa_rica_events.161 popup = yes }
	}
	# 21 ene 1837: la carta de Paredes
	else_if = {
		limit = {
			game_date >= 1837.1.21
			has_variable = ca_bocas_del_toro_lost
			NOT = { has_variable = ca_cos_bocas_carta_fired }
		}
		set_variable = ca_cos_bocas_carta_fired
		trigger_event = { id = ca_costa_rica_events.162 popup = yes }
	}

	else_if = {
		limit = {
			game_date >= 1837.4.19
			has_variable = ca_cos_bocas_protesta_federal
			NOT = { has_variable = ca_cos_bocas_clamor_fired }
		}
		set_variable = ca_cos_bocas_clamor_fired
		trigger_event = { id = ca_costa_rica_events.17 popup = yes }
	}
}

	}

	# Umbral de soberania: 40 (50 si se castigo la conspiracion, 30 si se ignoro)

		limit = {

			NOT = { has_variable = ca_cos_golpe_resuelto }

					}
					AND = {
						has_variable = ca_cos_conspiracion_ignorada
						ca_cos_soberania_min_trigger = { VALUE = 30 }
					}
					AND = {
						NOT = { has_variable = ca_cos_conspiracion_castigada }
						NOT = { has_variable = ca_cos_conspiracion_ignorada }
						ca_cos_soberania_min_trigger = { VALUE = 40 }
					}
				}
			}
			trigger_event = { id = ca_costa_rica_events.20 popup = yes }
		}
		else = {
			trigger_event = { id = ca_costa_rica_events.330 popup = yes }
		}
	}
}

					}
				}
			}
		}
		set_variable = ca_cos_revuelta_fired
		trigger_event = { id = ca_costa_rica_events.350 popup = yes }
	}
}

	else_if = {
		limit = {
			game_date >= 1844.12.10
			has_variable = ca_cos_gobierno_oreamuno
			NOT = { has_variable = ca_cos_oreamuno_renuncia_fired }
		}
		set_variable = ca_cos_oreamuno_renuncia_fired
		trigger_event = { id = ca_costa_rica_events.221 popup = yes }
	}
	# May 1845: Carrillo, asesinado en El Salvador
	else_if = {
		limit = {
			game_date >= 1845.5.20
			has_global_variable = cafm_hc_carrillo
			OR = {
				has_variable = ca_cos_carrillo_exiliado
				has_variable = ca_cos_carrillo_desterrado
			}
			NOT = { has_variable = ca_cos_muerte_carrillo_fired }
		}
		set_variable = ca_cos_muerte_carrillo_fired
		trigger_event = { id = ca_costa_rica_events.222 popup = yes }
	}
	# 7 jun 1846: la caida de Gallegos
	else_if = {
		limit = {
			game_date >= 1846.6.7
			OR = {
				has_variable = ca_cos_gobierno_gallegos
				has_variable = ca_cos_gobierno_oreamuno
			}
			NOT = { has_variable = ca_cos_caida_gallegos_fired }
		}
		set_variable = ca_cos_caida_gallegos_fired
		trigger_event = { id = ca_costa_rica_events.223 popup = yes }
	}
	# Ene 1847: la Constitucion de 1847
	else_if = {
		limit = {
			game_date >= 1847.1.21
			NOT = { has_variable = ca_cos_constitucion_1847_fired }
		}
		set_variable = ca_cos_constitucion_1847_fired
		trigger_event = { id = ca_costa_rica_events.224 popup = yes }
	}
	# 11 abr 1847: los electores de 1847
	else_if = {
		limit = {
			game_date >= 1847.4.11
			NOT = { has_variable = cos_elec_1847_fired }
		}
		set_variable = cos_elec_1847_fired
		trigger_event = { id = ca_costa_rica_events.23 popup = yes }
	}
	# 31 ago 1848: la Republica (solo si gobierna Castro)
	else_if = {
		limit = {
			game_date >= 1848.8.31
			has_variable = ca_cos_gobierno_castro
			NOT = { has_variable = ca_cos_republica_fired }
		}
		set_variable = ca_cos_republica_fired
		trigger_event = { id = ca_costa_rica_events.231 popup = yes }
	}
	# Sep 1848: la bandera
	else_if = {
		limit = {
			game_date >= 1848.9.29
			has_variable = ca_cos_republica_1848
			NOT = { has_variable = ca_cos_bandera_1848_fired }
		}
		set_variable = ca_cos_bandera_1848_fired
		trigger_event = { id = ca_costa_rica_events.232 popup = yes }
	}
	# Nov 1848: la reforma del sufragio
	else_if = {
		limit = {
			game_date >= 1848.11.22
			NOT = { has_variable = ca_cos_reforma_1848_fired }
		}
		set_variable = ca_cos_reforma_1848_fired
		trigger_event = { id = ca_costa_rica_events.233 popup = yes }
	}
	# 15 nov 1849: el general Quiros desconoce a Castro
	else_if = {
		limit = {
			game_date >= 1849.11.15
			has_variable = ca_cos_gobierno_castro
			NOT = { has_variable = ca_cos_renuncia_castro_fired }
		}
		set_variable = ca_cos_renuncia_castro_fired
		trigger_event = { id = ca_costa_rica_events.234 popup = yes }
	}
	# 2-9 dic 1849: los electores de 1849 (si Castro cayo o si no gobernaba)
	else_if = {
		limit = {
			game_date >= 1849.12.9
			NOT = { has_variable = ca_cos_gobierno_castro }
			NOT = { has_variable = cos_elec_1849_fired }
		}
		set_variable = cos_elec_1849_fired
		trigger_event = { id = ca_costa_rica_events.24 popup = yes }
	}
	# 4 jun 1850: el intento de golpe del general Manuel Quiros
	else_if = {
		limit = {
			game_date >= 1850.6.4
			NOT = { has_variable = ca_cos_quiros_1850_fired }
		}
		set_variable = ca_cos_quiros_1850_fired
		trigger_event = { id = ca_costa_rica_events.235 popup = yes }
	}
}

# Lo llama el pulso mensual de je_ca_historic_situation_in_costarica. Scope: COS.
# -------------------------------------------------------------------------
ca_cos_consequences_monthly = {
	# Bloque 1 - Quien gobierna
	if = {
		limit = { has_variable = ca_gobierno_carrillo_dictadura game_date < 1842.4.1 NOT = { has_variable = ca_cos_dictadura_push } }
		set_variable = { name = ca_cos_dictadura_push days = 365 }
		ca_cos_change_soberania_effect = { VALUE = 5 }
	}
	if = {
		limit = { has_variable = ca_cos_castro_depuesto NOT = { has_variable = ca_cos_c_castro } }
		set_variable = ca_cos_c_castro
		add_modifier = { name = ca_cos_castro_depuesto_mod months = 24 }
	}
	# Bloque 2 - Morazan y la Federacion
	if = {
		limit = { has_variable = ca_cos_estado_federal_autonomo NOT = { has_variable = ca_cos_c_autonomo } }
		set_variable = ca_cos_c_autonomo
		c:UCA ?= { ca_frca_loyalty_up = { TAG = cos } }
	}
	if = {
		limit = { has_variable = ca_cos_separada_1838 NOT = { has_variable = ca_cos_c_separada } }
		set_variable = ca_cos_c_separada
		set_global_variable = ca_sep_cos_fired
		c:UCA ?= { set_variable = { name = frca_loyalty_cos value = 0 } }
	}
	if = {
		limit = {
			has_variable = ca_intervencion_morazan_fired
			NOT = { has_variable = ca_cos_c_intervencion }
			ca_frca_phase_ge = { VALUE = 4 }
		}
		set_variable = ca_cos_c_intervencion
		c:UCA ?= { ca_frca_front_change = { FRONT = conservative VALUE = 1 } }
	}
	if = {
		limit = { has_variable = ca_informe_jimenez_leaked NOT = { has_variable = ca_cos_c_jimenez } }
		set_variable = ca_cos_c_jimenez
		c:UCA ?= { change_relations = { country = root value = -20 } }
		ca_fvs_bar_change = { VALUE = -10 }
	}
	if = {
		limit = { has_variable = ca_cos_nicaragua_desconfia NOT = { has_variable = ca_cos_c_nicaragua } }
		set_variable = ca_cos_c_nicaragua
		c:NIC ?= { change_relations = { country = root value = -20 } }
	}
	if = {
		limit = { has_variable = ca_cos_morazan_fusilado NOT = { has_variable = ca_cos_c_fusilado } }
		set_variable = ca_cos_c_fusilado
		c:ELS ?= { change_relations = { country = root value = -20 } }
		c:HON ?= { change_relations = { country = root value = -20 } }
	}
	# Bloque 3 - Politica interna
	if = {
		limit = {
			has_journal_entry = je_ca_cos_nueva_liga
			OR = {
				has_variable = ca_cos_ligueros_indultados
				has_variable = ca_cos_amnistia_1837
				has_variable = ca_cos_reformas_moderadas
			}
			NOT = { has_variable = ca_cos_c_liga }
		}
		set_variable = ca_cos_c_liga
		if = {
			limit = { has_variable = ca_cos_ligueros_indultados }
			je:je_ca_cos_nueva_liga = { add_progress = { name = ca_cos_reunificacion_bar value = 10 } }
		}
		if = {
			limit = { has_variable = ca_cos_amnistia_1837 }
			je:je_ca_cos_nueva_liga = { add_progress = { name = ca_cos_reunificacion_bar value = 10 } }
		}
		if = {
			limit = { has_variable = ca_cos_reformas_moderadas }
			je:je_ca_cos_nueva_liga = { add_progress = { name = ca_cos_reunificacion_bar value = 10 } }
		}
	}
	if = {
		limit = {
			OR = {
				has_variable = ca_cos_sublevados_arrestados
				has_variable = ca_cos_mora_desterrado
			}
			NOT = { has_variable = ca_cos_c_represion }
		}
		set_variable = ca_cos_c_represion
		add_radicals = { value = small_radicals culture = cu:costa_rican }
		set_variable = ca_cos_conspirators_may_return
	}
	if = {
		limit = { has_variable = ca_cos_renuncia_rechazada NOT = { has_variable = ca_cos_c_renuncia } }
		set_variable = ca_cos_c_renuncia
		add_modifier = { name = ca_cos_renuncia_rechazada_mod months = 24 }
	}
	# Bloque 4 - Instituciones y economia
	if = {
		limit = {
			has_variable = ca_cos_constitucion_1847
			NOT = { has_variable = ca_cos_c_1847 }
			OR = {
				has_law = law_type:law_parliamentary_republic
				has_law = law_type:law_presidential_republic
			}
		}
		set_variable = ca_cos_c_1847
		active_law:lawgroup_governance_principles ?= {
			add_amendment = {
				type = amendment_costa_rica_1st_republic
				sponsor = root.ig:ig_intelligentsia
			}
		}
	}
	if = {
		limit = { has_variable = ca_cos_universidad_santo_tomas NOT = { has_variable = ca_cos_c_universidad } }
		set_variable = ca_cos_c_universidad
		ca_enl_bar_change = { VALUE = 15 }
	}
	if = {
		limit = { has_variable = ca_cos_cafe_a_londres NOT = { has_variable = ca_cos_c_cafe } }
		set_variable = ca_cos_c_cafe
		ca_coffee_bar_change = { VALUE = 10 }
		c:GBR ?= { change_relations = { country = root value = 20 } }
	}
	# Bloque 5 - Finales de la fragmentacion
	if = {
		limit = { has_variable = ca_cos_unitaria NOT = { has_variable = ca_cos_c_unitaria } }
		set_variable = ca_cos_c_unitaria
		add_modifier = { name = ca_cos_republica_unitaria_mod months = 36 }
		s:STATE_COSTA_RICA.region_state:COS ?= { add_modifier = { name = ca_cos_ciudades_agraviadas months = 36 } }
	}
}
