# ============================================================
# CAFM - Efectos de la Deuda Inglesa
# root = país deudor (UCA, GUA, ELS, HON, NIC, COS)
# ============================================================

# Baja un tramo de deuda (4 -> 3 -> 2 -> 1 -> saldada)
# Mismo patrón que hispaniola_events.4
ca_british_debt_reduce_tier_effect = {
	if = {
		limit = { je:je_ca_british_debt ?= { has_modifier = modifier_ca_british_debt_4 } }
		je:je_ca_british_debt ?= {
			remove_modifier = modifier_ca_british_debt_4
			add_modifier = { name = modifier_ca_british_debt_3 months = -1 }
		}
	}
	else_if = {
		limit = { je:je_ca_british_debt ?= { has_modifier = modifier_ca_british_debt_3 } }
		je:je_ca_british_debt ?= {
			remove_modifier = modifier_ca_british_debt_3
			add_modifier = { name = modifier_ca_british_debt_2 months = -1 }
		}
	}
	else_if = {
		limit = { je:je_ca_british_debt ?= { has_modifier = modifier_ca_british_debt_2 } }
		je:je_ca_british_debt ?= {
			remove_modifier = modifier_ca_british_debt_2
			add_modifier = { name = modifier_ca_british_debt_1 months = -1 }
		}
	}
	else_if = {
		limit = { je:je_ca_british_debt ?= { has_modifier = modifier_ca_british_debt_1 } }
		je:je_ca_british_debt ?= {
			remove_modifier = modifier_ca_british_debt_1
		}
	}
}

# Obligación diplomática tras un acuerdo (igual que en Haití):
# si GBR le debía un favor al deudor, se cancela; si no, el deudor queda debiéndole uno a GBR
ca_british_debt_obligation_effect = {
	if = {
		limit = {
			c:GBR ?= { owes_obligation_to = root }
		}
		c:GBR ?= {
			set_owes_obligation_to = {
				country = root
				setting = no
			}
		}
	}
	else = {
		c:GBR ?= { save_scope_as = ca_debt_britain_temp }
		set_owes_obligation_to = {
			country = scope:ca_debt_britain_temp
			setting = yes
		}
	}
}
