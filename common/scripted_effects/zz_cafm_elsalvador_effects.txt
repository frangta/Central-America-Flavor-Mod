# EL ÚLTIMO BASTIÓN: límites, umbrales y deriva
# =====================================================================
ca_els_disolucion_clamp = {
	if = {
		limit = { var:els_disolucion > 100 }
		set_variable = { name = els_disolucion value = 100 }
	}
	if = {
		limit = { var:els_disolucion < 0 }
		set_variable = { name = els_disolucion value = 0 }
	}
}

ca_els_disolucion_thresholds = {
	if = {
		limit = {
			var:els_disolucion >= ca_els_disolucion_sola
			NOT = { has_modifier = els_ultimo_bastion }
		}
		add_modifier = { name = els_ultimo_bastion }
	}
	else_if = {
		limit = {
			var:els_disolucion < ca_els_disolucion_sola
			has_modifier = els_ultimo_bastion
		}
		remove_modifier = els_ultimo_bastion
	}
}

ca_els_disolucion_drift = {
	# --- Sube: la Federación se deshace ---
	change_variable = { name = els_disolucion add = 1 }
	# Cada estado que ya no está en la Unión
	if = {
		limit = { NOT = { exists = c:UCA } }
		change_variable = { name = els_disolucion add = 2 }
	}
	# El tesoro no da para sostener un gobierno federal
	if = {
		limit = { has_variable = els_tesoro_vacio }
		change_variable = { name = els_disolucion add = 2 }
	}
	# El ejército ya no obedece al gobierno civil
	if = {
		limit = { ig:ig_armed_forces ?= { ig_approval < 0 } }
		change_variable = { name = els_disolucion add = 1 }
	}
	# --- Baja: todavía se puede salvar ---
	# Los liberales sostienen la Unión
	if = {
		limit = { ig:ig_intelligentsia ?= { is_in_government = yes } }
		change_variable = { name = els_disolucion subtract = 1 }
	}
	# Hay otro estado que sigue en la Federación
	if = {
		limit = { has_variable = els_aliado_federal }
		change_variable = { name = els_disolucion subtract = 2 }
	}
	# El gobierno paga al ejército federal
	if = {
		limit = { has_modifier = els_pagar_ejercito_federal }
		change_variable = { name = els_disolucion subtract = 2 }
	}
}

		limit = {

			has_modifier = els_regimen_malespin
		}
		remove_modifier = els_regimen_malespin
	}

ca_els_malespin_drift = {
	# --- Sube ---
	# Las Fuerzas Armadas descontentas con el gobierno civil
	if = {
		limit = { ig:ig_armed_forces ?= { ig_approval < 0 } }
		change_variable = { name = els_malespin add = 2 }
	}
	# El obispo Viteri lo respalda
	if = {
		limit = { has_variable = els_viteri_aliado }
		change_variable = { name = els_malespin add = 2 }
	}
	# El Salvador en guerra
	if = {
		limit = { is_at_war = yes }
		change_variable = { name = els_malespin add = 1 }
	}
	# Gobierno sin legitimidad
	if = {
		limit = { government_legitimacy < 50 }
		change_variable = { name = els_malespin add = 1 }
	}
	# --- Baja ---
	# Los liberales en el gobierno
	if = {
		limit = { ig:ig_intelligentsia ?= { is_in_government = yes } }
		change_variable = { name = els_malespin subtract = 1 }
	}
	# La Iglesia le ha retirado el apoyo
	if = {
		limit = { has_variable = els_viteri_roto }
		change_variable = { name = els_malespin subtract = 3 }
	}
	# Barrios y Cabañas conspiran contra él
	if = {
		limit = { has_variable = els_barrios_cabanas }
		change_variable = { name = els_malespin subtract = 2 }
	}
}

# EL CAMINO A LA ARADA: límites y deriva
# =====================================================================
ca_els_arada_clamp = {
	if = {
		limit = { var:els_arada > 100 }
		set_variable = { name = els_arada value = 100 }
	}
	if = {
		limit = { var:els_arada < 0 }
		set_variable = { name = els_arada value = 0 }
	}
}

ca_els_arada_drift = {
	# --- Sube: nos acercamos a la guerra ---
	# Los liberales guatemaltecos exiliados empujan
	if = {
		limit = { has_variable = els_asilo_liberales }
		change_variable = { name = els_arada add = 2 }
	}
	# Pagamos a la Montaña guatemalteca
	if = {
		limit = { has_variable = els_financia_montana }
		change_variable = { name = els_arada add = 2 }
	}
	# Honduras está con nosotros
	if = {
		limit = { has_variable = els_alianza_ocotepeque }
		change_variable = { name = els_arada add = 3 }
	}
	# El ejército está contento y pagado
	if = {
		limit = { ig:ig_armed_forces ?= { ig_approval > 0 } }
		change_variable = { name = els_arada add = 1 }
	}
	# --- Baja: la guerra se aleja ---
	# El bloqueo británico nos ahoga
	if = {
		limit = { has_variable = els_bloqueo_britanico }
		change_variable = { name = els_arada subtract = 2 }
	}
	# El añil no se vende
	if = {
		limit = { has_variable = els_anil_en_crisis }
		change_variable = { name = els_arada subtract = 1 }
	}
}

# Crea un personaje una sola vez.
# Uso: ca_els_create_once = { KEY = viteri TEMPLATE = els_jorge_viteri_character_template }
ca_els_create_once = {
	if = {
		limit = {
			NOT = { has_global_variable = cafm_hc_$KEY$ }
			NOT = { has_variable = els_created_$KEY$ }
		}
		set_variable = els_created_$KEY$
		create_character = { template = $TEMPLATE$ }
	}
}
