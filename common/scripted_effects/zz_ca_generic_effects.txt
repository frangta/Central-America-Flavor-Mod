# Barra de la Federacion: solo si el pais es la UCA o un estado miembro, y solo con el diario.
ca_g16_bar = {
	if = {
		limit = {
			exists = c:UCA
			OR = {
				this = c:UCA
				is_subject_of = c:UCA
			}
		}
		ca_frca_bar_add = { VALUE = $VALUE$ }
	}
}

# Lealtad del estado (scope: estado). En la UCA, la del estado nucleo de esa region; en un miembro, la suya.
ca_g16_state_loyalty_up = { ca_g16_state_loyalty = { DIR = up } }
ca_g16_state_loyalty_down = { ca_g16_state_loyalty = { DIR = down } }
ca_g16_state_loyalty = {
	if = {
		limit = { owner ?= c:UCA }
		if = { limit = { state_region = s:STATE_GUATEMALA } ca_frca_loyalty_$DIR$ = { TAG = gua } }
		else_if = { limit = { state_region = s:STATE_SAN_SALVADOR } ca_frca_loyalty_$DIR$ = { TAG = els } }
		else_if = { limit = { state_region = s:STATE_HONDURAS } ca_frca_loyalty_$DIR$ = { TAG = hon } }
		else_if = { limit = { state_region = s:STATE_NICARAGUA } ca_frca_loyalty_$DIR$ = { TAG = nic } }
		else_if = { limit = { state_region = s:STATE_COSTA_RICA } ca_frca_loyalty_$DIR$ = { TAG = cos } }
	}
	else_if = {
		limit = {
			exists = c:UCA
			owner = { is_subject_of = c:UCA }
		}
		owner = { ca_state_loyalty_$DIR$ = yes }
	}
}

# Lealtad del pais miembro (scope: pais).
ca_g16_country_loyalty_down = {
	if = {
		limit = {
			exists = c:UCA
			is_subject_of = c:UCA
		}
		ca_state_loyalty_down = yes
	}
}

# .97: el frente segun el vecino (Mexico -> mexicano; Gran Bretana o Mosquitia -> britanico)
ca_g16_border_front = {
	if = {
		limit = { scope:border_neighbor = c:MEX }
		ca_frca_front_change = { FRONT = mexican VALUE = 1 }
	}
	else_if = {
		limit = {
			OR = {
				scope:border_neighbor = c:GBR
				scope:border_neighbor = c:MKT
			}
		}
		ca_frca_front_change = { FRONT = british VALUE = 1 }
	}
}

# .146: una disputa entre centroamericanos baja la Patria Grande de quien la tenga.
ca_unity_change_all = {
	if = {
		limit = { has_journal_entry = je_ca_patria_grande }
		ca_unity_change = { VALUE = -5 }
	}
}
