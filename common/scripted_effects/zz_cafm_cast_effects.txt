# =====================================================================
# CAFM — Cast nacional: mecanismo (fase 1)
#
# Diseño: claude/CAFM_NATIONAL_CAST_1836_1850.md (v3).
#
# Reglas
#  1. Cada plantilla registra a su personaje al crearse:
#       set_variable = cafm_cast_<cast>                              (su cast)
#       set_global_variable = { name = cafm_hc_<clave> value = this } (registro)
#  2. Durante la UCA, los líderes de cada estado viven en la UCA como
#     políticos. Al separarse un estado, todo su cast pasa al nuevo país
#     (cafm_cast_activate_<estado>) y se fija su gobernante por fecha.
#  3. Nadie se crea dos veces: todo pasa por cafm_spawn_or_transfer.
#  4. Una muerte histórica deja la marca global cafm_dead_<clave>. Un
#     personaje marcado no se vuelve a crear ni a transferir.
#  5. El gobernante en funciones de la UCA nunca se transfiere.
#
# VERIFICAR con effects.log / triggers.log de 1.13–1.14:
#  - transfer_character (efecto de personaje con país de destino).
#  - every_scope_character como efecto.
#  - Parámetros $X$ en scripted effects anidados.
#  - Que la referencia global a un personaje muerto deje de ser válida.
# =====================================================================


# ---------------------------------------------------------------------
# cafm_spawn_or_transfer
# Trae a un personaje histórico al país del scope actual.
#   KEY      = clave del registro (cafm_hc_<KEY>)
#   TEMPLATE = plantilla de personaje
#   RULER    = yes / no (hacerlo gobernante)
#
# Ejemplo:
#   c:GUA = {
#       cafm_spawn_or_transfer = {
#           KEY = galvez
#           TEMPLATE = gua_mariano_galvez_character_template
#           RULER = yes
#       }
#   }
# ---------------------------------------------------------------------
cafm_spawn_or_transfer = {
	save_scope_as = cafm_cast_target
	if = {
		# Ya existe: se trae aquí
		limit = {
			NOT = { has_global_variable = cafm_dead_$KEY$ }
			global_var:cafm_hc_$KEY$ ?= { always = yes }
		}
		global_var:cafm_hc_$KEY$ = {
			if = {
				limit = {
					NOT = {
						scope:cafm_cast_target = {
							any_scope_character = { this = global_var:cafm_hc_$KEY$ }
						}
					}
				}
				transfer_character = scope:cafm_cast_target
			}
			if = {
				limit = {
					always = $RULER$
					NOT = { scope:cafm_cast_target.ruler ?= this }	# CAFM Z7: ya gobierna, no se repite
				}
				set_character_as_ruler = yes
			}
		}
	}
	else_if = {
		# Nunca existió (o se perdió su referencia) y no murió: se crea
		limit = {
			NOT = { has_global_variable = cafm_dead_$KEY$ }
		}
		create_character = {
			template = $TEMPLATE$
			ruler = $RULER$
		}
	}
}


# ---------------------------------------------------------------------
# cafm_cast_transfer_from_uca
# Pasa al país del scope a todos los personajes del cast CAST que estén
# en la UCA, salvo el gobernante de la UCA y quien tenga cafm_cast_stays.
#   CAST = gua / els / hon / nic / cos / alt / pnm / cip
# ---------------------------------------------------------------------
cafm_cast_transfer_from_uca = {
	save_scope_as = cafm_cast_target
	if = {
		limit = {
			exists = c:UCA
			NOT = { c:UCA ?= this }
		}
		c:UCA = {
			every_scope_character = {
				limit = {
					has_variable = cafm_cast_$CAST$
					NOT = { has_variable = cafm_cast_stays }
					NOT = { c:UCA.ruler ?= this }
				}
				transfer_character = scope:cafm_cast_target
			}
		}
	}
}


# ---------------------------------------------------------------------
# cafm_cast_historical_death
# Muerte histórica de un personaje que sigue en el país del scope.
#   KEY = clave del registro
# ---------------------------------------------------------------------
cafm_cast_historical_death = {
	if = {
		limit = {
			NOT = { has_global_variable = cafm_dead_$KEY$ }
			global_var:cafm_hc_$KEY$ ?= { always = yes }
			any_scope_character = { this = global_var:cafm_hc_$KEY$ }
		}
		global_var:cafm_hc_$KEY$ = { kill_character = yes }
		set_global_variable = cafm_dead_$KEY$
	}
}


# =====================================================================
# MUERTES HISTÓRICAS EN LA UCA (pulso mensual, solo modo histórico)
# Solo matan si el personaje sigue en la UCA. Si su estado ya se separó,
# la muerte la llevan los eventos de ese país (por ejemplo, Zepeda en
# ca_nicaragua_events.13).
# =====================================================================
cafm_cast_uca_historical_deaths = {
	# José Zepeda, asesinado en el golpe de León (25 ene 1837)
	if = {
		limit = {
			game_date >= 1837.1.25
			NOT = { has_global_variable = cafm_zepeda_dead }
		}
		cafm_cast_historical_death = { KEY = zepeda }
		if = {
			limit = { has_global_variable = cafm_dead_zepeda }
			set_global_variable = cafm_zepeda_dead		# compatibilidad con Nicaragua fase 1
		}
	}
	# José Gregorio Salazar, vicepresidente federal, muerto por las tropas
	# de Carrera al tomar la capital (1 feb 1838)
	if = {
		limit = { game_date >= 1838.2.1 }
		cafm_cast_historical_death = { KEY = gregorio_salazar }
	}
	# Francisco Gómez, asesinado en Guatemala (17 may 1838)
	if = {
		limit = { game_date >= 1838.5.17 }
		cafm_cast_historical_death = { KEY = francisco_gomez_els }
	}
	# Juan Galindo, muerto tras la batalla cerca de Tegucigalpa (30 ene 1840)
	if = {
		limit = { game_date >= 1840.1.30 }
		cafm_cast_historical_death = { KEY = galindo }
	}
}

cafm_cast_set_ruler_pnm = {
	if = {
		limit = { game_date < 1851.1.1 }
		cafm_spawn_or_transfer = { KEY = tomas_herrera TEMPLATE = pnm_tomas_de_herrera_character_template RULER = yes }
	}
}

cafm_cast_activate_pnm = {
	if = {
		limit = { c:PNM ?= this }
		cafm_create_cast_pnm = { ALL = no }		# CAFM Z15
		cafm_cast_set_ruler_pnm = yes
	}
}

# =====================================================================
# CAFM Z7 — SUCESIÓN HISTÓRICA (pulso mensual, solo IA en modo histórico)
# La tabla de gobernantes se aplicaba solo al separarse el estado. Ahora,
# si el mundo es histórico (ca_world_historic) y el país es IA, cada mes se
# pone al gobernante de la fecha. Nunca toca a un jugador.
# =====================================================================
cafm_cast_historic_succession = {
	if = {
		limit = {
			is_ai = yes
			has_global_variable = ca_world_historic
		}
		if = {
			limit = { c:GUA ?= this }
			cafm_cast_set_ruler_gua = yes
		}
		else_if = {
			limit = { c:ELS ?= this }
			cafm_cast_set_ruler_els = yes
		}
		else_if = {
			limit = { c:HON ?= this }
			cafm_cast_set_ruler_hon = yes
		}
		else_if = {
			limit = { c:NIC ?= this }
			cafm_cast_set_ruler_nic = yes
		}
		else_if = {
			limit = { c:ALT ?= this }
			cafm_cast_set_ruler_alt = yes
		}
		else_if = {
			limit = { c:PNM ?= this }
			cafm_cast_set_ruler_pnm = yes
		}
		# CAFM Z14: viajes de los que no gobiernan y muertes fuera de la UCA
		cafm_cast_historic_travels = yes
		cafm_cast_world_historical_deaths = yes
	}
}


# =====================================================================
# CAFM Z14 — VIAJES DE LOS PERSONAJES QUE NO GOBIERNAN
# (pulso mensual, solo en el mundo histórico y hacia países IA)
# Diseño: claude/CAFM_NATIONAL_CAST_1836_1850.md, §5.
# =====================================================================

# ---------------------------------------------------------------------
# cafm_cast_move
# Lleva al país del scope a un personaje que ya existe. No hace nada si:
#  - murió (cafm_dead_<KEY>);
#  - gobierna algún país;
#  - ya está aquí;
#  - está en el exilio (no está en ningún país);
#  - lo tiene un jugador.
#   KEY = clave del registro (cafm_hc_<KEY>)
# ---------------------------------------------------------------------
cafm_cast_move = {
	save_scope_as = cafm_cast_target
	if = {
		limit = {
			NOT = { has_global_variable = cafm_dead_$KEY$ }
			global_var:cafm_hc_$KEY$ ?= { is_ruler_of_any_country = no }
			NOT = { any_scope_character = { this = global_var:cafm_hc_$KEY$ } }
			any_country = {
				is_ai = yes
				any_scope_character = { this = global_var:cafm_hc_$KEY$ }
			}
		}
		global_var:cafm_hc_$KEY$ = { transfer_character = scope:cafm_cast_target }
	}
}

# ---------------------------------------------------------------------
# Itinerarios (scope: el país de destino; lo llama cafm_cast_historic_succession)
# Certeza: V verificado · P en parte (solo el año o la etapa).
#  Barrundia (UCA): Los Altos tras el triunfo de Carrera (1838, P) · El Salvador
#    (1839, P) · exilio de abr 1840 (ca_central_america.77) · Guatemala (P; en la
#    Asamblea en mar 1847) · El Salvador, asilo de Vasconcelos (8 ago 1849, V).
#  Pedro Molina (UCA): diputado en Guatemala (1838, V) · El Salvador (1839–40; diputado
#    en 1840, V) · exilio de abr 1840 (.77) · con Morazán en Costa Rica (1842, V) ·
#    Guatemala hasta su muerte en 1854 (P).
#  Carlos Salazar (UCA): jefe interino de Guatemala (30 ene–13 abr 1839, V) · El Salvador
#    (P) · Costa Rica con Morazán (abr 1842, V); muere en San José en 1867.
#  Cabañas (HON): Costa Rica con Morazán (1842, V) · El Salvador tras la muerte de
#    Morazán (V, sin fecha exacta) hasta 1851.
#  Diego Vigil (ELS): sale con Morazán el 8 abr 1840; vive en Granada (V); muere el
#    10 ene 1845 (V).
#  Sacasa: sin datos fiables después de 1840; se queda donde esté.
# ---------------------------------------------------------------------
cafm_cast_historic_travels = {
	# --- Los Altos ---
	if = {
		limit = { c:ALT ?= this }
		if = {
			limit = { game_date >= 1838.8.1 game_date < 1839.3.1 }
			cafm_cast_move = { KEY = barrundia }
		}
	}
	# --- El Salvador ---
	else_if = {
		limit = { c:ELS ?= this }
		if = {
			limit = {
				OR = {
					AND = { game_date >= 1839.3.1 game_date < 1840.4.8 }
					AND = { game_date >= 1849.8.8 game_date < 1851.1.1 }
				}
			}
			cafm_cast_move = { KEY = barrundia }
		}
		if = {
			limit = { game_date >= 1839.4.13 game_date < 1840.4.8 }
			cafm_cast_move = { KEY = pedro_molina }
			cafm_cast_move = { KEY = carlos_salazar }
		}
		if = {
			limit = { game_date >= 1842.10.1 game_date < 1851.1.1 }
			cafm_cast_move = { KEY = cabanas }
		}
	}
	# --- Guatemala ---
	else_if = {
		limit = { c:GUA ?= this }
		if = {
			limit = { game_date >= 1842.6.1 game_date < 1849.8.8 }
			cafm_cast_move = { KEY = barrundia }
		}
		if = {
			limit = {
				OR = {
					AND = { game_date >= 1838.3.1 game_date < 1839.4.13 }
					AND = { game_date >= 1843.1.1 game_date < 1851.1.1 }
				}
			}
			cafm_cast_move = { KEY = pedro_molina }
		}
		if = {
			limit = { game_date >= 1839.1.30 game_date < 1839.4.13 }
			cafm_cast_move = { KEY = carlos_salazar }
		}
	}
	# --- Costa Rica ---
	else_if = {
		limit = { c:COS ?= this }
		if = {
			limit = { game_date >= 1842.4.13 game_date < 1842.9.15 }
			cafm_cast_move = { KEY = pedro_molina }
			cafm_cast_move = { KEY = cabanas }
		}
		if = {
			limit = { game_date >= 1842.4.13 game_date < 1851.1.1 }
			cafm_cast_move = { KEY = carlos_salazar }
		}
	}
	# --- Nicaragua ---
	else_if = {
		limit = { c:NIC ?= this }
		if = {
			limit = { game_date >= 1840.4.8 game_date < 1845.1.10 }
			cafm_cast_move = { KEY = vigil }
		}
	}
}

# Muertes históricas fuera de la UCA (scope: el país IA que tiene al personaje)
cafm_cast_world_historical_deaths = {
	# Diego Vigil, en Granada (10 ene 1845)
	if = {
		limit = { game_date >= 1845.1.10 }
		cafm_cast_historical_death = { KEY = vigil }
	}
	# Agustín Guzmán, herido en el ataque a la Ciudad de Guatemala (oct 1849; día desconocido)
	if = {
		limit = { game_date >= 1849.10.1 }
		cafm_cast_historical_death = { KEY = agustin_guzman }
	}
}


# =====================================================================
# CAFM Z14 — LA RESTAURACIÓN DE LOS ALTOS (1848–1849)
# Scope: GUA (pulso mensual). Solo en el mundo histórico.
#  26 ago 1848: Quetzaltenango se separa otra vez (ca_los_altos_1848.1, jugador o IA).
#  8 may 1849: si Guatemala y Los Altos son IA, Guatemala reincorpora Los Altos.
#  Guzmán pasa antes a Guatemala; su muerte (oct 1849) la lleva el pulso de muertes.
# =====================================================================
cafm_altos_1848_pulse = {
	if = {
		limit = {
			c:GUA ?= this
			has_global_variable = ca_world_historic
			game_date >= 1848.8.26
			game_date < 1849.5.8
			NOT = { exists = c:ALT }
			NOT = { has_global_variable = ca_altos_1848_fired }
			p:x42124E.state.owner ?= this
			p:x78C465.state.owner ?= this
			p:xD9BEFE.state.owner ?= this
			p:xBD8628.state.owner ?= this
		}
		set_global_variable = ca_altos_1848_fired
		trigger_event = { id = ca_los_altos_1848.1 popup = yes }
	}
	else_if = {
		limit = {
			c:GUA ?= this
			is_ai = yes
			is_at_war = no
			has_global_variable = ca_world_historic
			has_global_variable = ca_altos_1848_fired
			NOT = { has_global_variable = ca_altos_1848_ended }
			game_date >= 1849.5.8
			c:ALT ?= {
				is_ai = yes
				is_at_war = no
			}
		}
		set_global_variable = ca_altos_1848_ended
		cafm_spawn_or_transfer = { KEY = agustin_guzman TEMPLATE = cafm_agustin_guzman RULER = no }
		annex = c:ALT
	}
}
