# LA MONTAÑA: límites, umbrales y deriva
# =====================================================================
ca_gua_montana_clamp = {
	if = {
		limit = { var:gua_montana > 100 }
		set_variable = { name = gua_montana value = 100 }
	}
	if = {
		limit = { var:gua_montana < 0 }
		set_variable = { name = gua_montana value = 0 }
	}
}

ca_gua_montana_thresholds = {
	# 40: aparece Carrera como general rebelde
	if = {
		limit = {
			var:gua_montana >= ca_gua_montana_carrera
			NOT = { has_global_variable = cafm_hc_carrera }
			NOT = { has_variable = gua_carrera_created }
		}
		set_variable = gua_carrera_created
		create_character = { template = gua_rafael_carrera_character_template }
	}
	# 80: la Montaña es incontenible
	if = {
		limit = {
			var:gua_montana >= ca_gua_montana_incontenible
			NOT = { has_modifier = gua_montana_incontenible }
		}
		add_modifier = { name = gua_montana_incontenible }
	}
	else_if = {
		limit = {
			var:gua_montana < ca_gua_montana_incontenible
			has_modifier = gua_montana_incontenible
		}
		remove_modifier = gua_montana_incontenible
	}
}

ca_gua_montana_drift = {
	# --- Sube ---
	# Los Códigos de Livingston siguen en vigor
	if = {
		limit = {
			has_variable = gua_livingston_vigente
			NOT = { has_variable = gua_livingston_suspendido }
		}
		change_variable = { name = gua_montana add = 2 }
	}
	# El cólera recorre el país
	if = {
		limit = { has_variable = gua_colera_activo }
		change_variable = { name = gua_montana add = 2 }
	}
	# El campesinado está descontento
	if = {
		limit = { ig:ig_rural_folk ?= { ig_approval < 0 } }
		change_variable = { name = gua_montana add = 1 }
	}
	# La Iglesia está fuera del gobierno
	if = {
		limit = { ig:ig_devout ?= { is_in_government = no } }
		change_variable = { name = gua_montana add = 1 }
	}
	# Siguen las ventas de tierras comunales
	if = {
		limit = {
			NOT = { has_variable = gua_ejidos_garantizados }
		}
		change_variable = { name = gua_montana add = 2 }
	}
	# --- Baja ---
	# Los ejidos están garantizados
	if = {
		limit = { has_variable = gua_ejidos_garantizados }
		change_variable = { name = gua_montana subtract = 2 }
	}
	# La Iglesia está en el gobierno
	if = {
		limit = { ig:ig_devout ?= { is_in_government = yes } }
		change_variable = { name = gua_montana subtract = 1 }
	}
	# El campesinado está contento
	if = {
		limit = { ig:ig_rural_folk ?= { ig_approval > 0 } }
		change_variable = { name = gua_montana subtract = 2 }
	}
	# Los comisionados calman la Montaña (botón)
	if = {
		limit = { has_modifier = gua_comisionados_montana }
		change_variable = { name = gua_montana subtract = 1 }
	}
}

# LOS LUCÍOS: límites, umbrales y deriva

ca_gua_lucios_clamp = {
	if = {
		limit = { var:gua_lucios > 100 }
		set_variable = { name = gua_lucios value = 100 }
	}
	if = {
		limit = { var:gua_lucios < 0 }
		set_variable = { name = gua_lucios value = 0 }
	}
}

ca_gua_lucios_thresholds = {
	# 90: los Lucíos dominan el oriente
	if = {
		limit = {
			var:gua_lucios >= ca_gua_lucios_sitio
			NOT = { has_modifier = gua_lucios_dominan }
		}
		add_modifier = { name = gua_lucios_dominan }
	}
	else_if = {
		limit = {
			var:gua_lucios < ca_gua_lucios_sitio
			has_modifier = gua_lucios_dominan
		}
		remove_modifier = gua_lucios_dominan
	}
}

ca_gua_lucios_drift = {
	# --- Sube ---
	change_variable = { name = gua_lucios add = 1 }
	# El trabajo forzoso de 1846
	if = {
		limit = { has_variable = gua_trabajo_forzoso }
		change_variable = { name = gua_lucios add = 2 }
	}
	# El estanco del aguardiente
	if = {
		limit = { has_variable = gua_estanco_aguardiente }
		change_variable = { name = gua_lucios add = 2 }
	}
	# El Salvador los apoya
	if = {
		limit = { has_variable = gua_els_hostil }
		change_variable = { name = gua_lucios add = 1 }
	}
	# --- Baja ---
	# El Tratado de Zacapa
	if = {
		limit = { has_variable = gua_tratado_zacapa }
		change_variable = { name = gua_lucios subtract = 2 }
	}
	# Carrera manda el ejército
	if = {
		limit = { has_variable = gua_carrera_restituido }
		change_variable = { name = gua_lucios subtract = 2 }
	}
	# El juez de tierras del Convenio de Guadalupe
	if = {
		limit = { has_variable = gua_juez_de_tierras }
		change_variable = { name = gua_lucios subtract = 1 }
	}
}

# Crea a un personaje de Guatemala una sola vez. Uso:
# ca_gua_create_once = { KEY = mejia TEMPLATE = gua_teodoro_mejia_character_template }
ca_gua_create_once = {
	if = {
		limit = {
			NOT = { has_global_variable = cafm_hc_$KEY$ }
			NOT = { has_variable = gua_created_$KEY$ }
		}
		set_variable = gua_created_$KEY$
		create_character = { template = $TEMPLATE$ }
	}
}
