# =========================================================================
# CAFM - Tipos de gobierno centroamericanos (version corregida + Costa Rica)
#
# REEMPLAZA a common/government_types/ca_government_types.txt
#
# El juego usa el PRIMER tipo valido de la lista. Por eso:
#  1. El archivo se llama 00_ca_... para cargar ANTES que el vanilla
#     00_government_types.txt (si no, los tipos vanilla ganan siempre).
#  2. Dentro del archivo el orden va de lo mas especifico a lo mas general:
#       A. Costa Rica - alternativos   (variables de rutas alternativas)
#       B. Costa Rica - historicos     (variables de enmiendas/eventos)
#       C. Costa Rica - dinamicos      (leyes + grupos de interes)
#       D. Centroamerica - genericos   (cualquier pais de cultura CA)
#
# TODOS exigen cultura centroamericana (only_latin_centralamericans_cultures,
# ya definido en zz_ca_scripted_triggers.txt). Los de Costa Rica exigen
# ademas el tag y su variable historica.
# =========================================================================


# =========================================================================

# Confederacion de Costa Rica (pacto de la nueva Liga, evento 360)
gov_ca_cos_confederacion = {
	transfer_of_power = presidential_elective
	new_leader_on_reform_government = no

	male_ruler = "RULER_TITLE_CA_CONFEDERATION_PRESIDENT"
	female_ruler = "RULER_TITLE_CA_CONFEDERATION_PRESIDENT_FEMALE"

	possible = {
		only_latin_centralamericans_cultures = yes
		OR = {
			c:CCR ?= this
			has_variable = ca_cos_confederacion
		}
	}
}

# Republica del Olimpo (liberalismo positivista, 1880-1914)

	new_leader_on_reform_government = no

	male_ruler = "RULER_TITLE_CA_PRESIDENT_OF_THE_REPUBLIC"
	female_ruler = "RULER_TITLE_CA_PRESIDENT_OF_THE_REPUBLIC_FEMALE"

	possible = {
		only_latin_centralamericans_cultures = yes
		c:COS ?= this
		game_date >= 1880.1.1
		has_law_or_variant = law_type:law_presidential_republic
		OR = {
			has_law_or_variant = law_type:law_census_voting
			has_law_or_variant = law_type:law_wealth_voting
		}
		ig:ig_intelligentsia ?= { is_in_government = yes }
	}
}

# CORREGIDO: jefe_supremo_var no se seteaba nunca; ahora tambien vale
# con un gobernante caudillo.

	transfer_of_power = dictatorial
	new_leader_on_reform_government = no

	male_ruler = "RULER_TITLE_CA_PERPETUAL_DICTATOR"
	female_ruler = "RULER_TITLE_CA_PERPETUAL_DICTATOR_FEMALE"

	possible = {
		only_latin_centralamericans_cultures = yes
		has_law_or_variant = law_type:law_autocracy
		OR = {
			has_variable = jefe_supremo_var
			ruler ?= { has_ideology = ideology:ideology_central_america_caudillo }
		}
	}

	on_government_type_change = { change_to_dictatorial = yes }
	on_post_government_type_change = { post_change_to_dictatorial = yes }
}
