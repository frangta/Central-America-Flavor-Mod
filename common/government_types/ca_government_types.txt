	female_ruler = "RULER_TITLE_PERPETUAL_DICTATOR"

		has_law_or_variant = law_type:law_presidential_republic
		country_has_voting_franchise = yes
		has_variable = jefe_supremo_var
		AND = {
            country_has_primary_culture = cu:central_american
            country_has_primary_culture = cu:costa_rican

	}

		change_to_dictatorial = yes
	}

		post_change_to_dictatorial = yes
	}

gov_ca_junta = {
	transfer_of_power = dictatorial

	male_ruler = "RULER_TITLE_GENERAL"
	female_ruler = "RULER_TITLE_GENERAL"

	possible = {
        this = c:UCA
 	has_law_or_variant = law_type:law_presidential_republic
		OR = {
			has_law_or_variant = law_type:law_ca_central_american_unitary
			has_law_or_variant = law_type:law_ca_central_american_federal
		}
		country_has_voting_franchise = no
		OR = {
			has_law_or_variant = law_type:law_mass_conscription
			has_law_or_variant = law_type:law_professional_army
		}
		ig:ig_armed_forces ?= {
			is_in_government = yes
		}

	}

	on_government_type_change = {
		change_to_dictatorial = yes
	}
	on_post_government_type_change = {
		post_change_to_dictatorial = yes
	}
}
