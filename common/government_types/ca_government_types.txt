# 1. Director Supremo (Histórico para estados de la Federación antes de 1847)
gov_ca_director_supremo = {
	transfer_of_power = presidential_elective
	new_leader_on_reform_government = no

	male_ruler = "RULER_TITLE_SUPREME_DIRECTOR"
	female_ruler = "RULER_TITLE_SUPREME_DIRECTOR"



	possible = {
		OR = {
		
			country_has_primary_culture = cu:central_american
			country_has_primary_culture = cu:costa_rican
			country_has_primary_culture = cu:panamanian
			country_has_primary_culture = cu:nicaraguan
			country_has_primary_culture = cu:guatemalan
			country_has_primary_culture = cu:el_salvadorian
			country_has_primary_culture = cu:honduran
		}
	
	    OR = {
	    	this = c:UCA
	    	has_law_or_variant = law_type:law_presidential_republic
	    	OR = {
	    		has_law_or_variant = law_type:law_ca_central_american_confederated
	    		has_law_or_variant = law_type:law_ca_central_american_federal
	    	}		
	    }

	}
}

	new_leader_on_reform_government = no

	male_ruler = "RULER_TITLE_JEFE_DE_ESTADO"
	female_ruler = "RULER_TITLE_JEFE_DE_ESTADO"



	possible = {
		has_law_or_variant = law_type:law_presidential_republic
		
		has_variable = costa_rica_1830s
	}
}

# 3. Jefe Supremo / Dictador Perpetuo (Caudillismo Fuerte)
gov_ca_jefe_supremo = {
	transfer_of_power = dictatorial
	new_leader_on_reform_government = no

	male_ruler = "RULER_TITLE_PERPETUAL_DICTATOR"
	female_ruler = "RULER_TITLE_PERPETUAL_DICTATOR"



	possible = {
		has_law_or_variant = law_type:law_autocracy
		has_variable = jefe_supremo_var
		OR = {
			country_has_primary_culture = cu:central_american
			country_has_primary_culture = cu:costa_rican
			country_has_primary_culture = cu:panamanian
			country_has_primary_culture = cu:nicaraguan
			country_has_primary_culture = cu:guatemalan
			country_has_primary_culture = cu:el_salvadorian
			country_has_primary_culture = cu:honduran
		}
	}
	
	on_government_type_change = { change_to_dictatorial = yes }
	on_post_government_type_change = { post_change_to_dictatorial = yes }
}

# 7. República Presidencial Democrática (Genérica post-Caudillos)

		NOT = { has_law_or_variant = law_type:law_land_reform_banana_republic }
	}
}
