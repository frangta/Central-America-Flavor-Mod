# 1. CASA - Central American South Africa Company
establish_casa_company = {
    is_shown = { any_scope_state = { OR = { state_region = s:STATE_ZAMBEZI state_region = s:STATE_KAZEMBE } } NOT = { exists = c:CASA } }
    possible = { is_subject = no NOT = { any_primary_culture = { has_discrimination_trait = african_heritage } } }
    when_taken = {
        create_country = {
            tag = CASA origin = this
            every_scope_state = { limit = { OR = { state_region = s:STATE_ZAMBEZI state_region = s:STATE_KAZEMBE } } set_state_owner = c:CASA add_claim = ROOT }
            on_created = { activate_law = law_type:law_presidential_republic activate_law = law_type:law_autocracy activate_law = law_type:law_frontier_colonization set_institution_investment_level = { institution = institution_colonial_affairs level = 2 } }
        }
        hidden_effect = {
            create_diplomatic_pact = { country = c:CASA type = chartered_company }
            c:CASA = { add_journal_entry = { type = je_company_independence } top_overlord = { random_primary_culture = { save_scope_as = new_primary } root.religion = { save_scope_as = overlord_religion } } random_primary_culture = { save_scope_as = old_primary } add_primary_culture = scope:new_primary remove_primary_culture = scope:old_primary set_state_religion = scope:overlord_religion }
        }
    }
}
