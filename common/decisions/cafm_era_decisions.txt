cafm_the_rights_of_man = {
	is_shown = {
		is_ca_culture_country = yes
		NOT = { has_modifier = cafm_era_rights_of_man }
		NOT = { has_modifier = cafm_era_political_ideology }
		NOT = { has_modifier = cafm_era_state_and_government }
		NOT = { has_modifier = cafm_era_dreams_of_empire }
		NOT = { has_modifier = cafm_era_means_of_production }
		NOT = { has_modifier = cafm_era_flaws_of_democracy }
	}
	possible = {
		has_technology_researched = rationalism
	}
	when_taken = {
		cafm_clear_era_modifiers = yes
		add_modifier = { name = cafm_era_rights_of_man }
	}
	ai_chance = {
		value = 100
	}
}
