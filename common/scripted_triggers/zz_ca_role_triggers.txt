ca_role_can_support_federation = {
	custom_tooltip = {
		text = ca_button_cooldown_tt
		NOT = { has_variable = support_cooldown }
	}
	custom_tooltip = {
		text = ca_button_alignment_max_tt
		OR = {
			NOT = { has_variable = ca_federalist_influence }
			var:ca_federalist_influence < 3
		}
	}
}
