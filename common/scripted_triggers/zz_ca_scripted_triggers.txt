ca_countries_state_neighbour = {
	custom_tooltip = {
		text = ca_tooltip_countries_state_neighbour
		scope:actor = {
			any_scope_state = {
				is_homeland_of_country_cultures = scope:actor
				OR = {
					region = {
						any_scope_state = {
							is_incorporated = yes
							owner = scope:target_country
						}
					}
					any_neighbouring_state = {
						is_incorporated = yes
						owner = scope:target_country
					}
				}
			}
		}
	}
}



ca_centralamerican_countries_can_unify_trigger = {
	
	scope:actor = {
		prestige > scope:target_country.prestige
	}
	
	custom_tooltip = {
		text = ca_tooltip_centralamerican_countries_share_primary_culture
		ca_centralamerican_share_primary_culture = yes
		scope:actor = {
			"relations(scope:target_country)" >= relations_threshold:neutral
		}
	}
	
	ca_countries_state_neighbour = yes
}
