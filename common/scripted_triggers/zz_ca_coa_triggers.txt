coa_def_central_american_ensign_trigger = {
	exists = c:UCA
	c:UCA ?= { save_temporary_scope_as = custom_overlord }
	coa_def_ensign_trigger = yes
}

coa_def_frca_ensign_trigger = {
	exists = c:UCA
	c:UCA = { has_variable = is_frca_var }
	c:UCA ?= { save_temporary_scope_as = custom_overlord }
	coa_def_ensign_trigger = yes
}

coa_def_soconosco_uca_trigger = {
	exists = c:CIP
	c:CIP = { 
		is_subject_of = c:UCA
		NOT = { owns_entire_state_region = STATE_CHIAPAS } 
	}
}

coa_def_soconosco_trigger = {
	exists = c:CIP
	c:CIP = { 
		NOT = { owns_entire_state_region = STATE_CHIAPAS } 
	}
	OR = {
		c:CIP = { is_subject_of = c:ALT }
		c:CIP = { is_subject_of = c:GUA }
	}
}
