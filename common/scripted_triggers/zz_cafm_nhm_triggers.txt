# =====================================================================
# CAFM — Triggers de Nicaragua (fase 2), Honduras (fase 2) y la Mosquitia.
# =====================================================================

# Quien vive la cadena de Nicaragua: el estado, o la UCA mientras Nicaragua
# siga siendo suya (antes del 30 de abril de 1838).
ca_nic_is_nicaragua = {
	OR = {
		c:NIC ?= this
		AND = {
			c:UCA ?= this
			NOT = { exists = c:NIC }
			any_scope_state = {
				state_region = s:STATE_NICARAGUA
			}
		}
	}
}

ca_hon_is_honduras = {
	OR = {
		c:HON ?= this
		AND = {
			c:UCA ?= this
			NOT = { exists = c:HON }
			any_scope_state = {
				state_region = s:STATE_HONDURAS
			}
		}
	}
}

ca_mkt_is_mosquitia = {
	c:MKT ?= this
}

# Las opciones alternativas solo salen fuera del modo historico.
ca_nic_alt_history = {
	NOT = { has_global_variable = ca_world_historic }
	NOT = { has_variable = nic_historic_mode }
}

ca_hon_alt_history = {
	NOT = { has_global_variable = ca_world_historic }
	NOT = { has_variable = hon_historic_mode }
}

ca_mkt_alt_history = {
	NOT = { has_global_variable = ca_world_historic }
	NOT = { has_variable = mkt_historic_mode }
}
