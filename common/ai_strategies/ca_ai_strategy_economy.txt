# Z2 (D): economia por era para los paises centroamericanos.
# Peso 80: por debajo de las estrategias administrativas de cada pais (100-150),
# asi que solo mandan cuando el pais no tiene la suya (Los Altos, Chiapas, Panama, o
# cuando la estrategia propia no se cumple).

# Anil y grana: la economia de 1836, hasta que llega el cafe.
ai_strategy_ca_era_dye = {
    icon = "gfx/interface/icons/ai_strategy_icons/agricultural_expansion.dds"
    type = administrative

    building_group_weights = {
        bg_plantations = 3.0
        bg_agriculture = 1.5
        bg_light_industry = 1.2
    }

    goods_stances = {
        dye = { stance = wants_export }
        fabric = { stance = wants_export }
        grain = { stance = wants_high_supply }
    }

    possible = {
        is_ca_culture_country = yes
        game_date < 1860.1.1
        NOT = { has_journal_entry = je_ca_coffee_boom }
        NOT = { has_variable = ca_coffee_boom_consolidated }
        NOT = { has_journal_entry = je_ca_agro_export_boom }
    }

    weight = { value = 80 }
}
