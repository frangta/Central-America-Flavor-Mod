ca_golden_grain_bar = {
	name = "ca_golden_grain_bar_name"
	desc = "ca_golden_grain_bar_desc"
	
	default_green = yes
	start_value = 0
	min_value = 0
	max_value = 100

	monthly_progress = {
		# 1. Producción Base
		if = {
			limit = {
				
				any_scope_state = {
					state_has_building_type_levels = {
						target = bt:building_coffee_plantation
						value >= 10
					}
				}
			}
			add = {
				desc = "golden_grain_production_base"
				value = 1
			}
		}

		# 2. Infraestructura
		if = {
			limit = {
				has_technology_researched = railways
				any_scope_state = {
					has_building = building_coffee_plantation
					has_building = building_railway
				}
			}
			add = {
				desc = "golden_grain_infrastructure"
				value = 1
			}
		}

		# 3. Exportaciones
		if = {
			limit = {
				OR = {
					has_law_or_variant = law_type:law_free_trade
					has_law_or_variant = law_type:law_mercantilism
				}
			}
			add = {
				desc = "golden_grain_exports"
				value = 1
			}
		}

		# 4. Apoyo de Oligarquía
		if = {
			limit = {
				ig:ig_landowners = {
					is_powerful = yes
					ig_approval > 5 # CORRECCIÓN: ig_approval en vez de approval
				}
			}
			add = {
				desc = "golden_grain_oligarchy_support"
				value = 0.5
			}
		}

		# 5. Penalización: Falta de Mano de Obra
		if = {
			limit = {
				
				any_scope_state = {
					has_building = building_coffee_plantation
					pop_employment_building = {
						building = building_coffee_plantation
						
					}
				}
			}
			add = {
				desc = "golden_grain_labor_shortage"
				value = -2
			}
		}
	}
}
