ca_golden_grain_bar = {
	name = "ca_golden_grain_bar_name"
	desc = "ca_golden_grain_bar_desc"
	
	default_green = yes
	start_value = 0
	min_value = 0
	max_value = 100

	monthly_progress = {
		# 1. Producción Base
		if = {
			limit = {
				# CORRECCIÓN: Para medir niveles, verificamos que en algún estado haya café nivel > 10
				any_owned_state = {
					state_building_level = {
						building = building_coffee_plantation
						value >= 10
					}
				}
			}
			add = {
				desc = "golden_grain_production_base"
				value = 1
			}
		}

		# 2. Infraestructura
		if = {
			limit = {
				has_technology_researched = railways
				any_owned_state = {
					has_building = building_coffee_plantation
					has_building = building_railway
				}
			}
			add = {
				desc = "golden_grain_infrastructure"
				value = 1
			}
		}

		# 3. Exportaciones
		if = {
			limit = {
				OR = {
					has_law = law_type:law_free_trade
					has_law = law_type:law_mercantilism
				}
			}
			add = {
				desc = "golden_grain_exports"
				value = 1
			}
		}

		# 4. Apoyo de Oligarquía
		if = {
			limit = {
				ig:ig_landowners = {
					is_powerful = yes
					ig_approval > 5 # CORRECCIÓN: ig_approval en vez de approval
				}
			}
			add = {
				desc = "golden_grain_oligarchy_support"
				value = 0.5
			}
		}

		# 5. Penalización: Falta de Mano de Obra
		if = {
			limit = {
				# CORRECCIÓN: Evaluamos la ocupación del edificio específico, no del grupo.
				any_owned_state = {
					has_building = building_coffee_plantation
					building_employment_fraction = {
						building = building_coffee_plantation
						value < 0.75
					}
				}
			}
			add = {
				desc = "golden_grain_labor_shortage"
				value = -2
			}
		}
	}
}

federalists_vs_separatists_progress_core_progress_bar = {
	name = "federalists_vs_separatists_progress_core_progress_bar_name"
	desc = "federalists_vs_separatists_progress_core_progress_bar_federalists_desc"
	second_desc = "federalists_vs_separatists_progress_core_progress_bar_separatists_desc"

	double_sided_gold = yes

	start_value = 100
	min_value = 0
	max_value = 200

	monthly_progress = {
		# ===============================================
		# FACTORES FEDERALISTAS (+)
		# ===============================================
		if = {
			limit = {
				ig:ig_intelligentsia = { is_in_government = yes }
			}
			add = {
				desc = "federalists_intelligentsia_in_gov"
				value = 1
			}
		}

		if = {
			limit = {
				has_law = law_type:law_ca_central_american_federal
			}
			add = {
				desc = "federalists_law_active"
				value = 0.5
			}
		}

		if = {
			limit = {
				legitimacy >= 90
			}
			add = {
				desc = "federalists_high_legitimacy"
				value = 0.5
			}
		}

		if = {
			limit = {
				any_interest_group = {
					leader = {
						has_ideology = ideology:ideology_ca_federalist # CORRECCIÓN: Prefijo ideology: añadido
					}
				}
			}
			add = {
				desc = "federalist_ig_leader_bonus"
				value = 3
			}
		}

		# ===============================================
		# FACTORES SEPARATISTAS (-)
		# ===============================================
		if = {
			limit = {
				ig:ig_landowners = { 
					is_powerful = yes
					ig_approval < -5 # CORRECCIÓN
				}
			}
			add = {
				desc = "separatists_angry_landowners"
				value = -2
			}
		}

		if = {
			limit = {
				ig:ig_devout = {
					is_powerful = yes
					ig_approval < -5 # CORRECCIÓN
				}
			}
			add = {
				desc = "separatists_angry_church"
				value = -1
			}
		}

		if = {
			limit = {
				legitimacy < 25
			}
			add = {
				desc = "separatists_low_legitimacy"
				value = -1
			}
		}

		if = {
			limit = {
				# CORRECCIÓN: Separamos la validación de región de la de estado
				any_owned_state = {
					state_region = { is_homeland = cu:central_american }
					state_turmoil > 0.25
				}
			}
			add = {
				desc = "separatists_local_turmoil"
				value = -1
			}
		}

		if = {
			limit = {
				has_variable = voting_ca_self_determination
			}
			add = {
				desc = "separatists_may_30_decree_active"
				value = -10 
			}
		}        
	}
