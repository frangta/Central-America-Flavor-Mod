ca_pn_zone_low = {
	scope = country
	is_shown = {
		OR = {
			NOT = { has_variable = ca_pacificacion_progress }
			var:ca_pacificacion_progress < 34
		}
	}
}
ca_pn_zone_mid = {
	scope = country
	is_shown = {
		has_variable = ca_pacificacion_progress
		var:ca_pacificacion_progress >= 34
		var:ca_pacificacion_progress < 67
	}
}

	scope = country
	is_shown = {
		has_variable = ca_pacificacion_progress
		var:ca_pacificacion_progress >= 67
