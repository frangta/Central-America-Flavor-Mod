ca_role_support_federation = {
	scope = country
	is_shown = {
		has_journal_entry = je_ca_state_role_in_federation
	}
	is_valid = {
		ca_role_can_support_federation = yes
	}
	effect = {
		ca_ai_sgui_mark = yes # CAFM P8: marca que la IA usa los scripted_gui
		ca_role_do_support_federation = yes
	}
	ai_is_valid = {
		is_ai = yes
		NOT = { has_variable = ca_role_separatist }
	}
	ai_chance = {
		base = 30
		modifier = {
			ig:ig_intelligentsia ?= { is_in_government = yes }
			add = 30
		}
	}
	ai_frequency = 1
}
