ca_mm_select_gua = {
	scope = country
	effect = {
		if = {
			limit = {

			}
			remove_variable = ca_minimap_selected
		}
		else = {
			set_variable = { name = ca_minimap_selected value = 1 }
		}
	}

ca_mm_alt_loyal = {
	scope = country
	is_shown = {
		OR = {
			AND = {
				has_variable = frca_loyalty_alt
				var:frca_loyalty_alt = 3
			}
			# P4: en 1836 Los Altos se dibuja como parte de Guatemala
			AND = {
				NOT = { has_variable = frca_loyalty_alt }
				NOT = { exists = c:ALT }
				has_variable = frca_loyalty_gua
				var:frca_loyalty_gua = 3
			}
		}
	}
}

ca_mm_alt_separated = {

	is_shown = {
		OR = {
			AND = {
				has_variable = frca_loyalty_alt
				var:frca_loyalty_alt <= 0
			}
			# P4: Los Altos existe pero ya no esta en el diario
			AND = {
				NOT = { has_variable = frca_loyalty_alt }
				exists = c:ALT
			}
			# P4: antes de que exista, sigue a Guatemala
			AND = {
				NOT = { has_variable = frca_loyalty_alt }
				NOT = { exists = c:ALT }
				OR = {
					NOT = { has_variable = frca_loyalty_gua }
					var:frca_loyalty_gua <= 0
				}
			}
		}
	}
}
