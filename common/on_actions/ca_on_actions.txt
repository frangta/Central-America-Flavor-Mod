	on_actions = { ca_on_acquired_technology }

ca_on_acquired_technology = {
	effect = {
		if = {
			limit = {
				is_ca_culture_country = yes
				has_technology_researched = prospecting
				NOT = { has_variable = ca_survey_progress }
			}
			set_variable = {
				name = ca_survey_progress
				value = 0
			}
		}
	}
}

	on_actions = { ca_on_game_started }
}

ca_on_game_started = {
	effect = {
		every_country = {
			limit = {
				is_ca_culture_country = yes
				has_technology_researched = prospecting
				NOT = { has_variable = ca_survey_progress }
			}
			set_variable = {
				name = ca_survey_progress
				value = 0
			}
		}
		# CAFM F14: diarios regionales desde el primer dia (Istmo, Talamanca, Soconusco)
		every_country = {
			limit = {
				OR = {
					ca_isthmus_holder = yes
					c:RDT ?= { is_subject_of = prev }
					c:CIP ?= this
				}
			}
			ca_regional_yearly = yes
		}
		# CAFM F13 (A4): la UCA recibe la Ilustracion al inicio
		c:UCA ?= { ca_add_enlightenment_journal = yes }
		# CAFM F2: generales de 1836 en sus formaciones
		ca_assign_starting_generals = yes
		# CAFM F3: respaldo por si el immediate del diario no corre al anadirlo desde la historia
		c:UCA ?= {
			if = {
				limit = {
					has_journal_entry = je_historic_central_america_falls_apart
					NOT = { has_variable = frca_phase }
				}
				ca_frca_journal_setup = yes
			}
		}
		# CAFM fix: eventos de arranque, una sola vez y solo para la UCA
		# (antes estaban en el pulso anual: salian cada año en todos los paises del mundo)
		c:UCA ?= {
			trigger_event = { id = ca_setup_events.3 days = 1 } # variables iniciales
			trigger_event = { id = ca_setup_events.4 days = 1 } # era de los caudillos
			# CAFM F20: fuera ca_setup_events.1 (la eleccion de camino la hace ca_central_america.119)
		}
	}
}
