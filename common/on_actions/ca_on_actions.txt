on_tech_researched = {
    effect = {
        if = {
            limit = {
                is_ca_culture_country = yes
                has_technology_researched = prospecting
                NOT = { has_variable = ca_survey_progress }
            }
            set_variable = {
                name = ca_survey_progress
                value = 0
            }
        }
    }
}

on_country_released_as_independent = {
    effect = {
        
        if = {
            limit = {
               
                has_dlc_feature = ip4_content
                                
                is_historical_spanish_american_colony = yes
                               
                NOT = { has_modifier = modifier_not_recognized_by_spain }
                
                c:SPA ?= { always = yes }

                
                NOT = { 
                    any_overlord_or_above = { c:SPA ?= this }
                }
            }


            add_modifier = {
                name = modifier_not_recognized_by_spain
            }


            add_journal_entry = {
                type = je_spanish_american_independence
            }
        }
    }
}
on_secession_country_created = {
    effect = {
        if = {
            limit = {
                has_dlc_feature = ip4_content

            }

        }
    }
}
