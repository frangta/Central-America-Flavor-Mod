# ====================================================================
# CAFM - LEYES NUEVAS
# ====================================================================
# A) INDEPENDIENTES (sin parent): necesitan posturas en tus ideologías.
#    Solo en grupos donde la ley funciona solo con modificadores
#    (trabajo, burocracia). Los grupos de impuestos y comercio definen
#    tipos impositivos/aranceles dentro de cada ley; por eso el Estanco
#    y las concesiones van como ENMIENDAS (ca_amendments.txt).
#
# B) VARIANTES (con parent): el juego las trata como su ley padre
#    (posturas de IG, journals, has_law_or_variant). Para no saltarse
#    la tecnología del padre, se promulgan DESDE la ley padre:
#    can_enact exige has_law_or_variant = <padre>.
# ====================================================================


# ====================================================================
# A) INDEPENDIENTES

# --------------------------------------------------------------------
# MANDAMIENTOS (lawgroup_labor_rights)
# Reparto forzoso de jornaleros indígenas a las fincas
# (Reglamento de Jornaleros, Guatemala, 1877).
# --------------------------------------------------------------------
law_ca_mandamientos = {
	group = lawgroup_labor_rights
	icon = "gfx/interface/icons/law_icons/serfdom.dds"
	progressiveness = -40

	is_visible = {
		only_latin_centralamericans_cultures = yes
	}

	can_enact = {
		only_latin_centralamericans_cultures = yes
	}

	disallowing_laws = {
		law_council_republic
	}

	modifier = {
		building_group_bg_plantations_throughput_add = 0.10
		building_group_bg_plantations_standard_of_living_add = -1
		state_peasants_mortality_mult = 0.05
		interest_group_ig_landowners_pol_str_mult = 0.10
		interest_group_ig_rural_folk_approval_add = -2
		state_radicals_from_political_movements_mult = 0.10
	}

	ai_will_do = {
		any_interest_group = {
			is_in_government = yes
			OR = {
				is_interest_group_type = ig_landowners
				has_ideology = ideology:ideology_ca_agroexportadores
			}
		}
	}

	ai_enact_weight_modifier = {
		value = 0
		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}

# --------------------------------------------------------------------
# MUNICIPALISMO (lawgroup_bureaucracy)
# Los cabildos y municipalidades administran y recaudan.
# Contrapeso local al poder central (lo que Braulio Carrillo suprimió).
# --------------------------------------------------------------------
law_ca_municipalismo = {
	group = lawgroup_bureaucracy
	icon = "gfx/interface/icons/law_icons/elected_bureaucrats.dds"
	progressiveness = 20

	is_visible = {
		only_latin_centralamericans_cultures = yes
	}

	can_enact = {
		only_latin_centralamericans_cultures = yes
		NOT = { has_law = law_type:law_ca_central_american_unitary }
	}

	disallowing_laws = {
		law_ca_junta
	}

	modifier = {
		country_authority_add = -25
		country_legitimacy_base_add = 10
		country_bureaucracy_mult = 0.10
		state_tax_waste_add = 0.05
		state_turmoil_effects_mult = -0.10
		interest_group_ig_petty_bourgeoisie_pol_str_mult = 0.10
	}

	ai_enact_weight_modifier = {
		value = 0
		if = {
			limit = {
				any_interest_group = {
					is_in_government = yes
					OR = {
						has_ideology = ideology:ideology_ca_liberal
						has_ideology = ideology:ideology_ca_constitutionalist
					}
				}
			}
			add = 200
		}
	}
}

# --------------------------------------------------------------------
# ESCUELA LAICA Y OBLIGATORIA (variante de Escuelas Públicas)
# Costa Rica 1869: primaria gratuita, obligatoria y costeada por el Estado.
# Se promulga desde Escuelas Públicas y requiere un Estado no confesional.
#
# OJO: las variantes NO heredan los modificadores del padre. Copia aquí
# el institution_modifier de law_public_schools de tu versión vanilla
# (game/common/laws/00_education.txt) y súmale lo de abajo.
# --------------------------------------------------------------------
law_ca_escuela_laica = {
	group = lawgroup_education_system
	parent = law_public_schools
	icon = "gfx/interface/icons/law_icons/public_schools.dds"
	progressiveness = 60

	is_visible = {
		only_latin_centralamericans_cultures = yes
	}

	can_enact = {
		only_latin_centralamericans_cultures = yes
		has_law_or_variant = law_type:law_public_schools
		OR = {
			has_law_or_variant = law_type:law_freedom_of_conscience
			has_law_or_variant = law_type:law_total_separation
		}
	}

	institution = institution_schools
	institution_modifier = {
		state_education_access_add = 0.02		# VERIFICAR contra law_public_schools
	}

	modifier = {
		country_academics_pol_str_mult = 0.10
		interest_group_ig_devout_approval_add = -1
		country_institution_cost_institution_schools_mult = 0.10
	}

	ai_enact_weight_modifier = {
		value = 0
		if = {
			limit = {
				any_interest_group = {
					is_in_government = yes
					has_ideology = ideology:ideology_ca_secularist
				}
			}
			add = 200
		}
	}
}
