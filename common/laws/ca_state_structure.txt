# ====================================================================
# CAFM - ESTRUCTURA DEL ESTADO (lawgroup_ca_state_structure)
# ====================================================================
# Leyes INDEPENDIENTES (sin parent) para que las ideologías del mod
# tengan postura propia sobre ellas.
#
# Tres "lados":
#   SOBERANO DE LA UNIÓN  (tiene ca_organization_apro, no es dependencia)
#       law_ca_central_american_federal
#       law_ca_central_american_unitary
#       law_ca_central_american_confederated
#   ESTADO MIEMBRO        (is_subject_type = subject_type_ca_dependency)
#       law_ca_federal_state        <- soberano federal
#       law_ca_confederated_state   <- soberano confederado
#       law_ca_departmental_state   <- soberano unitario      (NUEVA)
#       law_ca_autonomous_state     <- soberano federal/confederado + deseo de libertad alto
#   PAÍS CENTROAMERICANO INDEPENDIENTE (fuera de la Unión)
#       law_ca_sovereign_state                                (NUEVA)
#
# Al cambiar la ley del soberano, on_activate sincroniza a sus estados.
# Las claves antiguas se mantienen porque se usan en historia, ideologías,
# partidos, lobbies, flags, government_types y scripted_effects.
# ====================================================================


# ====================================================================
# LADO SOBERANO
# ====================================================================

# --------------------------------------------------------------------
# REPÚBLICA FEDERAL (statu quo histórico, 1824-1838)
# Lealtad de los estados y legitimidad a cambio de autoridad central.
# --------------------------------------------------------------------
law_ca_central_american_federal = {
	group = lawgroup_ca_state_structure
	icon = "gfx/interface/icons/law_icons/parliamentary_republic.dds"
	progressiveness = 50

	is_visible = {
		hidden_trigger = { exists = this } 
		ca_is_union_sovereign = yes
	}

	can_enact = {
		hidden_trigger = { exists = this } 
		ca_is_union_sovereign = yes
	}

	# Una federación necesita pacto constitucional: no convive con autocracia.
	disallowing_laws = {
		law_autocracy
	}

	modifier = {
		country_authority_add = -25
        country_legitimacy_base_add = 10
        state_tax_capacity_mult = -0.10
        state_turmoil_effects_mult = -0.10
        political_movement_radicalism_add = -0.05
        interest_group_ig_intelligentsia_pol_str_mult = 0.10
		country_liberty_desire_of_subjects_add = -0.1		# por semana; ajustar tras probar
		state_radicals_from_political_movements_mult = -0.10
	}

	on_activate = {
		ca_sync_union_member_laws_effect = { MEMBER_LAW = law_ca_federal_state KEEP_LAW = law_ca_autonomous_state }
	}

	can_impose = {
		always = no
	}

	ai_enact_weight_modifier = {
		value = 0
		if = {
			limit = {
				any_interest_group = {
					is_in_government = yes
					leader ?= { has_ideology = ideology:ideology_ca_federalist }
				}
			}
			add = 300
		}
		if = {
			limit = {
				any_interest_group = {
					is_in_government = yes
					leader ?= { has_ideology = ideology:ideology_ca_morazanista }
				}
			}
			add = 300
		}
		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}

# --------------------------------------------------------------------
# ESTADO FEDERADO
# --------------------------------------------------------------------
law_ca_federal_state = {
	group = lawgroup_ca_state_structure
	icon = "gfx/interface/icons/law_icons/elected_bureaucrats.dds"
	progressiveness = 40

	is_visible = {
		hidden_trigger = { exists = this } 
		ca_is_union_member_state = yes
	}

	can_enact = {
		hidden_trigger = { exists = this } 
		ca_is_union_member_state = yes
		overlord ?= { has_law = law_type:law_ca_central_american_federal }
	}

	modifier = {
        country_authority_add = -15
        country_legitimacy_base_add = 10
        state_tax_capacity_mult = -0.10
	}

	can_impose = {
		has_law = law_type:law_ca_central_american_federal
		scope:target_country ?= {
			is_subject_of = root
			is_subject_type = subject_type_ca_dependency
		}
	}

	ai_enact_weight_modifier = {
		value = 0
		if = {
			limit = { ca_has_mismatched_member_law = yes }
			add = 1000
		}
	}
}

# --------------------------------------------------------------------
# ESTADO AUTÓNOMO
# Concesión a un estado inquieto: compra estabilidad con autonomía.
# Solo con soberano federal o confederado y deseo de libertad >= 50.
# --------------------------------------------------------------------
law_ca_autonomous_state = {
	group = lawgroup_ca_state_structure
	icon = "gfx/interface/icons/law_icons/elected_bureaucrats.dds"
	progressiveness = 20

	is_visible = {
		hidden_trigger = { exists = this } 
		ca_is_union_member_state = yes
	}

	can_enact = {
		hidden_trigger = { exists = this } 
		ca_is_union_member_state = yes
		overlord ?= {
			OR = {
				has_law = law_type:law_ca_central_american_federal
				has_law = law_type:law_ca_central_american_confederated
			}
		}
		liberty_desire >= 50
	}

	modifier = {
		country_authority_add = 50
        state_tax_capacity_mult = 0.10
        country_leverage_resistance_mult = 0.50
        state_turmoil_effects_mult = -0.15
	}

	can_impose = {
		OR = {
			has_law = law_type:law_ca_central_american_federal
			has_law = law_type:law_ca_central_american_confederated
		}
		scope:target_country ?= {
			is_subject_of = root
			is_subject_type = subject_type_ca_dependency
			liberty_desire >= 50
		}
	}

	ai_enact_weight_modifier = {
		value = 0
		if = {
			limit = { liberty_desire >= 75 }
			add = 400
		}
	}
}
