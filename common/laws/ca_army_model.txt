law_ca_comunal_milicia = {
	group = lawgroup_army_model
	icon = "gfx/interface/icons/law_icons/peasant_levies.dds"

	progressiveness = -50

	is_visible = {
	        
        OR = {
            country_has_primary_culture = cu:central_american
            country_has_primary_culture = cu:costa_rican
            country_has_primary_culture = cu:panamanian
            country_has_primary_culture = cu:nicaraguan
            country_has_primary_culture = cu:guatemalan
            country_has_primary_culture = cu:el_salvadorian
            country_has_primary_culture = cu:honduran
        }
	}

	can_enact = {
	}


	modifier = {
		country_can_only_conscript_peasants_bool = yes
		unit_morale_loss_mult = 0.1
		unit_experience_gain_mult = -0.25
		state_conscription_rate_add = 0.04
		country_aristocrats_pol_str_mult = 0.25
        interest_group_ig_landowners_pol_str_mult = 0.25		
		country_military_goods_cost_mult = -0.1
		state_building_barrack_max_level_add = 25
		state_building_conscription_center_max_level_add = 25
	
	}

	can_impose = {
		OR = {
			AND = {
				is_in_same_power_bloc = scope:target_country
				AND = {
					modifier:country_can_impose_same_lawgroup_army_model_in_power_bloc_bool = yes
					has_law_or_variant = scope:law
				}
			}
			can_impose_law_default = yes
		}
	}

	on_enact = {
		custom_tooltip = QUALIFICATIONS_OFFICERS_PEASANT_LEVIES_DESC
        if = {
            limit = { NOT = { has_law_or_variant = law_type:law_peasant_levies } }
            custom_tooltip = aristocrats_af_ig_shift_effect_tt
        }		
		if = {
			limit = {
				NOT = {
					has_law_or_variant = law_type:law_peasant_levies
				}
			}
			custom_tooltip = aristocrats_af_ig_shift_effect_tt
		}
		if = {
			limit = {
				has_law_or_variant = law_type:law_professional_army
			}
			custom_tooltip = military_pops_af_ig_shift_effect_negative_tt
			custom_tooltip = non_military_pops_af_ig_shift_effect_negative_tt
		}
	}

	# AI will never return to peasant levies
	ai_will_do = {
		always = no
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}

law_ca_federal_army = {
	group = lawgroup_army_model
	icon = "gfx/interface/icons/law_icons/professional_army.dds"

	progressiveness = 25

	unlocking_technologies = { military_drill }

	is_visible = {
		# Restringido al ecosistema de Centroamérica
		OR = {
			exists = c:UCA
			country_has_primary_culture = cu:central_american
			country_has_primary_culture = cu:guatemalan
			country_has_primary_culture = cu:el_salvadorian
			country_has_primary_culture = cu:honduran
			country_has_primary_culture = cu:nicaraguan
			country_has_primary_culture = cu:costa_rican
			country_has_primary_culture = cu:panamanian
		}
	}

	modifier = {
		# Balance: Ejército central moderado, gran capacidad de movilización estatal
		unit_morale_loss_mult = -0.05
		state_conscription_rate_add = 0.03
		country_officers_pol_str_mult = 0.25
		state_building_barrack_max_level_add = 25
		state_building_conscription_center_max_level_add = 75
		unit_experience_gain_mult = 0.50
		interest_group_ig_intelligentsia_pol_str_mult = 0.25
	}

	can_enact = {
		
		NOT = { has_law = law_type:law_ca_central_american_confederated
                has_law = law_type:law_ca_central_american_unitary		}
	}

	on_enact = {
		if = {
			limit = {
				has_law_or_variant = law_type:law_peasant_levies
			}
			custom_tooltip = aristocrats_af_ig_shift_effect_negative_tt
		}
		if = {
			limit = {
				has_law_or_variant = law_type:law_professional_army
			}
			custom_tooltip = military_pops_af_ig_shift_effect_negative_tt
		}
		ig:ig_armed_forces = {
            remove_ideology = ideology_caudillismo
            remove_ideology = ideology_paternalistic
            add_ideology = ideology_central_american_federal
            set_interest_group_name = ig_ejercito_federal
        }
	}

	ai_enact_weight_modifier = {
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}

		# Los federalistas siempre intentarán impulsarla
		if = {
			limit = {
				any_interest_group = {
					is_in_government = yes
					leader = { has_ideology = ideology:ideology_ca_federalist }
				}
			}
			add = 500
		}
	}
}
