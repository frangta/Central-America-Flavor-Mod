# --------------------------------------------------------------------
# OLIGARQUÍA CAFETALERA
# Grandes fincas de café; el peón queda atado a la finca.
# --------------------------------------------------------------------
law_land_reform_coffee_oligarchy = {
	group = lawgroup_land_reform
	icon = "gfx/interface/icons/law_icons/tenant_farmers.dds"
	progressiveness = -20

	is_visible = {
		only_latin_centralamericans_cultures = yes
	}

	modifier = {
		interest_group_ig_landowners_pol_str_mult = 0.20
		country_peasants_pol_str_mult = -0.10
		building_coffee_plantation_throughput_add = 0.20
		building_tobacco_plantation_throughput_add = 0.05
		building_sugar_plantation_throughput_add = 0.05
		state_tax_capacity_add = 3
		state_migration_pull_mult = -0.10
	}

	ai_will_do = {
		any_interest_group = {
			is_in_government = yes
			OR = {
				is_interest_group_type = ig_landowners
				has_ideology = ideology:ideology_ca_coffee_liberal
				has_ideology = ideology:ideology_ca_agroexportadores
			}
		}
	}

	ai_enact_weight_modifier = {
		value = 0
		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}
