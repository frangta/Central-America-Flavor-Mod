# group = this is the law_group a law belongs to
# icon = graphical icon shown in-game
# modifier = {} modifier on country for having adopted this law

law_central_american_federation = {
	group = lawgroup_governance_principles
	parent = law_parliamentary_republic # Inherits ideological stances from parent

	icon = "gfx/interface/icons/law_icons/parliamentary_republic.dds"

	progressiveness = 100
	
	requires_law_or = {
	law_presidential_republic
	}

	is_visible = {
	        exists = c:UCA
            country_has_primary_culture = cu:central_american
        OR = {
            
            country_has_primary_culture = cu:costa_rican
            country_has_primary_culture = cu:panamanian
            country_has_primary_culture = cu:nicaraguan
            country_has_primary_culture = cu:guatemalan
            country_has_primary_culture = cu:el_salvadorian
            country_has_primary_culture = cu:honduran
        }
	}


	can_enact = {
	this = c:UCA
	exists = c:UCA
	}

	modifier = {
		country_legitimacy_headofstate_add = 10
		country_legitimacy_govt_total_votes_add = 10
		country_authority_add = 100
		political_movement_radicalism_add = -0.25
		country_economic_dependence_on_overlord_add = 0.6
		country_allow_enacting_decrees_in_subject_bool = yes
	}

	ai_enact_weight_modifier = {
		value = 0
		if = {
			limit = {
				has_journal_entry = je_pedro_republic
			}
			add = 100
		}
		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}
