# group = this is the law_group a law belongs to
# icon = graphical icon shown in-game
# modifier = {} modifier on country for having adopted this law
# ==============================================
# LEY FEDERAL (Status Quo / Histórico)
# Equilibrio: Bonos a la lealtad local, penalización a la autoridad central.
# ==============================================
law_ca_central_american_federal = {
    group = lawgroup_ca_state_structure
    icon = "gfx/interface/icons/law_icons/placeholder.dds" # Asegúrate de tener este icono o usa uno genérico
    
    progressiveness = 50


    is_visible = {
	C:UCA?= this
	    law_ca_valid_trigger = yes
        OR = {
            OR = {exists = c:UCA}

            country_has_primary_culture = cu:central_american
            country_has_primary_culture = cu:costa_rican
            country_has_primary_culture = cu:panamanian
            country_has_primary_culture = cu:nicaraguan
            country_has_primary_culture = cu:guatemalan
            country_has_primary_culture = cu:el_salvadorian
            country_has_primary_culture = cu:honduran
        }
    }


    can_enact = {
        law_ca_valid_trigger = yes
        NOT = { has_law = law_type:law_theocracy }
		
    }

 	disallowing_laws = {
		law_autocracy
	}
	can_impose = {
		scope:target_country ?= {
			has_variable = ca_organization_apro
			is_subject_of = root
		}
	}
    modifier = {
        country_authority_add = -25 
        country_legitimacy_base_add = 50
        state_standard_of_living_add = 0.5
        political_movement_radicalism_add = -0.10
    }

    ai_enact_weight_modifier = {
        value = 50
        if = {
            limit = { has_ideology = ideology:ideology_ca_federalist }
            add = 100
        }
    }
}
