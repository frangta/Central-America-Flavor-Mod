# group = this is the law_group a law belongs to
# icon = graphical icon shown in-game
# modifier = {} modifier on country for having adopted this law
# ==============================================
# LEY FEDERAL (Status Quo / Histórico)
# Equilibrio: Bonos a la lealtad local, penalización a la autoridad central.
# ==============================================
law_central_american_federal = {
    group = lawgroup_ca_organization
    icon = "gfx/interface/icons/law_icons/federal.dds" # Asegúrate de tener este icono o usa uno genérico
    
    progressiveness = 50


    is_visible = {
        is_ca_culture_country = yes 
    }


    can_enact = {
        NOT = { has_law = law_theocracy }
    }

    modifier = {

        country_authority_add = -25 
        

        state_standard_of_living_add = 0.5
        political_movement_radicalism_add = -0.10
        

        country_institutions_bureaucracy_mult = -0.10
    }

    ai_enact_weight_modifier = {
        value = 50
        if = {
            limit = { has_ideology = ideology_ca_federalist }
            add = 100
        }
    }
}

# ==============================================
# LEY UNITARIA (La meta de los Conservadores/Caudillos)
# Equilibrio: Mucha autoridad, pero genera radicales en las provincias.
# ==============================================
law_central_american_unitary = {
    group = lawgroup_ca_organization
    icon = "gfx/interface/icons/law_icons/unitary.dds"
    
    progressiveness = 0

    is_visible = {
        is_ca_culture_country = yes
    }

    modifier = {

        country_authority_add = 100
        country_tax_income_mult = 0.10 # Mejor recaudación
        

        state_turmoil_mult = 0.25 
        political_movement_radicalism_add = 0.15
        
 
        decree_cost_mult = -0.25
    }

    ai_enact_weight_modifier = {
        value = 0
        if = {
            limit = { has_ideology = ideology_ca_unitarian }
            add = 200
        }
    }
}

    icon = "gfx/interface/icons/law_icons/confederation.dds"

    is_visible = {
        is_ca_culture_country = yes
    }

    modifier = {
        country_authority_add = -100
        country_tax_income_mult = -0.50
        
        state_turmoil_mult = -0.50
        

        country_subjects_tax_income_mult = -0.75
    }
    

}
