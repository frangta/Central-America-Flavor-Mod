ca_cen_button_reorganize = {
	name = "ca_cen_button_reorganize"
	desc = "ca_cen_button_reorganize_desc"
	visible = {
		ca_frca_route_centralist = yes
		NOT = { has_variable = ca_cen_army_converted }
		ca_uca_button_tab = { TAB = gov }
	}
	possible = {
		custom_tooltip = { text = ca_button_cooldown_tt NOT = { has_variable = ca_cen_reorganize_cd } }
		ca_mm_can_pay = { VALUE = ca_money_small }
	}
	effect = {
		set_variable = { name = ca_cen_reorganize_cd days = 180 }
		add_treasury = { value = ca_money_small multiply = -1 }
		ca_cen_army_add = { VALUE = ca_cen_army_reorganize }
		ig:ig_armed_forces ?= { add_modifier = { name = ca_ig_approval_up months = 12 } }
	}
	ai_chance = { value = 0 }
}

	name = "ca_cen_button_conspiracy"
	desc = "ca_cen_button_conspiracy_desc"
	visible = {
		ca_frca_route_centralist = yes
		NOT = { has_variable = frca_morazan_removed }
		ca_cen_morazan_in_uca = yes
		ca_uca_button_tab = { TAB = crisis }
	}
	possible = {
		custom_tooltip = { text = ca_button_cooldown_tt NOT = { has_variable = ca_cen_conspiracy_cd } }
		NOT = { has_variable = ca_cen_coup_plot }
		ca_cen_can_conspire = yes
		ca_mm_can_pay = { VALUE = ca_money_medium }
	}
	effect = {
		set_variable = { name = ca_cen_conspiracy_cd days = 730 }
		add_treasury = { value = ca_money_medium multiply = -1 }
		hidden_effect = {
			random_list = {
				30 = {
					modifier = { if = { limit = { has_variable = ca_cen_army_converted } add = 20 } }
					modifier = { if = { limit = { ca_cen_unitarian_generals >= 2 } add = 10 } }
					modifier = { if = { limit = { ca_bloc_power_centralist >= 60 } add = 15 } }
					set_variable = { name = ca_cen_kill_result value = flag:success }
				}
				70 = {
					set_variable = { name = ca_cen_kill_result value = flag:failure }
				}
			}
		}
		trigger_event = { id = cafm_cen.3 popup = yes }
	}
	ai_chance = { value = 0 }
}

	name = "ca_cen_button_campaign"

	visible = {
		ca_frca_route_centralist = yes
		NOT = { has_variable = frca_morazan_removed }
		NOT = { has_variable = ca_cen_election_fired }
		game_date >= 1837.1.1
		ca_uca_button_tab = { TAB = gov }
	}
	possible = {
		custom_tooltip = { text = ca_button_cooldown_tt NOT = { has_variable = ca_cen_campaign_cd } }
		custom_tooltip = {
			text = ca_cen_campaign_max_tt
			OR = {
				NOT = { has_variable = ca_cen_campaign }
				var:ca_cen_campaign < 3
			}
		}
		ca_mm_can_pay = { VALUE = ca_money_small }
	}
	effect = {
		set_variable = { name = ca_cen_campaign_cd days = 120 }
		add_treasury = { value = ca_money_small multiply = -1 }
		if = {
			limit = { NOT = { has_variable = ca_cen_campaign } }
			set_variable = { name = ca_cen_campaign value = 0 }
		}
		change_variable = { name = ca_cen_campaign add = 1 }
		ig:ig_landowners ?= { add_modifier = { name = ca_ig_approval_up months = 12 } }
		ig:ig_intelligentsia ?= { add_modifier = { name = ca_ig_approval_down months = 12 } }
	}
	ai_chance = { value = 0 }
}
