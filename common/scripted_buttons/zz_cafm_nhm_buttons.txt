ca_button_nic_mediar = {
	name = "ca_button_nic_mediar"
	desc = "ca_button_nic_mediar_desc"

	visible = {
		ca_nic_is_nicaragua = yes
		ca_nic_division_activo = yes
	}

	possible = {
		custom_tooltip = {
			text = ca_button_cooldown_tt
			NOT = { has_variable = nic_mediar_cd }
		}
	}

	effect = {
		set_variable = { name = nic_mediar_cd days = 365 }
		add_treasury = -12000
		ig:ig_landowners ?= { add_modifier = { name = ca_ig_approval_up months = 24 } }
		ig:ig_intelligentsia ?= { add_modifier = { name = ca_ig_approval_up months = 24 } }
		custom_tooltip = {
			text = nic_division_down_10_tt
			change_variable = { name = nic_division subtract = 10 }
		}
	}

	ai_chance = { value = 55 }
}
