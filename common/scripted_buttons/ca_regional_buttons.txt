ca_button_talamanca_integrate = {
	name = "ca_button_talamanca_integrate"
	desc = "ca_button_talamanca_integrate_desc"
	visible = {
		has_journal_entry = je_ca_historic_situation_in_talamanca
		NOT = { has_variable = ca_talamanca_path_conquer }
		NOT = { has_variable = ca_talamanca_path_coexist }
	}
	possible = {
		NOT = { has_variable = ca_talamanca_path_integrate }
	}
	effect = {
		set_variable = ca_talamanca_path_integrate
		ca_talamanca_change = { VALUE = 5 }
	}
	ai_chance = { value = 40 }
}
