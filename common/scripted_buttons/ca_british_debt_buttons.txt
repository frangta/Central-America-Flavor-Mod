# root = país deudor

# --- Pedir a los bonistas una conversión de la deuda ---
je_ca_british_debt_negotiation_button = {
	name = "je_ca_british_debt_negotiation_button"
	desc = "je_ca_british_debt_negotiation_button_desc"

	visible = {
		exists = c:GBR
	}

	possible = {
		exists = c:GBR
		custom_tooltip = {
			text = je_ca_british_debt_negotiation_cooldown_tt
			NOT = { has_variable = ca_british_debt_negotiation_cd }
		}
		# No se renegocia el último tramo: ese hay que pagarlo
		je:je_ca_british_debt ?= {
			OR = {
				has_modifier = modifier_ca_british_debt_4
				has_modifier = modifier_ca_british_debt_3
				has_modifier = modifier_ca_british_debt_2
			}
		}
	}

	effect = {
		set_variable = { name = ca_british_debt_negotiation_cd years = 5 }
		save_scope_as = ca_debtor_scope
		c:GBR ?= {
			trigger_event = { id = ca_british_debt.3 popup = yes }
		}
	}

	ai_chance = {
		value = 15
	}
}

# --- Pago extraordinario: baja un tramo pagando con el tesoro ---
je_ca_british_debt_payment_button = {
	name = "je_ca_british_debt_payment_button"
	desc = "je_ca_british_debt_payment_button_desc"

	visible = {
		always = yes
	}

	possible = {
		gold_reserves >= 25000
		NOT = { has_variable = ca_british_debt_payment_cd }
	}

	effect = {
		add_treasury = -25000
		set_variable = { name = ca_british_debt_payment_cd years = 2 }
		ca_british_debt_reduce_tier_effect = yes
		change_relations = {
			country = c:GBR
			value = 10
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = { gold_reserves >= 75000 }
			add = 25
		}
	}
}

# --- Repudiar la deuda (hace fallar el diario) ---
je_ca_british_debt_repudiate_button = {
	name = "je_ca_british_debt_repudiate_button"
	desc = "je_ca_british_debt_repudiate_button_desc"

	visible = {
		always = yes
	}

	possible = {
		NOT = { has_modifier = modifier_ca_british_debt_repudiated }
	}

	effect = {
		add_modifier = {
			name = modifier_ca_british_debt_repudiated
			years = 20
		}
		change_relations = {
			country = c:GBR
			value = -50
		}
	}

	ai_chance = {
		value = 1
	}
}
