# ====================================================================
# CAFM - ENMIENDAS CENTROAMERICANAS
# ====================================================================
# Plantilla de cada enmienda (ver amendments.md):
#   parent           -> ley que define la postura de IG y movimientos
#   allowed_laws     -> leyes a las que puede adjuntarse
#   modifier / tax_modifier_*
#   possible         -> si se puede adjuntar
#   would_sponsor    -> root = IG; scope:approval = aprobación de la ley padre
#   ai_will_revoke / can_repeal / amendment_activism_multiplier
#
# Correcciones generales:
#  - 15 enmiendas tenían "set_variable = " vacío: rompía el parseo.
#  - possible ya no excluye la ley a la que se adjunta la enmienda.
#  - Todas tienen would_sponsor, ai_will_revoke y can_repeal.
#  - Valores bajados a la escala vanilla (aprobación ±1/±2,
#    autoridad <= 75; la enmienda vanilla más extrema da +50).
#  - Se conservan TODAS las claves (las usan scripted_effects y eventos).
# ====================================================================

# LEYES DE VAGANCIA
# Quien no prueba tener trabajo puede ser obligado a trabajar en fincas.
# --------------------------------------------------------------------
amendment_vagrancy_laws = {
	parent = law_commercialized_agriculture

	allowed_laws = {
		law_commercialized_agriculture
		law_tenant_farmers
		law_land_reform_coffee_oligarchy
		law_land_reform_hacienda_system
	}

	modifier = {
		building_group_bg_agriculture_throughput_add = 0.10
		building_group_bg_plantations_throughput_add = 0.05
		state_lower_strata_standard_of_living_add = -1
		interest_group_ig_landowners_pol_str_mult = 0.10
	}

	possible = {
		only_latin_centralamericans_cultures = yes
		NOR = {
			has_law_or_variant = law_type:law_regulatory_bodies
			has_law_or_variant = law_type:law_worker_protections
		}
	}

	would_sponsor = {
		is_interest_group_type = ig_landowners
	}

	ai_will_revoke = {
		any_interest_group = {
			is_in_government = yes
			OR = {
				is_interest_group_type = ig_trade_unions
				is_interest_group_type = ig_intelligentsia
			}
		}
	}

	can_repeal = {
		legitimacy >= legitimacy_to_repeal_amendment
	}

	amendment_activism_multiplier = 0.5
}

			has_law_or_variant = law_type:law_autocracy
			has_law_or_variant = law_type:law_oligarchy
		}
	}

	would_sponsor = {
		is_interest_group_type = ig_armed_forces
	}

	# Antes: OR = { NOT autocracia  NOT oligarquía } -> siempre verdadero.

		OR = {
			NOR = {
				has_law_or_variant = law_type:law_autocracy
				has_law_or_variant = law_type:law_oligarchy
			}
			any_interest_group = {
				is_in_government = yes
				is_interest_group_type = ig_intelligentsia
			}
		}
	}

	can_repeal = {
		legitimacy >= legitimacy_to_repeal_amendment
	}

	amendment_activism_multiplier = 0.25
}

# Decreto de Bases y Garantías (Braulio Carrillo, 1841)
amendment_ca_bases_y_garantias = {
	parent = law_autocracy

	allowed_laws = {
		law_autocracy
	}

	modifier = {
		country_authority_add = 75
		country_legitimacy_base_add = 10
		interest_group_ig_landowners_pol_str_mult = 0.20
		interest_group_ig_industrialists_pol_str_mult = 0.10
		interest_group_ig_rural_folk_pol_str_mult = -0.30
		country_institution_cost_institution_police_mult = -0.10
		country_bureaucracy_mult = 0.10
	}

	# Antes: NOT = { A B } (nunca bloqueaba) y on_activate ponía esas
	# mismas variables, invalidándose a sí misma. Ahora solo se impide
	# volver a promulgarla después de derogada.
	possible = {
		c:COS ?= this
		NOT = { has_variable = decreto_bases_garantias_derogado }
	}

	would_sponsor = {
		OR = {
			is_interest_group_type = ig_landowners
			is_interest_group_type = ig_armed_forces
			leader ?= { has_ideology = ideology:ideology_carrillismo }
		}
	}

	ai_will_revoke = {
		any_interest_group = {
			is_in_government = yes
			is_interest_group_type = ig_intelligentsia
		}
	}

	can_repeal = {
		legitimacy >= legitimacy_to_repeal_amendment
	}

	amendment_activism_multiplier = 1.0

	on_activate = {
		set_variable = decreto_bases_garantias_activo
	}

	on_deactivate = {
		remove_variable = decreto_bases_garantias_activo
		set_variable = decreto_bases_garantias_derogado
	}
}
