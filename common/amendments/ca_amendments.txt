amendment_vagrancy_laws = {
    parent = law_commercialized_agriculture  # Se añade a la agricultura comercial

    allowed_laws = {
        law_commercialized_agriculture
        law_tenant_farmers
    }

    modifier = {
        # Aumenta la eficiencia de plantaciones (Café/Banano)
        building_group_bg_agriculture_throughput_mult = 0.15
        
        # Reduce el nivel de vida de los campesinos/obreros
        state_standard_of_living_lower_strata_mult = -1.0
        
        # Aumenta el poder de los terratenientes
        interest_group_political_strength_mult_ig_landowners = 0.1
    }

    possible = {
        # Solo posible si no hay derechos laborales protegidos
        NOT = { has_law = law_type:law_regulatory_bodies }
        NOT = { has_law = law_type:law_worker_protections }
        OR = {
            country_has_primary_culture = cu:central_american
            country_has_primary_culture = cu:costa_rican
            country_has_primary_culture = cu:panamanian
            country_has_primary_culture = cu:nicaraguan
            country_has_primary_culture = cu:guatemalan
            country_has_primary_culture = cu:el_salvadorian
            country_has_primary_culture = cu:honduran
        }		
    }

    would_sponsor = {
        # Los terratenientes lo apoyan si son poderosos
        limit = {
            is_interest_group_type = ig_landowners
            is_powerful = yes
        }
        add = 50
    }
    
    amendment_activism_multiplier = 1.5 # Genera mucha controversia/activismo
}
