# 1. La UCA acepta intervenir en el estado (eventos .35-.40)
catalyst_ca_federal_aid_accepted = {
	category = cc_ca_federal_relations

	political_lobby_creation = {
		lobby_pro_federalist = {
			value = 0.2
			if = {
				limit = { liberty_desire < 25 }
				multiply = 3
			}
			else_if = {
				limit = { liberty_desire < 50 }
				multiply = 2
			}
		}

		lobby_pro_country = {
			value = 0.05
			if = {
				limit = { NOT = { has_strategic_adjacency = scope:target_country } }
				multiply = 0.5
			}

			if = {
				limit = { "relations(scope:target_country)" < relations_threshold:cordial }
				multiply = 0.5
			}
			else_if = {
				limit = { "relations(scope:target_country)" >= relations_threshold:friendly }
				multiply = 1.5
			}
		}
	}

	ai_country_goal_recalculation = {
		type = more_friendly
		chance = {
			value = 0.25

			if = {
				limit = { "relations(scope:target_country)" >= relations_threshold:friendly }
				multiply = 4.0
			}
			else_if = {
				limit = { "relations(scope:target_country)" >= relations_threshold:amicable }
				multiply = 3.0
			}
			else_if = {
				limit = { "relations(scope:target_country)" >= relations_threshold:cordial }
				multiply = 2.0
			}
		}
	}

	effect = {
		every_political_lobby = {
			limit = {
				is_political_lobby_type = lobby_pro_federalist
				target = scope:target_country
			}
			change_appeasement = {
				amount = 2
				factor = appeasement_federal_aid_received
			}
		}
		every_political_lobby = {
			limit = {
				is_political_lobby_type = lobby_anti_federalist
				target = scope:target_country
			}
			change_appeasement = {
				amount = -1
				factor = appeasement_federal_aid_received
			}
		}
	}
}
