ca_dependency = {
	groups = { general }
	texture = "gfx/interface/icons/lens_toolbar_icons/protectorate.dds"
	requires_approval = yes
	uses_random_approval = no
	can_use_obligations = no
	should_notify_third_parties = yes

	unlocking_technologies = {
	}

	selectable = {
	    is_ca_culture_country = yes
		NOT = { is_country_type = decentralized }
		is_revolutionary = no
		is_secessionist = no
		NOT = { is_subject = yes }
		NOT = { is_at_war = yes }
	}

	potential = {
		
		scope:target_country = {
			NOT = { is_country_type = decentralized }
			is_ca_culture_country = yes
			#country_rank <= rank_value:insignificant_power
			OR = {
				NOT = { is_subject = yes }
				is_subject_of = scope:actor
			}
			NOT = { is_at_war = yes }
		}
	}

	possible = {		
		
		NOT = { has_war_with = scope:target_country }
		NOT = { is_diplomatic_play_enemy_of = scope:target_country }

		
		NOT = { infamy >= 25 }
	}


    pact = {
        cost = 10
		#has_junior_participant = yes
		maintenance_paid_by = first_country
		income_transfer = 0.3
		second_country_gets_income_transfer = no
		income_transfer_based_on_second_country = yes

		max_target_involvement = 8000
		target_involvement_applies_to = source_country

        
		relations_progress_per_day = 1
        relations_improvement_max = 60


        subject_type = subject_type_ca_dependency

        second_modifier = {
				country_law_enactment_imposition_success_add = 0.1
        }
        
		actor_can_break = {
			is_diplomatic_play_committed_participant = no
		}

		target_can_break = { # Has to use diplomatic play to break
			always = no
		}

		manual_break_effect = {
			create_bidirectional_truce = { country = scope:target_country months = 60 }
		}

		auto_break_effect = {
			create_bidirectional_truce = { country = scope:target_country months = 60 }
		}
    }

	ai = {
		check_acceptance_for_will_propose = yes

		evaluation_chance = {
			value = 1
		}

		accept_score = {
			value = 0

			# Unification is a one-way door for the target, so the baseline is hostile.
			add = {
				desc = "DIPLOMATIC_ACCEPTANCE_BASE"
				value = -100
			}
			add = {
				desc = "DIPLOMATIC_ACCEPTANCE_GDP"
				value = "scope:actor.ai_gdp_comparison(root)"
				min = -10
				max = 10
			}
			if = {
					limit = {
						OR = {
							any_scope_treaty = {
								binds = scope:target_country
								any_scope_article = {
									has_type = trade_privilege
									source_country = scope:target_country
								}
							}
							scope:treaty ?= {
								any_scope_article_option = {
									has_type = trade_privilege
									source_country = scope:target_country
								}
							}
						}
					}
					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_TRADE_PRIVILEGES"
						value = 10
					}
			}
			if = {
					limit = {
						OR = {
							any_scope_treaty = {
								binds = scope:target_country
								any_scope_article = {
									has_type = alliance
								}
							}
							scope:treaty ?= {
								any_scope_article_option = {
									has_type = alliance
								}
							}
						}
					}
					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_ALLIANCE"
						value = 30
					}
			}
			else_if = {
					limit = {
						OR = {
							any_scope_treaty = {
								binds = scope:target_country
								any_scope_article = {
									has_type = defensive_pact
								}
							}
							scope:treaty ?= {
								any_scope_article_option = {
									has_type = defensive_pact
								}
							}
						}
					}
					add = {
						desc = "DIPLOMATIC_ACCEPTANCE_DEFENSIVE_PACT"
						value = 15
					}
			}			
			# Shared culture adds weight even when the cultures differ.
			if = {
				limit = {
					scope:actor.culture = scope:target_country.culture
				}
				add = {
					desc = "CA_UNIFY_ACCEPT_SHARED_RELIGION"
					value = 50
				}
			}
			# Shared religion adds weight even when the cultures differ.
			if = {
				limit = {
					scope:actor.religion = scope:target_country.religion
				}
				add = {
					desc = "CA_UNIFY_ACCEPT_SHARED_RELIGION"
					value = 10
				}
			}

			
			add = {
				desc = "DIPLOMATIC_ACCEPTANCE_RELATIONS"
				if = {
					limit = {
						"relations(scope:actor)" < relations_threshold:cordial
					}
					value = -75
				}
				if = {
					limit = {
						"relations(scope:actor)" >= relations_threshold:neutral
						"relations(scope:actor)" < relations_threshold:cordial
					}
					value = -25
				}
				if = {
					limit = {
						"relations(scope:actor)" >= relations_threshold:cordial
						"relations(scope:actor)" < relations_threshold:amicable
					}
					value = 10
				}
				if = {
					limit = {
						"relations(scope:actor)" >= relations_threshold:amicable
						"relations(scope:actor)" < relations_threshold:friendly
					}
					value = 35
				}
				if = {
					limit = {
						"relations(scope:actor)" >= relations_threshold:friendly
					}
					value = 75
				}
			}

			# Attitude
			if = {
				limit = {
					has_attitude = {
						who = scope:actor
						attitude = genial
					}
				}
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_ATTITUDE"
					value = 75
				}
			}
			if = {
				limit = {
					OR = {
						has_attitude = {
							who = scope:actor
							attitude = cooperative
						}
						has_attitude = {
							who = scope:actor
							attitude = conciliatory
						}
						has_attitude = {
							who = scope:actor
							attitude = loyal
						}
						has_attitude = {
							who = scope:actor
							attitude = protective
						}
					}
				}
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_ATTITUDE"
					value = 20
				}
			}
			if = {
				limit = {
					OR = {
						has_attitude = {
							who = scope:actor
							attitude = antagonistic
						}
						has_attitude = {
							who = scope:actor
							attitude = domineering
						}
						has_attitude = {
							who = scope:actor
							attitude = belligerent
						}
						has_attitude = {
							who = scope:actor
							attitude = rebellious
						}
					}
				}
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_ATTITUDE"
					value = -1000
				}
			}

			
			if = {
				limit = {
					is_in_same_power_bloc = scope:actor
				}
				add = {
					desc = "CA_UNIFY_ACCEPT_SAME_POWER_BLOC"
					value = 50
				}
			}

			
			add = {
				desc = "DIPLOMATIC_ACCEPTANCE_RANK"
				value = scope:actor.country_rank
				subtract = country_rank
				multiply = 15
				min = 0
				max = 75
			}

			
			if = {
				limit = {
					"scope:actor.infamy" >= infamy_threshold:infamous
					"scope:actor.infamy" < infamy_threshold:notorious
				}
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_INFAMY"
					value = -40
				}
			}
			if = {
				limit = {
					"scope:actor.infamy" >= infamy_threshold:notorious
				}
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_INFAMY"
					value = -120
				}
			}
		}

		will_propose = {
			would_accept_diplomatic_action = {
				actor = scope:target_country
				type = ca_unify_countries
			}
		}

		propose_score = {
			value = 10000
		}
	}
}
