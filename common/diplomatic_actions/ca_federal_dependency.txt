ca_federal_dependency = {
    groups = {
        general
    }
    texture = "gfx/interface/icons/lens_toolbar_icons/protectorate.dds"
    show_in_lens = no
	requires_approval = yes
    can_use_obligations = yes
	should_notify_third_parties = yes

    selectable = {
        is_ca_culture_country = yes
        NOT = { is_country_type = decentralized }
    }

    potential = {
        scope:target_country = {
            is_ca_culture_country = yes
            NOT = { is_country_type = decentralized }
            is_revolutionary = no
			
        }
    }

    pact = {
        cost = 25
        has_junior_participant = yes
        maintenance_paid_by = first_country 
        income_transfer = 0.2
        second_country_gets_income_transfer = no
        income_transfer_based_on_second_country = yes

        relations_progress_per_day = 1
        relations_improvement_max = 100

        target_requires_approval_to_break = yes

        subject_type = subject_type_ca_federal_dependency

        manual_break_effect = {
            create_bidirectional_truce = { country = scope:target_country months = 36 }
        }

        auto_break_effect = {
            create_bidirectional_truce = { country = scope:target_country months = 36 }
        }

        actor_can_break = {
            is_diplomatic_play_committed_participant = no
        }

        target_can_break = {
            is_diplomatic_play_committed_participant = no
        }
    }

    ai = {
		check_acceptance_for_will_propose = yes

		evaluation_chance = {
			value = 1
		}

		accept_score = {
			value = 0

			# Unification is a one-way door for the target, so the baseline is hostile.
			add = {
				desc = "DIPLOMATIC_ACCEPTANCE_BASE"
				value = -100
			}

			# Shared religion adds weight even when the cultures differ.
			if = {
				limit = {
					scope:actor.religion = scope:target_country.religion
				}
				add = {
					desc = "CA_UNIFY_ACCEPT_SHARED_RELIGION"
					value = 10
				}
			}

			
			add = {
				desc = "DIPLOMATIC_ACCEPTANCE_RELATIONS"
				if = {
					limit = {
						"relations(scope:actor)" < relations_threshold:cordial
					}
					value = -75
				}
				if = {
					limit = {
						"relations(scope:actor)" >= relations_threshold:neutral
						"relations(scope:actor)" < relations_threshold:cordial
					}
					value = -25
				}
				if = {
					limit = {
						"relations(scope:actor)" >= relations_threshold:cordial
						"relations(scope:actor)" < relations_threshold:amicable
					}
					value = 10
				}
				if = {
					limit = {
						"relations(scope:actor)" >= relations_threshold:amicable
						"relations(scope:actor)" < relations_threshold:friendly
					}
					value = 35
				}
				if = {
					limit = {
						"relations(scope:actor)" >= relations_threshold:friendly
					}
					value = 75
				}
			}

			# Attitude
			if = {
				limit = {
					has_attitude = {
						who = scope:actor
						attitude = genial
					}
				}
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_ATTITUDE"
					value = 75
				}
			}
			if = {
				limit = {
					OR = {
						has_attitude = {
							who = scope:actor
							attitude = cooperative
						}
						has_attitude = {
							who = scope:actor
							attitude = conciliatory
						}
						has_attitude = {
							who = scope:actor
							attitude = loyal
						}
						has_attitude = {
							who = scope:actor
							attitude = protective
						}
					}
				}
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_ATTITUDE"
					value = 20
				}
			}
			if = {
				limit = {
					OR = {
						has_attitude = {
							who = scope:actor
							attitude = antagonistic
						}
						has_attitude = {
							who = scope:actor
							attitude = domineering
						}
						has_attitude = {
							who = scope:actor
							attitude = belligerent
						}
						has_attitude = {
							who = scope:actor
							attitude = rebellious
						}
					}
				}
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_ATTITUDE"
					value = -1000
				}
			}

			
			if = {
				limit = {
					is_in_same_power_bloc = scope:actor
				}
				add = {
					desc = "CA_UNIFY_ACCEPT_SAME_POWER_BLOC" # CAFM fix
					value = 50
				}
			}

			
			add = {
				desc = "DIPLOMATIC_ACCEPTANCE_RANK"
				value = scope:actor.country_rank
				subtract = country_rank
				multiply = 15
				min = 0
				max = 75
			}

			
			if = {
				limit = {
					"scope:actor.infamy" >= infamy_threshold:infamous
					"scope:actor.infamy" < infamy_threshold:notorious
				}
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_INFAMY"
					value = -40
				}
			}
			if = {
				limit = {
					"scope:actor.infamy" >= infamy_threshold:notorious
				}
				add = {
					desc = "DIPLOMATIC_ACCEPTANCE_INFAMY"
					value = -120
				}
			}
		}

		will_propose = {
			scope:target_country = { # CAFM fix: ztr_unify_countries no existe (resto de otro mod)
				would_accept_diplomatic_action = {
					actor = root
					type = ca_federal_dependency
				}
			}
		}

		propose_score = {
			value = 10000
		}
	}
}
