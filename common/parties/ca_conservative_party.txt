ca_conservative_party = {
	color = { 108 64 179 }
	name = {
		first_valid = {
			triggered_desc = {
				desc = party_ca_conservative_uca
				trigger = { c:UCA ?= this }
			}
			triggered_desc = {
				desc = party_ca_conservative_gua
				trigger = { c:GUA ?= this }
			}
			triggered_desc = {
				desc = party_ca_conservative_els
				trigger = { c:ELS ?= this }
			}
			triggered_desc = {
				desc = party_ca_conservative_hon
				trigger = { c:HON ?= this }
			}
			triggered_desc = {
				desc = party_ca_conservative_nic
				trigger = { c:NIC ?= this }
			}
			triggered_desc = {
				desc = party_ca_conservative_cos
				trigger = { c:COS ?= this }
			}
			triggered_desc = {
				desc = party_ca_conservative_pnm
				trigger = { c:PNM ?= this }
			}
			triggered_desc = {
				desc = party_ca_conservative_generic
				trigger = { always = yes }
			}
		}
	}
	icon = { default = "gfx/interface/icons/political_parties/conservatism.dds" }

	valid_for_country = { is_ca_culture_country = yes }
	available_for_interest_group = {
		owner = { country_has_voting_franchise = yes }
	}

	join_weight = {
		value = 0
		add = { value = -5 desc = "ca_party_join_base" }

		# Ideologías de IG centrales
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_central_american_reactionary
					has_ideology = ideology:ideology_central_american_reactionary_2
					has_ideology = ideology:ideology_ca_criollo_paternalistic
					has_ideology = ideology:ideology_ca_criollo_stratocratic
					has_ideology = ideology:ideology_ca_criollo_elite
					has_ideology = ideology:ideology_ca_soberania_aristocratica
					has_ideology = ideology:ideology_ca_servile
					has_ideology = ideology:ideology_ca_moralist
					has_ideology = ideology:ideology_ca_papal_moralist
					# has_ideology = ideology:ideology_ca_conservadurismo  # pendiente: no existe aún
				}
			}
			add = { value = 12 desc = "ca_party_join_ig_core" }
		}
		# Ideologías de IG afines
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_ca_guilds
					has_ideology = ideology:ideology_ca_isolationist_expansionism
				}
			}
			add = { value = 5 desc = "ca_party_join_ig_affinity" }
		}
		# Ideologías de IG opuestas
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_ca_secularist
					has_ideology = ideology:ideology_ca_liberal
					has_ideology = ideology:ideology_central_american_federal
				}
			}
			add = { value = -10 desc = "ca_party_join_ig_hostile" }
		}
		# Ideologías de líder centrales
		if = {
			limit = {
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_ca_unitarian
						has_ideology = ideology:ideology_granadine_legitimism
						has_ideology = ideology:ideology_ca_jesuit_theocracy
						has_ideology = ideology:ideology_ca_imperial_monarchist
						has_ideology = ideology:ideology_ca_leader_autocratic_autonomist
						has_ideology = ideology:ideology_carrillismo
						has_ideology = ideology:ideology_ca_hermit_isolationism
						has_ideology = ideology:ideology_ca_mexican_annexationist
					}
				}
			}
			add = { value = 15 desc = "ca_party_join_leader_core" }
		}
		# Ideologías de líder afines
		if = {
			limit = {
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_ca_filibuster
						has_ideology = ideology:ideology_ca_peasant_caudillismo
						has_ideology = ideology:ideology_ca_tropical_bonapartism
						has_ideology = ideology:ideology_ca_morismo
						has_ideology = ideology:ideology_ca_moravian_pietism
					}
				}
			}
			add = { value = 6 desc = "ca_party_join_leader_affinity" }
		}
		# Ideologías de líder opuestas
		if = {
			limit = {
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_ca_morazanista
						has_ideology = ideology:ideology_olympus_liberalism
					}
				}
			}
			add = { value = -10 desc = "ca_party_join_leader_hostile" }
		}
		# Contexto: Iglesia y hacendados son la base natural del conservadurismo
		if = {
			limit = {
				OR = {
					is_interest_group_type = ig_devout
					is_interest_group_type = ig_landowners
				}
			}
			add = { value = 3 desc = "ca_party_join_ig_type" }
		}
		# Contexto: bajo un Estado unitario los centralistas cierran filas
		if = {
			limit = {
				has_ideology = ideology:ideology_central_american_reactionary_2
				owner = { has_law_or_variant = law_type:law_ca_central_american_unitary }
			}
			add = { value = 4 desc = "ca_party_join_unitary_context" }
		}
	}
}
