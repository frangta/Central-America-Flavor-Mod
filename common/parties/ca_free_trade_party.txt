ca_free_trade_party = {
	color = { 150 20 20 }
	name = {
		first_valid = {
			triggered_desc = {
				desc = party_ca_free_trade_uca
				trigger = { c:UCA ?= this }
			}
			triggered_desc = {
				desc = party_ca_free_trade_gua
				trigger = { c:GUA ?= this }
			}
			triggered_desc = {
				desc = party_ca_free_trade_els
				trigger = { c:ELS ?= this }
			}
			triggered_desc = {
				desc = party_ca_free_trade_hon
				trigger = { c:HON ?= this }
			}
			triggered_desc = {
				desc = party_ca_free_trade_nic
				trigger = { c:NIC ?= this }
			}
			triggered_desc = {
				desc = party_ca_free_trade_cos
				trigger = { c:COS ?= this }
			}
			triggered_desc = {
				desc = party_ca_free_trade_pnm
				trigger = { c:PNM ?= this }
			}
			triggered_desc = {
				desc = party_ca_free_trade_generic
				trigger = { always = yes }
			}
		}
	}
	icon = { default = "gfx/interface/icons/political_parties/free_trade.dds" }

	valid_for_country = { is_ca_culture_country = yes }
	available_for_interest_group = {
		owner = { country_has_voting_franchise = yes }
	}

	join_weight = {
		value = 0
		add = { value = -5 desc = "ca_party_join_base" }

		# Ideologías de IG centrales
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_ca_agroexportadores
					has_ideology = ideology:ideology_ca_coffee_liberal
					has_ideology = ideology:ideology_ca_colonialist
					has_ideology = ideology:ideology_ca_creole_caribbean
				}
			}
			add = { value = 12 desc = "ca_party_join_ig_core" }
		}
		# Ideologías de IG afines
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_ca_enclavist_pick
				}
			}
			add = { value = 5 desc = "ca_party_join_ig_affinity" }
		}
		# Ideologías de IG opuestas
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_ca_guilds
					has_ideology = ideology:ideology_ca_artesanado
					has_ideology = ideology:ideology_ca_separatist
				}
			}
			add = { value = -10 desc = "ca_party_join_ig_hostile" }
		}
		# Ideologías de líder centrales
		if = {
			limit = {
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_ca_filibuster
						has_ideology = ideology:ideology_coffee_liberalism
						has_ideology = ideology:ideology_ca_miskito_hegemony
						has_ideology = ideology:ideology_ca_canal_company_rule
						has_ideology = ideology:ideology_mercantile_transitism
						has_ideology = ideology:ideology_ca_corsair_republic
					}
				}
			}
			add = { value = 15 desc = "ca_party_join_leader_core" }
		}
		# Ideologías de líder afines
		if = {
			limit = {
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_slaver_filibusterism
						has_ideology = ideology:ideology_ca_cientifico_technocracy
						has_ideology = ideology:ideology_ca_soto_rosa_reformism
					}
				}
			}
			add = { value = 6 desc = "ca_party_join_leader_affinity" }
		}
		# Ideologías de líder opuestas
		if = {
			limit = {
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_anti_imperialist_sovereigntism
						has_ideology = ideology:ideology_ca_hermit_isolationism
						has_ideology = ideology:ideology_ca_agrarian_luddism
					}
				}
			}
			add = { value = -10 desc = "ca_party_join_leader_hostile" }
		}
		# Contexto: los exportadores presionan contra el mercantilismo y el proteccionismo
		if = {
			limit = {
				OR = {
					is_interest_group_type = ig_industrialists
					is_interest_group_type = ig_landowners
				}
				owner = {
					OR = {
						has_law_or_variant = law_type:law_mercantilism
						has_law_or_variant = law_type:law_protectionism
					}
				}
			}
			add = { value = 3 desc = "ca_party_join_trade_context" }
		}
	}
}
