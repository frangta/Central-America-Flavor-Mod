	color = { 64 173 59 }
	name = {
		first_valid = {
			triggered_desc = {
				desc = party_ca_agrarian_uca
				trigger = { c:UCA ?= this }
			}
			triggered_desc = {
				desc = party_ca_agrarian_gua
				trigger = { c:GUA ?= this }
			}
			triggered_desc = {
				desc = party_ca_agrarian_els
				trigger = { c:ELS ?= this }
			}

				desc = party_ca_agrarian_hon
				trigger = { c:HON ?= this }

				trigger = { c:NIC ?= this }

			triggered_desc = {
				desc = party_ca_agrarian_cos
				trigger = { c:COS ?= this }
			}
			triggered_desc = {
				desc = party_ca_agrarian_alt
				trigger = { c:ALT ?= this }
			}
			triggered_desc = {
				desc = party_ca_agrarian_generic
				trigger = { always = yes }
			}
		}
	}
	icon = { default = "gfx/interface/icons/political_parties/agrarians.dds" }

	valid_for_country = { is_ca_culture_country = yes }
	available_for_interest_group = {
		owner = { country_has_voting_franchise = yes }
	}

	join_weight = {
		value = 0
		add = { value = -5 desc = "ca_party_join_base" }

		# Ideologías de IG centrales
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_ca_artesanado
					has_ideology = ideology:ideology_ca_guilds
					has_ideology = ideology:ideology_law_regionalist_militia
					has_ideology = ideology:ideology_ca_comunal_milicia
					has_ideology = ideology:ideology_ca_indigenous_communalism
					has_ideology = ideology:ideology_ca_communalist
				}
			}
			add = { value = 12 desc = "ca_party_join_ig_core" }
		}
		# Ideologías de IG afines
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_ca_mutualism
				}
			}
			add = { value = 5 desc = "ca_party_join_ig_affinity" }
		}
		# Ideologías de IG opuestas
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_ca_criollo_stratocratic
					has_ideology = ideology:ideology_ca_colonialist
				}
			}
			add = { value = -10 desc = "ca_party_join_ig_hostile" }
		}
		# Ideologías de líder centrales
		if = {
			limit = {
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_ca_agrarian_luddism
						has_ideology = ideology:ideology_ca_peasant_caudillismo
						has_ideology = ideology:ideology_maya_autonomism
					}
				}
			}
			add = { value = 15 desc = "ca_party_join_leader_core" }
		}
		# Ideologías de líder afines
		if = {
			limit = {
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_ca_neo_mayan_restoration
						has_ideology = ideology:ideology_anti_imperialist_sovereigntism
						has_ideology = ideology:ideology_chacara_social_democracy
					}
				}
			}
			add = { value = 6 desc = "ca_party_join_leader_affinity" }
		}
		# Ideologías de líder opuestas
		if = {
			limit = {
				leader ?= {
					OR = {
						has_ideology = ideology:ideology_ca_canal_company_rule
					}
				}
			}
			add = { value = -10 desc = "ca_party_join_leader_hostile" }
		}
		if = {
			limit = { is_interest_group_type = ig_rural_folk }
			add = { value = 5 desc = "ca_party_join_ig_type" }
		}
	}
}
