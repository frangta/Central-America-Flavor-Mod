	icon = "gfx/interface/icons/ideology_icons/corporatist.dds"
	
	lawgroup_governance_principles = {
		law_monarchy = strongly_disapprove  # Los liberales criollos rechazaban las monarquías absolutas.
		law_presidential_republic = approve  # Apoyo a una república presidencial con fuertes características democráticas.
		law_theocracy = strongly_disapprove  # Rechazo hacia el control clerical del estado.
		law_parliamentary_republic = strongly_approve  # Apoyo fuerte a una república parlamentaria, reflejando un deseo por una democracia más representativa.
		law_council_republic = disapprove  # Desaprobación hacia sistemas de gobierno basados en consejos.
	}
	
	lawgroup_distribution_of_power = {	
		law_universal_suffrage = strongly_approve  # Apoyo fuerte al sufragio universal como parte de una democracia inclusiva.
		law_census_voting = approve  # Aprobación al sufragio censal, aunque preferían el sufragio universal.
		law_wealth_voting = disapprove  # Desaprobación hacia el voto basado en la riqueza, preferían una participación más amplia.
		law_technocracy = neutral  # Indiferencia hacia la tecnocracia, sin una postura fuerte a favor o en contra.
		law_single_party_state = strongly_disapprove  # Rechazo hacia estados de partido único, en favor de la competencia política.
		law_oligarchy = strongly_disapprove  # Rechazo hacia el gobierno oligárquico, preferían una distribución más equitativa del poder.
		law_landed_voting = disapprove  # Desaprobación hacia el voto basado en la propiedad de tierras, favor de una base de sufragio más amplia.
		law_anarchy = strongly_disapprove  # Rechazo fuerte hacia la anarquía.
		law_autocracy = strongly_disapprove  # Fuerte desaprobación hacia sistemas autocráticos.
	}
	
	lawgroup_bureaucracy = {
		law_hereditary_bureaucrats = strongly_disapprove  # Rechazo hacia burócratas hereditarios, preferían una administración basada en el mérito.
		law_appointed_bureaucrats = approve  # Aprobación hacia burócratas designados, como un paso hacia la meritocracia.
		law_elected_bureaucrats = strongly_approve  # Fuerte apoyo hacia burócratas electos, alineados con principios democráticos.
	}
	
	lawgroup_economic_system = {
        law_laissez_faire = strongly_approve  # Apoyo fuerte al laissez-faire y al libre mercado.
        law_agrarianism = disapprove  # Desaprobación hacia el agrarianismo, preferían economías más industrializadas.
		law_interventionism = disapprove  # Desaprobación hacia la intervención estatal en la economía.
		law_traditionalism = strongly_disapprove  # Rechazo hacia políticas tradicionales que limitan la modernización económica.
		law_industry_banned = strongly_disapprove  # Rechazo hacia prohibiciones de la industria, apoyando la industrialización.
		law_cooperative_ownership = neutral  # Indiferencia hacia la propiedad cooperativa, sin una postura fuerte a favor o en contra.
		law_command_economy = strongly_disapprove  # Fuerte desaprobación hacia economías dirigidas centralmente.
	}
	
	lawgroup_trade_policy = {
		law_free_trade = strongly_approve  # Apoyo fuerte hacia el libre comercio y la apertura de mercados.
		law_mercantilism = disapprove  # Desaprobación hacia el mercantilismo, en favor de políticas de comercio más liberales.
		law_isolationism = disapprove  # Desaprobación hacia el aislamiento económico, en favor de la integración global.
		law_protectionism = strongly_disapprove  # Fuerte desaprobación hacia políticas proteccionistas que limitan el comercio.
	}
	
	lawgroup_land_reform = {	
		law_commercialized_agriculture = strongly_approve  # Apoyo fuerte a la agricultura comercializada y la modernización agrícola.
		law_homesteading = approve  # Aprobación hacia la colonización de tierras y la distribución equitativa de la tierra.
		law_collectivized_agriculture = neutral  # Indiferencia hacia la agricultura colectivizada, sin una postura fuerte a favor o en contra.
		law_tenant_farmers = disapprove  # Desaprobación hacia leyes que favorecen a los arrendatarios, prefiriendo reformas más equitativas.
