
ideology_ca_criollo_elite = {
	icon = "gfx/interface/icons/ideology_icons/oligarchic.dds"
	
	lawgroup_governance_principles = {
		law_monarchy = neutral  # Las elites criollas podrían haber mostrado cierta apertura a una monarquía constitucional.
		law_presidential_republic = strongly_approve  # Preferían una república presidencial que les diera control.
		law_theocracy = neutral  # La influencia del clero era significativa, pero no necesariamente apoyaban una teocracia.
		law_parliamentary_republic = strongly_disapprove  # Preferían evitar sistemas de gobierno demasiado democráticos.
		law_council_republic = strongly_disapprove  # Similar al caso de la república parlamentaria.
	}
	
	lawgroup_distribution_of_power = {	
		law_oligarchy = strongly_approve  # Favorecían el control por una minoría privilegiada.
		law_autocracy = neutral  # Aceptaban cierta centralización del poder pero no era su preferencia principal.
		law_single_party_state = neutral  # Aceptaban el control centralizado, pero no necesariamente un partido único.
		law_technocracy = neutral  # Poca inclinación hacia la tecnocracia.
		law_anarchy = strongly_disapprove  # Rechazo fuerte hacia la anarquía.
		law_universal_suffrage = strongly_disapprove  # Fuerte desaprobación hacia el sufragio universal.
		law_census_voting = strongly_disapprove  # Similar al sufragio universal, fuerte desaprobación.
		law_wealth_voting = disapprove  # Desaprobación hacia el voto basado en la riqueza, pero menos que el sufragio universal.
		law_landed_voting = disapprove  # Preferían limitar el voto a las elites y terratenientes.
	}
	
	lawgroup_bureaucracy = {
		law_hereditary_bureaucrats = strongly_approve  # Preferían burócratas hereditarios que garantizaban su control.
		law_appointed_bureaucrats = disapprove  # Desaprobaban burócratas designados que podrían cambiar con el tiempo.
		law_elected_bureaucrats = disapprove  # Desaprobaban burócratas electos, ya que preferían el control hereditario.
	}
	
	lawgroup_economic_system = {
        law_agrarianism = strongly_approve  # Fuerte apoyo al agrarianismo que protegía los intereses terratenientes.
		law_traditionalism = neutral  # Aceptaban algunos aspectos del tradicionalismo, pero estaban abiertos a ciertas reformas.
		law_industry_banned = neutral  # No necesariamente en contra de la industria, pero preferían mantener la estructura agraria.
		law_laissez_faire = disapprove  # Desaprobación hacia la economía de mercado libre, preferían regulación.
		law_interventionism = strongly_disapprove  # Rechazaban la economía altamente intervenida.
		law_cooperative_ownership = strongly_disapprove  # Fuerte desaprobación hacia la propiedad cooperativa.
		law_command_economy = strongly_disapprove  # Desaprobaban la economía dirigida, preferían sistemas más tradicionales.
	}
	
	lawgroup_trade_policy = {
		law_mercantilism = strongly_approve  # Apoyo fuerte al mercantilismo que protegía sus intereses económicos.
		law_isolationism = neutral  # Podían aceptar cierto grado de aislamiento, pero también había interés en el comercio.
		law_protectionism = disapprove  # Menor desaprobación hacia el proteccionismo, pero no una postura fuerte.
		law_free_trade = strongly_disapprove  # Fuerte desaprobación hacia el libre comercio, preferían controlar el comercio.
	}
	
	lawgroup_land_reform = {
		law_tenant_farmers = strongly_approve  # Apoyo a los arrendatarios para mantener el control agrario.
		law_commercialized_agriculture = approve  # Apoyo a la agricultura comercializada para maximizar la producción.
		law_serfdom = neutral  # Serfdom era menos relevante en el contexto de las elites criollas.
		law_homesteading = strongly_disapprove  # Fuerte desaprobación hacia reformas que promovieran la colonización.
		law_collectivized_agriculture = strongly_disapprove  # Rechazo a la agricultura colectivizada que amenazaba el control de tierras.
	}
}

		law_interventionism = disapprove  # Desaprobación hacia la intervención estatal en la economía.

		law_industry_banned = strongly_disapprove  # Rechazo hacia prohibiciones de la industria, apoyando la industrialización.
		law_cooperative_ownership = neutral  # Indiferencia hacia la propiedad cooperativa, sin una postura fuerte a favor o en contra.
		law_command_economy = strongly_disapprove  # Fuerte desaprobación hacia economías dirigidas centralmente.
	}
	
	lawgroup_trade_policy = {
		law_free_trade = strongly_approve  # Apoyo fuerte hacia el libre comercio y la apertura de mercados.
		law_mercantilism = disapprove  # Desaprobación hacia el mercantilismo, en favor de políticas de comercio más liberales.
		law_isolationism = disapprove  # Desaprobación hacia el aislamiento económico, en favor de la integración global.
		law_protectionism = strongly_disapprove  # Fuerte desaprobación hacia políticas proteccionistas que limitan el comercio.
	}
	
	lawgroup_land_reform = {	
		law_commercialized_agriculture = strongly_approve  # Apoyo fuerte a la agricultura comercializada y la modernización agrícola.
		law_homesteading = approve  # Aprobación hacia la colonización de tierras y la distribución equitativa de la tierra.
		law_collectivized_agriculture = neutral  # Indiferencia hacia la agricultura colectivizada, sin una postura fuerte a favor o en contra.
		law_tenant_farmers = disapprove  # Desaprobación hacia leyes que favorecen a los arrendatarios, prefiriendo reformas más equitativas.
