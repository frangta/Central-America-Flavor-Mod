# central_american con la mitad de los 6 estados.

	states = {
		STATE_SAN_SALVADOR
		STATE_HONDURAS
		STATE_NICARAGUA

		has_technology_researched = nationalism
		ca_formable_consolidation_done = yes
	}

		ca_formable_ai_ready = yes
	}
