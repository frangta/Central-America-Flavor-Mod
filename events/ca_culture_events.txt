ca_culture.2 = {
	type = country_event
	placement = root
	title = ca_culture.2.t
	desc = ca_culture.2.d
	flavor = ca_culture.2.f
	event_image = { video = "unspecific_research" }
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3

	trigger = {
		has_journal_entry = je_ca_enlightenment
		NOT = { has_variable = ca_university_founded }
		OR = {
			var:ca_enl_bar >= 20
			AND = { c:ELS ?= this game_date >= 1841.2.16 }
			AND = { c:COS ?= this game_date >= 1843.4.3 }
			game_date >= 1850.1.1
		}
	}
	option = {
		name = ca_culture.2.a
		default_option = yes
		set_variable = ca_university_founded
		ca_enl_bar_change = { VALUE = 15 }
		add_treasury = { value = ca_money_small multiply = -1 }
		ig:ig_devout ?= { add_modifier = { name = ca_ig_approval_down months = 24 } }
		ig:ig_intelligentsia ?= { add_modifier = { name = ca_ig_approval_up months = 24 } }
		ai_chance = { base = 50
		modifier = { trigger = { ig:ig_intelligentsia ?= { is_in_government = yes } } add = 30 } }
	}
	option = {
		name = ca_culture.2.b
		set_variable = ca_university_founded
		ca_enl_bar_change = { VALUE = 5 }
		ig:ig_devout ?= { add_modifier = { name = ca_ig_approval_up months = 24 } }
		ai_chance = { base = 50
		modifier = { trigger = { ig:ig_devout ?= { is_in_government = yes } } add = 30 } }
	}
}
