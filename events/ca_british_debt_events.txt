# ------------------------------------------------------------
# .1 - Deuda saldada (deudor)
# ------------------------------------------------------------
ca_british_debt.1 = {
	type = country_event
	placement = root

	title = ca_british_debt.1.t
	desc = ca_british_debt.1.d
	flavor = ca_british_debt.1.f

	event_image = {
		video = "europenorthamerica_london_center"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	duration = 3

	trigger = {
		# Evento de completado de je_ca_british_debt
	}

	immediate = {
		c:GBR ?= {
			save_scope_as = britain_scope
		}
	}

	# Mantener buen crédito en Londres
	option = {
		name = ca_british_debt.1.a
		default_option = yes
		add_modifier = {
			name = modifier_ca_british_debt_good_credit
			days = very_long_modifier_time
		}
		change_relations = {
			country = c:GBR
			value = 20
		}
		ai_chance = { base = 70 } # F21
	}

	# Presumir la soberanía financiera
	option = {
		name = ca_british_debt.1.b
		add_modifier = {
			name = modifier_ca_british_debt_sovereign_finances
			days = very_long_modifier_time
		}
		ai_chance = { base = 30 } # F21
	}
}

# ------------------------------------------------------------
# .3 - El deudor pide conversión de la deuda (GBR decide)
# ------------------------------------------------------------
ca_british_debt.3 = {
	type = country_event
	placement = root

	title = ca_british_debt.3.t
	desc = ca_british_debt.3.d
	flavor = ca_british_debt.3.f

	event_image = {
		video = "africa_diplomats_negotiating"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	duration = 3

	trigger = {
		# Disparado por je_ca_british_debt_negotiation_button
	}

	# Aceptar la conversión
	option = {
		name = ca_british_debt.3.a
		default_option = yes
		scope:ca_debtor_scope = {
			trigger_event = { id = ca_british_debt.4 popup = yes }
		}
		show_as_tooltip = {
			scope:ca_debtor_scope = {
				ca_british_debt_reduce_tier_effect = yes
				ca_british_debt_obligation_effect = yes
			}
		}
		change_relations = {
			country = scope:ca_debtor_scope
			value = 20
		}
		ai_chance = {
			base = 60
		}
	}

	# Rechazar: los bonistas exigen el pago íntegro
	option = {
		name = ca_british_debt.3.b
		scope:ca_debtor_scope = {
			trigger_event = { id = ca_british_debt.5 popup = yes }
		}
		change_relations = {
			country = scope:ca_debtor_scope
			value = -20
		}
		ai_chance = {
			base = 40
		}
	}
}
