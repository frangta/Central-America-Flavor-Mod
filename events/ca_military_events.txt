ca_military.3 = {
	type = country_event
	placement = root
	title = ca_military.3.t
	desc = ca_military.3.d
	flavor = ca_military.3.f
	event_image = { video = "unspecific_military_parade" }
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3

	trigger = {
		has_journal_entry = je_ca_military_reform
		NOT = { has_variable = ca_military_academy }
		OR = {
			var:ca_military_bar >= 20
			game_date >= 1870.1.1
		}
	}
	option = {
		name = ca_military.3.a
		default_option = yes
		set_variable = ca_military_academy
		set_variable = ca_foreign_instructors
		add_treasury = { value = ca_money_medium multiply = -1 }
		ca_military_bar_change = { VALUE = 20 }
		ig:ig_armed_forces ?= { add_modifier = { name = ca_ig_approval_down months = 24 } }
		ai_chance = { base = 50
		modifier = { trigger = { gold_reserves < ca_money_medium } factor = 0 } }
	}
	option = {
		name = ca_military.3.b
		set_variable = ca_military_academy
		ca_military_bar_change = { VALUE = 10 }
		ig:ig_armed_forces ?= { add_modifier = { name = ca_ig_approval_up months = 24 } }
		ai_chance = { base = 50 }
	}
}
