namespace = ca_stronghold_events

# CAFM P11: ciclo de los bastiones (strongholds) en los estados de un pais.
# Federal y unitario: surge (.1/.2) -> se consolida (.3/.4) -> se fortalece (.5/.6) -> mantenimiento (.7/.8)
# -> decae (.9/.10) -> resurge (.11/.12). Liberal (.13), servil (.14), conservador (.15) y separatista (.16).
# Pulso anual ca_p11_yearly (zz_ca_p11_on_actions.txt).

ca_stronghold_events.1 = {
	type = country_event
	placement = scope:ca_p11_state
	title = ca_stronghold_events.1.t
	desc = ca_stronghold_events.1.d
	flavor = ca_stronghold_events.1.f
	event_image = { video = "southamerica_election" }
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3
	cooldown = { days = normal_modifier_time }

	trigger = {
		ca_p11_country = yes
		ca_p11_movement_ge = { TYPE = movement_ca_federalist VALUE = 0.05 }
		any_scope_state = {
			is_incorporated = yes
			NOT = { ca_p11_state_has_fed_sh = yes }
			NOT = { has_modifier = waning_federal_stronghold }
			NOT = { ca_p11_state_has_uni_sh = yes }
		}
	}
	cancellation_trigger = {
		NOT = { exists = scope:ca_p11_state }
	}

	immediate = {
		random_scope_state = {
			limit = {
				is_incorporated = yes
				NOT = { ca_p11_state_has_fed_sh = yes }
				NOT = { has_modifier = waning_federal_stronghold }
				NOT = { ca_p11_state_has_uni_sh = yes }
			}
			save_scope_as = ca_p11_state
		}
	}
	option = {
		name = ca_stronghold_events.1.a
		default_option = yes
		scope:ca_p11_state = {
			add_modifier = { name = federal_stronghold_small_nation }
			set_variable = { name = ca_sh_fed_grow days = 730 }
		}
		ai_chance = { base = 60 }
	}
	option = {
		name = ca_stronghold_events.1.b
		scope:ca_p11_state = {
			add_radicals_in_state = { value = small_radicals strata = middle }
		}
		ai_chance = { base = 40 }
	}
}
