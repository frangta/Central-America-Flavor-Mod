ca_movement_events.1 = {
	type = country_event
	placement = root
	title = ca_movement_events.1.t
	desc = ca_movement_events.1.d
	flavor = ca_movement_events.1.f
	event_image = { video = "southamerica_aristocrats" }
	icon = "gfx/interface/icons/event_icons/event_protest.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3
	cooldown = { days = normal_modifier_time }

	trigger = {
		ca_p11_country = yes
		ca_p11_movement_ge = { TYPE = movement_ca_federalist VALUE = 0.03 }
	}
	cancellation_trigger = {
		NOT = { exists = scope:ca_p11_movement }
	}

	immediate = {
		random_political_movement = {
			limit = { is_political_movement_type = movement_ca_federalist }
			save_scope_as = ca_p11_movement
		}
	}
	option = {
		name = ca_movement_events.1.a
		default_option = yes
		scope:ca_p11_movement ?= { add_modifier = { name = ca_modifier_pop_attraction_low months = 24 } }
		ig:ig_intelligentsia ?= { add_modifier = { name = ca_p11_ig_emboldened months = 24 } }
		ai_chance = { base = 50 }
	}
	option = {
		name = ca_movement_events.1.b
		scope:ca_p11_movement ?= { add_modifier = { name = ca_modifier_radicalism_low months = 24 } }
		add_radicals = { value = small_radicals strata = middle }
		ai_chance = { base = 30 }
	}
	option = {
		name = ca_movement_events.1.c
		scope:ca_p11_movement ?= { add_modifier = { name = ca_modifier_pop_attraction_low months = 12 } }
		ig:ig_intelligentsia ?= { add_modifier = { name = ca_p11_ig_coopted months = 24 } }
		ig:ig_landowners ?= { add_modifier = { name = ca_p11_ig_humiliated months = 12 } }
		ai_chance = { base = 20 }
	}
}
