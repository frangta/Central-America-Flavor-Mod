## .2 — El Estado dividido (reutiliza ca_nicaragua_events.2.*)
## Lo dispara ca_play_nicaragua_effect 7 días después de crear NIC (eventos .1 y ca_central_america.80).
ca_nicaragua_events.2 = {
	type = country_event
	placement = ROOT
	title = ca_nicaragua_events.2.t
	desc = ca_nicaragua_events.2.d
	flavor = ca_nicaragua_events.2.f
	duration = 3
	event_image = { video = "southamerica_aristocrats" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/southamerica/aristocrats"
	icon = "gfx/interface/icons/event_icons/event_fire.dds"

	trigger = {
		c:NIC ?= this
		game_date < 1837.1.1
		NOT = { has_variable = nic_e2_fired }
	}

	immediate = {
		set_variable = nic_e2_fired
		if = {
			limit = { c:UCA ?= { has_variable = historic_central_american } }
			set_variable = nic_historic_mode
		}
		s:STATE_NICARAGUA.region_state:NIC ?= {
			add_modifier = { name = nic_agitacion_aristocratica months = 60 }
		}
	}

	# Histórica: Zepeda gobierna desde León contra los caudillos
	option = {
		name = ca_nicaragua_events.2.a
		default_option = yes
		add_modifier = { name = nic_gobierno_desde_leon months = 60 }
		add_radicals = { value = small_radicals pop_type = aristocrats }
		ai_chance = { base = 80 }
	}

	# Alternativa: pedir tutela federal (sube el apoyo federal en la Fase 2)
	option = {
		name = ca_nicaragua_events.2.b
		add_modifier = { name = nic_tutela_federal months = 36 }
		set_variable = nic_pidio_federacion
		c:UCA ?= { change_relations = { country = ROOT value = 20 } }
		ai_chance = { base = 20 }
	}
}

		c:NIC ?= this
		game_date >= 1836.4.1

		ruler ?= { has_variable = is_zepeda }
		NOT = { has_variable = nic_e6_fired }
	}

	}

		set_variable = nic_zepeda_reforms

		ig:ig_landowners ?= { add_modifier = { name = ca_ig_approval_down months = 24 } }

		ai_chance = { base = 90 }
	}

	# Alternativa: el pacto granadino (sin conspiración; Zepeda no corre peligro)

		name = ca_nicaragua_events.6.b
		set_variable = nic_pacto_granadino

		ig:ig_landowners ?= { add_modifier = { name = ca_ig_approval_up months = 24 } }
		ai_chance = { base = 10 }
	}

## .13 — La noche del cuartel (25 ene 1837, o barra = 100)
## Modo histórico + fecha alcanzada: Zepeda muere siempre.
## Barra a 100 antes de la fecha, o modo aleatorio: 70 % muere / 30 % sobrevive.
ca_nicaragua_events.13 = {
	type = country_event
	placement = ROOT
	title = ca_nicaragua_events.13.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = ca_nicaragua_events.13.d_triunfa
				trigger = { has_variable = nic_golpe_triunfa }
			}
			triggered_desc = {
				desc = ca_nicaragua_events.13.d_fracasa
				trigger = { always = yes }
			}
		}
	}
	flavor = ca_nicaragua_events.13.f
	duration = 3
	event_image = { video = "southamerica_war_civilians" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_skull.dds"

	trigger = {
		c:NIC ?= this
		has_journal_entry = je_ca_nic_conspiracy
		NOT = { has_variable = nic_e13_fired }
	}

	immediate = {
		set_variable = nic_e13_fired
		set_variable = { name = nic_conspiracy value = 100 }
		if = {
			limit = {
				has_variable = nic_historic_mode
				game_date >= 1837.1.25
			}
			set_variable = nic_golpe_triunfa
		}
		else = {
			random_list = {
				70 = { set_variable = nic_golpe_triunfa }
				30 = { set_variable = nic_golpe_fracasa }
			}
		}
	}

	option = {
		name = ca_nicaragua_events.13.a
		default_option = yes
		if = {
			limit = { has_variable = nic_golpe_triunfa }
			custom_tooltip = nic_golpe_triunfa_tt
			if = {
				limit = { ruler ?= { has_variable = is_zepeda } }
				set_variable = nic_zepeda_murdered
				# El registro anti-duplicados pasa a ser una bandera: el puntero al muerto daría errores
				remove_global_variable = cafm_hc_zepeda
				set_global_variable = cafm_zepeda_dead
				ruler = { kill_character = yes }
				trigger_event = { id = ca_nicaragua_events.14 days = 1 }
			}
		}
		else = {
			custom_tooltip = nic_golpe_fracasa_tt
			set_variable = nic_zepeda_survived
			if = {
				limit = { ruler ?= { has_variable = is_zepeda } }
				ruler = { add_modifier = { name = nic_heroe_nacional } }
			}
			add_modifier = { name = nic_paz_institucional months = 60 }
			ig:ig_landowners ?= { add_modifier = { name = nic_terratenientes_en_desgracia months = 60 } }
		}
		ai_chance = { base = 100 }
	}
}
