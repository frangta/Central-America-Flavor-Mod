ca_nicaragua_events.01 = {
	type = country_event
	placement = s:STATE_NICARAGUA
	title = ca_nicaragua_events.01.t
	desc = ca_nicaragua_events.01.d
	flavor = ca_nicaragua_events.01.f
	duration = 3
	event_image = {
		video = "gfx/event_pictures/southamerica_aristocrats.bk2"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/southamerica/aristocrats"
	icon = "gfx/interface/icons/event_icons/event_fire.dds"
	trigger = {
        exists = c:UCA
        this = c:UCA 
        game_date <= 1836.1.15
	}
	
	immediate = {
        s:STATE_NICARAGUA = {
        add_state_trait = state_trait_mining_nic
		add_modifier = { name = ca_uca_leones_vs_granadinos }
        add_modifier = { name = separatism_stronghold }
        add_modifier = { name = liberal_stronghold_small_nation }
        add_modifier = { name = conservative_stronghold_small_nation }
        }
	}

	option = {
		name = ca_nicaragua_events.01.a
		default_option = yes

	}
	option = {
		name = ca_nicaragua_events.02.a
		trigger = {
			c:UCA ?= {
			has_variable = ca_state_selection_to_playing
			}
		}
        ca_create_nicaragua_effect = yes 
		ca_play_nicaragua_effect = yes
        play_as = c:NIC
	}
}

# 1. 1837: José Núñez y el fin de la Federación
ca_nicaragua_events.19 = {
    type = country_event
    placement = ROOT
    title = ca_nicaragua_events.19.t
    desc = ca_nicaragua_events.19.d
    icon = "gfx/interface/icons/event_icons/event_election.dds"
    duration = 3

    trigger = {
        c:NIC ?= this
        in_election_campaign = yes
        NOT = { has_variable = nic_elec_1837_fired }
        game_date >= 1837.1.1
        game_date <= 1838.1.1
    }
    
    immediate = { 
        set_variable = { name = nic_elec_1837_fired value = yes } 
    }

    # Opción A (Histórica): José Núñez asume como moderado de León
    option = {
        name = ca_nicaragua_events.19.a
        default_option = yes
        ig:ig_intelligentsia ?= { party = { add_momentum = momentum_medium } }
        ig:ig_petty_bourgeoisie ?= { party = { add_momentum = momentum_medium } }
    }

    # Opción B (Alternativa): Los conservadores de Granada imponen su fuerza
    option = {
        name = ca_nicaragua_events.19.b
        ig:ig_landowners ?= { party = { add_momentum = momentum_large } }
        add_radicals = { value = small_radicals pop_type = academics }
    }
}
