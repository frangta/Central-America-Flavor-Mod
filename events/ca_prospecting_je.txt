je_national_geological_survey = {
    icon = "gfx/interface/icons/event_icons/event_map.dds"
    
    # CONEXIÓN CON LA BARRA DE PROGRESO
    progressbar = ca_prospecting_bar
    
    # ¿Quién puede ver y activar esto?
    possible = {
        is_ca_culture_country = yes # Usamos el trigger del Paso 1
        has_technology_researched = prospecting # Requiere tecnología de Prospección

    # ¿Cuándo se considera activa? (Mismas condiciones)

    }
    
    # CONDICIÓN DE VICTORIA
    complete = {
        # Se completa cuando la variable llega a 100
        has_variable = ca_survey_progress
        var:ca_survey_progress >= 100
    }

        # O disparas un evento de celebración
        trigger_event = { id = ca_prospecting.100 }
    }
    
    weight = 100
}
