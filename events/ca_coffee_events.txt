	duration = 3

		name = ca_coffee_events.1.b
		set_variable = ca_coffee_1_fired
		ca_coffee_bar_change = { VALUE = -5 }
		ig:ig_landowners ?= { add_modifier = { name = ca_ig_approval_down months = 36 } }
