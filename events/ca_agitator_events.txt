namespace = ca_agitator_events

# CAFM P11: nacen agitadores federalistas (.1-.4) y unitarios (.5-.8).
# Disparo: pulso mensual ca_p11_monthly. Maximo 2 agitadores por ideologia en el pais.

ca_agitator_events.1 = {
	type = country_event
	placement = root
	title = ca_agitator_events.1.t
	desc = ca_agitator_events.1.d
	flavor = ca_agitator_events.1.f
	event_image = { video = "europenorthamerica_rich_and_poor" }
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3
	cooldown = { days = 1825 }

	trigger = {
		ca_p11_country = yes
		ca_p11_movement_ge = { TYPE = movement_ca_federalist VALUE = 0.05 }
		ig:ig_intelligentsia ?= { is_marginal = no }
		NOT = {
			any_scope_character = {
				has_role = character_role_agitator
				has_ideology = ideology:ideology_ca_federalist
				count >= 2
			}
		}
	}

	immediate = {
		save_scope_as = ca_p11_country_scope
	}
	option = {
		name = ca_agitator_events.1.a
		default_option = yes
		create_character = {
			is_agitator = yes
			interest_group = ig_intelligentsia
			ideology = ideology_ca_federalist
			traits = { inspirational_orator }
		}
		random_political_movement = {
			limit = { is_political_movement_type = movement_ca_federalist }
			add_modifier = { name = ca_modifier_pop_attraction_low months = 12 }
		}
		ai_chance = { base = 60 }
	}
	option = {
		name = ca_agitator_events.1.b
		ig:ig_intelligentsia ?= { add_modifier = { name = ca_p11_ig_humiliated months = 12 } }
		add_radicals = { value = small_radicals interest_group = ig:ig_intelligentsia }
		ai_chance = { base = 40 }
	}
}
