# CAFM P3 (punto 3): "El Congreso propone...". Leyes federalistas que el Congreso abre por evento.
# Las dispara ca_fed_congress_monthly (zz_ca_p3_effects.txt) desde el pulso del diario de la Federacion.
# La paralisis federal sigue bloqueando al jugador y a la IA; solo estos eventos abren leyes.
# Los eventos de promulgacion .6-.14 siguen saliendo durante la ley.

ca_federalist_events.21 = {
	type = country_event
	placement = root
	title = ca_federalist_events.21.t
	desc = ca_federalist_events.21.d
	flavor = ca_federalist_events.21.f
	event_image = { video = "asia_politician_parliament_motion" }
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3

	trigger = {
		ca_fed_congress_can_propose = yes
		NOT = { has_law = law_type:law_ca_escuela_laica }
	}

	# a. Presentar al Congreso
	option = {
		name = ca_federalist_events.21.a
		default_option = yes
		ca_fed_congress_present = { LAW = law_ca_escuela_laica }
		ai_chance = {
			base = 60
			modifier = { trigger = { ig:ig_intelligentsia ?= { is_in_government = yes } } add = 20 }
		}
	}
	# b. Negociar con los Estados (cohesion >= 50; cuesta 15)
	option = {
		name = ca_federalist_events.21.b
		trigger = { ca_cohesion_ge = { VALUE = 50 } }
		ca_fed_congress_states_pact = { LAW = law_ca_escuela_laica }
		ai_chance = {
			base = 30
			modifier = { trigger = { ca_cohesion_ge = { VALUE = 65 } } add = 30 }
		}
	}
	# c. Aplazar (no vuelve en 3 anos)
	option = {
		name = ca_federalist_events.21.c
		ca_fed_congress_postpone = { N = 21 }
		ig:ig_intelligentsia ?= { add_modifier = { name = ca_ig_approval_down months = 12 } }
		ai_chance = {
			base = 10
			modifier = { trigger = { ig:ig_landowners ?= { is_powerful = yes } } add = 20 }
		}
	}
}
