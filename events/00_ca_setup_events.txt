				    }

			}
			
			create_country = {
				tag = DG6

				province = p:xA087F6

				   }
				  }

	desc = ca_setup_events.1_event.d
	flavor = ca_setup_events.1_event.f
	
	event_image = {
		video = ""
	}
	
	icon = "gfx/interface/icons/event_icons/event_default.dds"
	
	on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	duration = 3
	
	trigger = {
	    is_subject_of = c:UCA
		exists = c:COS
		this = c:COS 
	}
	
	immediate = {
	}
	
	option = {
		 name = ca_setup_events.1.#.a
		 default_option = yes
		create_character = {
		   template = cos_braulio_carrillo
            ruler = yes
            save_scope_as = cos_braulio_carrillo_scope		
		}
	    create_diplomatic_pact = {
				country = scope:scope_RDT
				type = puppet
			}
	}
	

}
