		video = "gfx/event_pictures/southamerica_aristocrats.bk2"

        exists = c:UCA

		name = ca_costa_rica_events.1.a
		default_option = yes

	option = {
		name = ca_costa_rica_events.1.b
		trigger = {
			c:UCA ?= {
			has_variable = ca_state_selection_to_playing
			}
		}
        ca_create_costarica_effect = yes
		ca_play_costarica_effect = yes

	}

# Evento 11: La Maniobra Maestra de Carrillo
ca_costa_rica_events.11 = {
    type = country_event
    placement = root
    
    title = ca_costa_rica_events.11.t
    desc = ca_costa_rica_events.11.d
    flavor = ca_costa_rica_events.11.f
    
    icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    
    trigger = {
        always = no
    }
    
    option = {
        name = ca_costa_rica_events.11.a
        # Opción A (Histórica): Ascensos y honores (Anexión pacífica)
        default_option = yes
        
        # Anexar a Guanacaste (PC2) pacíficamente
        if = {
            limit = { exists = c:PC2 }
            annex = c:PC2
        }
        
        # Bono de lealtad por la vía diplomática
        add_loyalists = { value = 0.10 culture = cu:costa_rican }
        
        # Nicaragua pierde confianza
        if = {
            limit = { exists = c:NIC }
            c:NIC = {
                add_modifier = { name = ca_federal_paralysis months = 24 } # Simula la desconfianza/caos interno en León
            }
        }
    }
    
    option = {
        name = ca_costa_rica_events.11.b
        # Opción B (La Purga): Ejecutar a los líderes
        
        if = {
            limit = { exists = c:PC2 }
            annex = c:PC2
        }
        
        # Mucha autoridad, pero muchos radicales
        add_modifier = { name = ca_add_authority months = 36 }
        add_radicals = { value = 0.15 culture = cu:costa_rican }
        
        # Nicaragua se enfurece y gana un reclamo/motivo de guerra
        if = {
            limit = { exists = c:NIC }
            change_relations = { country = c:NIC value = -50 }
            c:NIC = { add_claim = s:STATE_COSTA_RICA }
        }
    }
    
    option = {
        name = ca_costa_rica_events.11.c
        # Opción C (Amnistía Silenciosa): Sin teatro político
        
        if = {
            limit = { exists = c:PC2 }
            annex = c:PC2
        }
        
        # Modificador de vulnerabilidad fronteriza
        add_modifier = { name = ca_funding_separatists_modifier months = 60 }
    }
}

ca_costa_rica_events.99 = {
    type = country_event
    placement = root
    
    title = ca_costa_rica_events.99.t
    desc = ca_costa_rica_events.99.d
    flavor = ca_costa_rica_events.99.f
    
    icon = "gfx/interface/icons/event_icons/event_military.dds"
    
    trigger = {
        always = no # Disparado por el on_fail del Diario
    }
    
    option = {
        name = ca_costa_rica_events.99.a
        # "¡Hemos perdido el Norte!"
        
        # Guanacaste (PC2) rompe lazos
        if = {
            limit = { 
                exists = c:PC2
                c:PC2 = { is_subject_of = c:COS }
            }
            c:PC2 = { make_independent = yes }
            
            # Guanacaste se alía con Nicaragua si existe
            if = {
                limit = { exists = c:NIC }
                
                create_treaty = {
                    first_country = c:PC2
                    second_country = c:NIC
                    is_draft = no
                    binding_period = { years = 5 }
                    articles_to_create = {
                        { article = alliance }
                    }
                }
                s:STATE_COSTA_RICA = { add_claim = c:NIC } 
            }
        }
        
        # Penalización masiva de prestigio y radicales
        add_prestige = -20
        add_radicals = { value = 0.20 culture = cu:costa_rican }
    }
}

			change_relations = { country = c:UCA value = 50 }
		}

	}

	immediate = {

	}

	# Historica: fusilamiento en San Jose, 15 de septiembre de 1842

			limit = { ruler ?= { has_variable = is_morazan } }
