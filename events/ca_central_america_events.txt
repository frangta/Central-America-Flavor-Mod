## A local newspaper proposed the formation of a state...
ca_central_america.1 = { 
	type = country_event
	placement = s:STATE_GUATEMALA	
	title = ca_central_america.1.t
	desc = ca_central_america.1.d
	flavor = ca_central_america.1.f
	
	event_image = {
		video = "southamerica_aristocrats"
	}
	
	icon = "gfx/interface/icons/event_icons/event_default.dds"
	
	on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	duration = 3
	
	trigger = {
		game_date = 1836.5.1
		exists = c:UCA
		this = c:UCA 
	}
	
	option = {
		name = ca_central_america.1.a #support the creation of a new state
		default_option = yes
		set_variable = more_liked_ALT
        change_variable = { name = frca_support add = 2 }
        add_loyalists_in_state = { value = 0.1 }		
		ai_chance = {
			base = 50
		}
	}
	option = {
		name = ca_central_america.1.b #We have to stop the owner of this newspaper
		trigger = {
			c:UCA = {
				is_enacting_law = law_type:law_censorship
			}
		}
		scope:ca_guatemala_1836_1838 = {
			set_variable = censorship_liked_ALT
		}	
	}	
	option = {
		name = ca_central_america.1.c #Ignore the situation in Guatemala
		scope:ca_guatemala_1836_1838 = {
			set_variable = ignore_ALT
		}
        change_variable = { name = frca_support subtract = 5 }
        # Aumenta el radicalismo regional
        add_radicals_in_state = { value = 0.15 }		
	}
}

	}

	duration = 3
	
	trigger = {
                game_date >= 1836.1.1
		OR = {		
			AND = {
				this = c:UCA 
				p:x47C5F8 = { owner = c:UCA }
			}
			AND = {
				this = c:COS 
				p:x47C5F8 = { owner = c:COS }
			}
		}
               NOT = { has_variable = bocas_del_toro_crisis_fired }
	}
	
immediate = { 
		set_variable = bocas_del_toro_crisis_fired
	}
	
	option = {
		name = ca_central_america.6.a # Proclaim authority
		default_option = yes
		create_country = {
			tag = PP1
			origin = ROOT # Usa ROOT para que funcione sin importar si es UCA o COS
			province = p:x47C5F8
			on_created = {
				save_scope_as = scope_PP1
				set_country_type = decentralized
				set_variable = ca_bocas_de_toro_country	
				create_character = {
					first_name = "John"
					last_name = "Gallagher"
					historical = yes
					is_general = yes
					interest_group = ig_armed_forces
					ideology = ideology:ideology_federalist # Asegúrate de que esta ideología exista en tu mod
					ruler = yes
					ig_leader = yes			
					traits = { explorer }
				}
			}	
		}	
		# Pacto diplomático con el creador
		create_diplomatic_pact = {
			country = scope:scope_PP1
			type = ca_dependency
		}
		set_variable = proclaim_pp1_uca
		
		# Le avisa a Colombia
		if = {
			limit = { exists = c:CLM }
			c:CLM = {		
				trigger_event = { 
					id = ca_central_america.12
					months = 6
				}
			}
		}
		ai_chance = { base = 33 }
	}
	
	option = {
		name = ca_central_america.6.b # Let Britain's influence grow
		set_owner_of_provinces = { country = c:GBR provinces = { x47C5F8 } } # Corregido al HEX correcto de Bocas

			limit = { exists = c:GBR }
			c:GBR = { trigger_event = { id = ca_central_america.7 } }
		}
		ai_chance = { base = 33 }
	}	
	
	option = {
		name = ca_central_america.6.c # Abandon ambitions
		# Quizás aquí deberíamos darle la provincia directamente a CLM o dejarla descentralizada
		ai_chance = { base = 33 }
	}

ca_central_america.13 = {
	type = country_event
	placement = ROOT
	title = ca_central_america.13.t
	desc = ca_central_america.13.d
	flavor = ca_central_america.13.f
	duration = 3
	event_image = { video = "ep1_transfer_of_authority" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_fire.dds"
	
	trigger = {
		this = c:UCA
	}

	immediate = {
		
		capital = {
			state_region = { save_scope_as = capital_region_scope }
		}

		# --- SOCONUSCO (CIP) ---
		if = {
			limit = {
				exists = s:STATE_CHIAPAS.region_state:UCA
				NOT = { scope:capital_region_scope = s:STATE_CHIAPAS }
			}			
			create_country = {
				tag = CIP
				origin = ROOT				
				province = p:x1030C0 
				province = p:x4DBDE9 
				province = p:xD03040 
				province = p:xA087F6
				province = p:x8E26C5
				on_created = {
					set_variable = { name = ca_secession_country }
					save_scope_as = scope_CIP
				}
			}
		}	
			
		# --- GUATEMALA (GUA) sin Soconusco ---
		if = {
			limit = {
				exists = s:STATE_GUATEMALA.region_state:UCA
				NOT = { scope:capital_region_scope = s:STATE_GUATEMALA }
				NOT = { has_variable = guatemala_integrate_soconusco }
			}
			s:STATE_GUATEMALA = {
				add_homeland = cu:guatemalan
				remove_homeland = cu:central_american	
			}			
			create_country = {
				tag = GUA
				origin = ROOT
				state = s:STATE_GUATEMALA.region_state:UCA
				province = p:x31E49A 
				on_created = {
					set_variable = { name = ca_secession_country }
					save_scope_as = scope_GUA
					add_primary_culture = cu:guatemalan
					remove_primary_culture = cu:central_american
				}
			}
		}

		# --- GUATEMALA (GUA) con Soconusco ---
		if = {
			limit = {
				exists = s:STATE_GUATEMALA.region_state:UCA
				NOT = { scope:capital_region_scope = s:STATE_GUATEMALA }
				has_variable = guatemala_integrate_soconusco
			}
			s:STATE_GUATEMALA = {
				add_homeland = cu:guatemalan
				remove_homeland = cu:central_american	
			}
			create_country = {
				tag = GUA
				origin = ROOT
				state = s:STATE_GUATEMALA.region_state:UCA
				on_created = {
					set_variable = { name = ca_secession_country }
					save_scope_as = scope_GUA
					add_primary_culture = cu:guatemalan
					remove_primary_culture = cu:central_american
				}
			}
			if = {
				limit = { exists = s:STATE_CHIAPAS.region_state:UCA }
				s:STATE_CHIAPAS.region_state:UCA = { set_state_owner = scope:scope_GUA }
			}
		}

		# --- LOS ALTOS (ALT) con Soconusco ---
		if = {
			limit = {
				has_variable = quetzaltenango_integrate_soconusco
