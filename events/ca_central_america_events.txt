		NOT = { has_variable = nic_civil_peace_var }
	}

	immediate = {
		set_variable = nic_civil_war_var
		random_scope_state = {
			limit = {
				is_capital = no
				state_region = s:STATE_NICARAGUA
				turmoil > 0.2
			}
			save_scope_as = secession_state
		}
		create_country = {
			tag = NIC
			origin = this
			state = scope:secession_state
		}
		create_diplomatic_pact = {
			country = c:NIC
			type = ca_autonomous_province
		}		
		c:NIC = {
			save_scope_as = scope_NIC
			save_scope_as = secession_country
			capital = {
				save_scope_as = secession_capital
			}
			set_relations = { country = c:UCA value = -50 }
			create_character = {
				first_name = jose
				last_name = nunez
				historical = yes
				ig_leader = yes
				interest_group = ig_armed_forces
				ruler = yes
				birth_date = 1800.03.03
				female = no
				culture = cu:central_american
				ideology = ideology_reformer
				religion = rel:catholic
				traits = {
					basic_diplomat
				}
			}
			ig:ig_landowners = {
				remove_ideology = ideology_paternalistic
				add_ideology = ideology_central_american_reactionary
				set_interest_group_name = ig_hacendados
			}
			ig:ig_armed_forces = {
				remove_ideology = ideology_paternalistic
			}
			ig:ig_devout = {
				add_ideology = ideology_central_american_reactionary
				set_interest_group_name = ig_catholic_church
			}
			activate_law = law_type:law_freedom_of_conscience
			activate_law = law_type:law_mercantilism
			activate_law = law_type:law_wealth_voting
			it = {
				limit = { has_variable = historic_central_american }
				set_variable = {
					name = nic_civil_war_var
					days = 90
				}
			}
			it = {
				limit = { has_variable = random_central_american }
				random_events = {
					50 = ca_central_america.27
					50 = ca_central_america.28
				}
			}
		}
	}

	option = {
		name = central_america.4.a
		default_option = yes
		scope:secession_country = {
			change_relations = {
				country = ROOT
