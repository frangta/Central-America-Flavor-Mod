ca_central_america.1 = { 
    type = country_event
    placement = scope:guatemala_state # Ahora apunta al scope correcto    
    
    title = ca_central_america.1.t
    desc = ca_central_america.1.d
    flavor = ca_central_america.1.f
    
    event_image = {
        video = "southamerica_aristocrats"
    }
    
    
    icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
    
    on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    
    duration = 3
    
    trigger = {
        game_date >= 1836.5.1
        exists = c:UCA
        this = c:UCA 
       
        NOT = { has_global_variable = ca_los_altos_proposal_fired }
    }

    
    immediate = {
        set_global_variable = ca_los_altos_proposal_fired
        
        
        s:STATE_GUATEMALA = {
            region_state:UCA = {
                save_scope_as = guatemala_state
            }
        }
        
        c:GUA ?= { save_scope_as = guatemala_country }
    }
    
   
    option = {
        name = ca_central_america.1.a 
        default_option = yes
        
        # Guardamos la variable en el país sujeto de Guatemala (si existe) o en la UCA
        scope:guatemala_country ?= { set_variable = more_liked_ALT }
        
        if = {
            limit = { has_variable = frca_support }
            custom_tooltip = { text = frca_support_1a set_bar_progress = { value = 2 name = frca_support_bar } }
        }
        
       
        scope:guatemala_state = {
                                
                                                     
            add_loyalists = {
                value = 0.1
                culture = cu:central_american
            }
        }        
        
        ai_chance = { base = 50 }
    }

    
    option = {
        name = ca_central_america.1.b 
        trigger = {
            
            OR = {
                has_law_or_variant = law_type:law_censorship
                has_law_or_variant = law_type:law_outlawed_dissent
                is_enacting_law = law_type:law_censorship
            }
        }
        
        scope:guatemala_country ?= { set_variable = censorship_liked_ALT }   
        
        
        scope:guatemala_state = {
            add_modifier = {
                name = ca_newspaper_censored_modifier
                months = 12
            }
        }
        ig:ig_intelligentsia = { add_modifier = { name = ca_ig_approval_down months = 12 } }
    }   


    option = {
        name = ca_central_america.1.c 
        
        scope:guatemala_country ?= { set_variable = ignore_ALT }
        
        if = {
            limit = { has_variable = frca_support }
            custom_tooltip = { text = frca_support_1c set_bar_progress = { value = -5 name = frca_support_bar } }
        }
        
        
        scope:guatemala_state = {
                                
                                                     
            add_radicals = {
                value = 0.15
                culture = cu:central_american
            }
            
            add_modifier = {
                name = ca_separatist_agitation_modifier
                months = 12
            }
        }        
    }
}

ca_central_america.2 = { 
    type = country_event
    placement = root
    title = ca_central_america.2.t
    desc = ca_central_america.2.d
    flavor = ca_central_america.2.f
    
    
    event_image = {
        video = "unspecific_politicians_arguing"
    }
    icon = "gfx/interface/icons/event_icons/event_gavel.dds"
    
    on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    duration = 3
    
    trigger = {
        # triggered by je
        this = c:UCA
        has_variable = frca_support
    }

    
    immediate = {
        s:STATE_GUATEMALA = {
            region_state:UCA = {
                save_scope_as = guatemala_state
            }
        }
        c:GUA ?= { save_scope_as = guatemala_country }
    }

    
    option = {
        name = ca_central_america.2.a 
        default_option = yes
        
        custom_tooltip = { text = frca_support_2a set_bar_progress = { value = -10 name = frca_support_bar } }

        set_variable = { name = los_altos_autonomy_approved_and_granted value = 1 }
        
        
        scope:guatemala_country ?= {
            ig:ig_landowners = { add_modifier = { name = ca_ig_approval_down months = 24 } }
        }
        
        ai_chance = {
            base = 50
        }
    }

    # OPCIÓN B: El Congreso Federal RECHAZA el nuevo estado
    option = {
        name = ca_central_america.2.b 
        
        set_variable = rejects_ALT
        
        custom_tooltip = { text = frca_support_2b set_bar_progress = { value = 10 name = frca_support_bar } }
        
        
        scope:guatemala_state = {
            add_modifier = { name = ca_separatist_agitation_modifier months = 12 }
            add_radicals = { value = 0.10 pop_type = academics }
            add_radicals = { value = 0.10 pop_type = shopkeepers }
            add_radicals = { value = 0.10 pop_type = clerks }
            add_radicals = { value = 0.10 pop_type = aristocrats }
        }
        
        ai_chance = {
            base = 50
        }
    }
}
