ca_central_america.1 = { 
    type = country_event
    placement = scope:guatemala_state    
    
    title = ca_central_america.1.t
    desc = ca_central_america.1.d
    flavor = ca_central_america.1.f
    
    event_image = {
        video = "southamerica_aristocrats"
    }
    
    icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
    
    on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    
    duration = 3
    
    trigger = {
        game_date >= 1836.5.1
        exists = c:UCA
        this = c:UCA 
       
        NOT = { has_global_variable = ca_los_altos_proposal_fired }
    }

    immediate = {
        set_global_variable = ca_los_altos_proposal_fired
        
        s:STATE_GUATEMALA = {
            region_state:UCA = {
                save_scope_as = guatemala_state
            }
        }
        
        c:GUA ?= { save_scope_as = guatemala_country }
    }
    
    option = {
        name = ca_central_america.1.a 
        default_option = yes
        
        scope:guatemala_country ?= { set_variable = more_liked_ALT }
        
        if = {
            limit = { has_variable = frca_support }
            custom_tooltip = { 
                text = frca_support_1a 
                je:je_historic_central_america_falls_apart = {
                    add_progress = { value = 20 name = frca_support_bar } 
                }
            }
        }
        
        scope:guatemala_state = {
            add_loyalists = {
                value = 0.1
                culture = cu:central_american
            }
        }        
        
        ai_chance = { base = 50 }
    }

    option = {
        name = ca_central_america.1.b 
        trigger = {
            OR = {
                has_law_or_variant = law_type:law_censorship
                has_law_or_variant = law_type:law_outlawed_dissent
                is_enacting_law = law_type:law_censorship
            }
        }
        
        scope:guatemala_country ?= { set_variable = censorship_liked_ALT }   
        
        scope:guatemala_state = {
            add_modifier = {
                name = ca_newspaper_censored_modifier
                months = 12
            }
        }
        ig:ig_intelligentsia = { add_modifier = { name = ca_ig_approval_down months = 12 } }
    }   

    option = {
        name = ca_central_america.1.c 
        
        scope:guatemala_country ?= { set_variable = ignore_ALT }
        
        if = {
            limit = { has_variable = frca_support }
            custom_tooltip = { 
                text = frca_support_1c 
                je:je_historic_central_america_falls_apart = {
                    add_progress = { value = -5 name = frca_support_bar } 
                }
            }
        }
        
        scope:guatemala_state = {
            add_radicals = {
                value = 0.15
                culture = cu:central_american
            }
            
            add_modifier = {
                name = ca_separatist_agitation_modifier
                months = 12
            }
        }        
    }
}

        }

        NOT = { has_global_variable = bocas_del_toro_crisis_fired } # Global para evitar que salte para COS y UCA a la vez
        OR = {        
            AND = {
                this = c:UCA 
                p:x47C5F8 = { owner = c:UCA }
            }
            AND = {
                this = c:COS 
                p:x47C5F8 = { owner = c:COS }
            }
        }
    }
    
    immediate = { 
        set_global_variable = bocas_del_toro_crisis_fired
        
        # Encontramos y guardamos el estado al que pertenece Bocas del Toro
        p:x47C5F8 = {
            state = { save_scope_as = bocas_state }
        }
    }
    
    
    option = {
        name = ca_central_america.6.a

        create_country = {
            tag = PP1
            origin = ROOT

                scope:bocas_state = {
                    set_owner_of_provinces = {
                        country = c:PP1
                        provinces = { x47C5F8 }
                    }
                }
                
                
                create_character = {
                    first_name = "John"
                    last_name = "Gallagher"
                    historical = yes
                    is_general = yes
                    interest_group = ig_armed_forces
                    ideology = ideology:ideology_market_liberal 
                    ruler = yes
                    ig_leader = yes            
                    traits = { explorer }
                }
            }   
        }   
        
        
        create_diplomatic_pact = {
            country = scope:scope_PP1
            type = puppet
        }
        
        set_variable = proclaim_pp1_uca
        
        
        if = {
            limit = { exists = c:CLM }
            c:CLM ?= {        
                trigger_event = { 
                    id = ca_central_america.12
                    months = 6
                }
            }
        }

            limit = { exists = c:GBR }
            change_relations = { country = c:GBR value = -30 }
        }
        
        ai_chance = { base = 33 }
    }

        name = ca_central_america.6.b 
        
        
        scope:bocas_state = {
            set_owner_of_provinces = {
                country = c:GBR
                provinces = { x47C5F8 }
            }
        }
        
        
        if = {
            limit = { exists = c:GBR }
            change_relations = { country = c:GBR value = 50 }
