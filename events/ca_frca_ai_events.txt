ca_frca_ai.2 = {
	type = country_event
	hidden = yes

	trigger = {
		is_ai = yes
		OR = {
			ca_frca_ai_historic = yes
			ca_frca_ai_dynamic = yes
		}
	}

	immediate = {
		# 1. Bocas del Toro con poco control (< 30)
		if = {
			limit = {
				ca_bocas_held_by_uca = yes
				NOT = { ca_bocas_control_ge = { VALUE = 30 } }
				ca_mm_region_ready = { KEY = boc_garrison }
				ca_mm_can_pay = { VALUE = ca_money_medium }
			}
			ca_mm_act_boc_garrison = yes
		}
		else_if = {
			limit = {
				ca_bocas_held_by_uca = yes
				NOT = { ca_bocas_control_ge = { VALUE = 30 } }
				NOT = { has_global_variable = ca_bocas_native_pact }
				ca_mm_region_ready = { KEY = boc_native_pact }
				ca_cohesion_can_spend = { VALUE = 5 }
			}
			ca_mm_act_boc_native_pact = yes
		}
		else_if = {
			limit = {
				ca_bocas_held_by_uca = yes
				NOT = { ca_bocas_control_ge = { VALUE = 30 } }
				exists = c:GBR
				ca_mm_region_ready = { KEY = boc_protest }
			}
			ca_mm_act_boc_protest = yes
		}
		# 2. Frente britanico (Mosquitia) en Tenso o peor
		else_if = {
			limit = {
				ca_frca_front_ge = { FRONT = british VALUE = 1 }
				ca_mm_region_ready = { KEY = mos_commissioner }
				ca_cohesion_can_spend = { VALUE = 10 }
				ca_mm_can_pay = { VALUE = ca_money_small }
			}
			ca_mm_act_mos_commissioner = yes
		}
		else_if = {
			limit = {
				ca_frca_front_ge = { FRONT = british VALUE = 1 }
				exists = c:MKT
				ca_mm_region_ready = { KEY = mos_king }
				ca_cohesion_can_spend = { VALUE = 5 }
			}
			ca_mm_act_mos_king = yes
		}
		else_if = {
			limit = {
				ca_frca_front_ge = { FRONT = british VALUE = 1 }
				exists = c:GBR
				ca_mm_region_ready = { KEY = mos_protest }
			}
			ca_mm_act_mos_protest = yes
		}
	}
}
