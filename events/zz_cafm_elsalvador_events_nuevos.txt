ca_el_salvador_events.14 = {
	type = country_event
	placement = ROOT
	title = ca_el_salvador_events.14.t
	desc = ca_el_salvador_events.14.d
	flavor = ca_el_salvador_events.14.f
	duration = 3
	event_image = { video = "unspecific_politicians_arguing" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	trigger = {
		ca_els_is_elsalvador = yes
	}

	immediate = {
		set_variable = { name = els_disolucion value = ca_els_disolucion_inicio }
		if = {
			limit = { NOT = { has_journal_entry = je_ca_els_bastion } }
			add_journal_entry = { type = je_ca_els_bastion }
		}
	}
	option = {
		name = ca_el_salvador_events.14.a
		default_option = yes
		set_variable = els_sostener_union
		ig:ig_intelligentsia ?= { add_modifier = { name = ca_ig_approval_up months = 36 } }
		custom_tooltip = {
			text = els_disolucion_down_5_tt
			change_variable = { name = els_disolucion subtract = 5 }
		}
		ai_chance = { base = 70 }
	}

	option = {
		name = ca_el_salvador_events.14.b
		trigger = {
			ca_els_alt_history = yes
		}
		set_variable = els_tesoro_vacio
		custom_tooltip = {
			text = els_disolucion_up_10_tt
			change_variable = { name = els_disolucion add = 10 }
		}
		ig:ig_landowners ?= { add_modifier = { name = ca_ig_approval_up months = 36 } }
		ai_chance = { base = 30 }
	}

}
