ca_guatemala_events.38 = {
	type = country_event
	placement = ROOT
	title = ca_guatemala_events.38.t
	desc = ca_guatemala_events.38.d
	flavor = ca_guatemala_events.38.f
	duration = 3
	event_image = { video = "asia_politician_parliament_motion" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	trigger = {
		ca_gua_is_guatemala = yes
	}

	immediate = {
		set_variable = { name = gua_montana value = ca_gua_montana_inicio }
		if = {
			limit = { NOT = { has_journal_entry = je_ca_gua_montana } }
			add_journal_entry = { type = je_ca_gua_montana }
		}
	}
	option = {
		name = ca_guatemala_events.38.a
		default_option = yes
		set_variable = gua_livingston_vigente
		add_modifier = { name = gua_codigos_livingston }
		ig:ig_devout ?= { add_modifier = { name = ca_ig_approval_down months = 36 } }
		ig:ig_rural_folk ?= { add_modifier = { name = ca_ig_approval_down months = 36 } }
		ig:ig_intelligentsia ?= { add_modifier = { name = ca_ig_approval_up months = 36 } }
		custom_tooltip = {
			text = gua_montana_up_10_tt
			change_variable = { name = gua_montana add = 10 }
		}
		ai_chance = { base = 70 }
	}

	option = {
		name = ca_guatemala_events.38.b
		trigger = {
			ca_gua_alt_history = yes
		}
		set_variable = gua_livingston_vigente
		set_variable = gua_livingston_capital
		add_modifier = { name = gua_livingston_capital months = 36 }
		custom_tooltip = {
			text = gua_montana_up_4_tt
			change_variable = { name = gua_montana add = 4 }
		}
		ai_chance = { base = 20 }
	}

	option = {
		name = ca_guatemala_events.38.c
		trigger = {
			ca_gua_alt_history = yes
		}
		set_variable = gua_livingston_suspendido
		ig:ig_intelligentsia ?= { add_modifier = { name = ca_ig_approval_down months = 36 } }
		ig:ig_devout ?= { add_modifier = { name = ca_ig_approval_up months = 36 } }
		custom_tooltip = {
			text = gua_montana_down_5_tt
			change_variable = { name = gua_montana subtract = 5 }
		}
		ai_chance = { base = 10 }
	}

}
