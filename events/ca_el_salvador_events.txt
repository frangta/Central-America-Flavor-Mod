## introduccion de El Salvador
ca_el_salvador_events.1 = {
	type = country_event
	placement = s:STATE_SAN_SALVADOR.region_state:UCA
	title = ca_el_salvador_events.1.t
	desc = ca_el_salvador_events.1.d
	flavor = ca_el_salvador_events.1.f
	duration = 3
	event_image = {
		video = "gfx/event_pictures/southamerica_aristocrats.bk2"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/southamerica/aristocrats"
	icon = "gfx/interface/icons/event_icons/event_fire.dds"
	trigger = {
        exists = c:UCA
        this = c:UCA 
        game_date <= 1836.1.1
	}
	
	immediate = {
        s:STATE_SAN_SALVADOR.region_state:UCA = {
        state_region = { add_state_trait = state_trait_mining_els } # CAFM fix: add_state_trait es de state_region
        add_modifier = { name = ca_uca_liberal_stronghold }
        add_modifier = { name = federal_stronghold }
        }
	}

	option = {
		name = ca_el_salvador_events.1.a
		default_option = yes

	}
	option = {
		name = ca_el_salvador_events.1.b
		trigger = {
			c:UCA ?= {
			has_variable = ca_state_selection_to_playing
			}
		}
        ca_create_elsalvador_effect = yes 
        ca_play_elsalvador_effect = yes		
        play_as = c:ELS
	}
}

# 1. 1838: La salida de Silva y la elección de Menéndez
ca_el_salvador_events.9 = {
    type = country_event
    event_image = { video = "southamerica_election" } # CAFM fix: sin imagen el evento muestra el cuadro magenta
    placement = ROOT
    title = ca_el_salvador_events.9.t
    desc = ca_el_salvador_events.9.d
    icon = "gfx/interface/icons/event_icons/event_election.dds"
    duration = 3

    trigger = {
        c:ELS ?= this
        in_election_campaign = yes
        NOT = { has_variable = els_elec_1838_fired }
        game_date >= 1838.2.1
        game_date <= 1839.1.1
    }
    
    immediate = { 
        set_variable = { name = els_elec_1838_fired value = yes } 
    }

    # Opción A (Histórica): Timoteo Menéndez (Liberal Moderado)
    option = {
        name = ca_el_salvador_events.9.a
        default_option = yes
        ig:ig_intelligentsia ?= { party = { add_momentum = momentum_large } }
        ig:ig_petty_bourgeoisie ?= { party = { add_momentum = momentum_medium } }
    }

    # Opción B (Alternativa): Oficiales radicales pro-Morazán exigen el control total
    option = {
        name = ca_el_salvador_events.9.b
        ig:ig_armed_forces ?= { party = { add_momentum = momentum_large } }
        add_radicals = { value = small_radicals pop_type = aristocrats }
    }

    # Opción C (Alternativa): Los conservadores intentan un golpe político
    option = {
        name = ca_el_salvador_events.9.c
        ig:ig_landowners ?= { party = { add_momentum = momentum_large } }
        ig:ig_devout ?= { party = { add_momentum = momentum_medium } }
    }
}

    type = country_event

    icon = "gfx/interface/icons/event_icons/event_election.dds"

        c:ELS ?= this
        in_election_campaign = yes
        NOT = { has_variable = els_elec_1848_fired }
        game_date >= 1848.2.1
        game_date <= 1849.1.1
    }

    option = {
        name = ca_el_salvador_events.13.a
        default_option = yes
        ig:ig_intelligentsia ?= { party = { add_momentum = momentum_large } }
        ig:ig_armed_forces ?= { party = { add_momentum = momentum_medium } } # Los militares lo apoyan por la inminente guerra

            limit = { exists = c:GUA }

        name = ca_el_salvador_events.13.b

        ig:ig_landowners ?= { party = { add_momentum = momentum_medium } }
        add_radicals = { value = medium_radicals pop_type = officers } # Los oficiales querían la guerra
    }
