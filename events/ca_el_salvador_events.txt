# 3. 1844: La Dictadura de Francisco Malespín
# CAFM F15 (E4): lado del estado de "El desafio de Malespin" (B1)
ca_el_salvador_events.11 = {
	type = country_event
	placement = root
	title = ca_el_salvador_events.11.t
	desc = ca_el_salvador_events.11.d
	flavor = ca_el_salvador_events.11.f
	event_image = { video = "unspecific_military_parade" }
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3

	trigger = {
		c:ELS ?= this
		NOT = { has_variable = els_elec_1844_fired }
		exists = ruler
		OR = {
			game_date >= 1844.2.1
			ig:ig_armed_forces ?= { is_powerful = yes }
		}
		game_date <= 1845.1.1
	}
	immediate = {
		set_variable = { name = els_elec_1844_fired value = yes }
		if = {
			limit = { has_global_variable = cafm_hc_malespin }
			global_var:cafm_hc_malespin = { save_scope_as = els_strongman }
		}
		else = {
			ruler = { save_scope_as = els_strongman }
		}
	}
	option = {
		name = ca_el_salvador_events.11.a
		default_option = yes
		ig:ig_armed_forces ?= { party ?= { add_momentum = momentum_very_large } }
		ig:ig_devout ?= { party ?= { add_momentum = momentum_large } }
		if = {
			limit = { is_subject_type = subject_type_ca_confederative_partner }
			overlord = { ca_conf_cohesion_change = { VALUE = -10 } }
			c:UCA ?= { ca_frca_front_change = { FRONT = caudillos VALUE = 1 } }
		}
		ai_chance = { base = 70
		modifier = { trigger = { has_global_variable = ca_world_historic } add = 300 } }
	}
	option = {
		name = ca_el_salvador_events.11.b
		ig:ig_intelligentsia ?= { party ?= { add_momentum = momentum_large } }
		ig:ig_petty_bourgeoisie ?= { party ?= { add_momentum = momentum_medium } }
		add_radicals = { value = large_radicals pop_type = officers }
		add_radicals = { value = medium_radicals pop_type = soldiers }
		ai_chance = { base = 20 }
	}
	option = {
		name = ca_el_salvador_events.11.c
		trigger = {
			is_subject_type = subject_type_ca_confederative_partner
		}
		overlord = { ca_conf_cohesion_change = { VALUE = 10 } }
		ig:ig_armed_forces ?= { add_modifier = { name = ca_ig_approval_down months = 24 } }
		ai_chance = { base = 10 }
	}
}
