ca_el_salvador_events.01 = {
	type = country_event
	placement = s:STATE_SAN_SALVADOR
	title = ca_el_salvador_events.01.t
	desc = ca_el_salvador_events.01.d
	flavor = ca_el_salvador_events.01.f
	duration = 3
	event_image = {
		video = "gfx/event_pictures/southamerica_aristocrats.bk2"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/southamerica/aristocrats"
	icon = "gfx/interface/icons/event_icons/event_fire.dds"
	trigger = {
        exists = c:UCA
        this = c:UCA 
        game_date <= 1836.1.15
	}
	
	immediate = {
        s:STATE_SAN_SALVADOR = {
        add_state_trait = state_trait_mining_els
        add_modifier = { name = ca_uca_liberal_stronghold }
        add_modifier = { name = federal_stronghold }
        }
	}

	option = {
		name = ca_el_salvador_events.01.a
		default_option = yes

	}
	option = {
		name = ca_el_salvador_events.02.a
		trigger = {
			c:UCA ?= {
			has_variable = ca_state_selection_to_playing
			}
		}
        ca_create_elsalvador_effect = yes 
        ca_play_elsalvador_effect = yes		
        play_as = c:ELS
	}
}
