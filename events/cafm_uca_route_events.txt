cafm_uca_route.7 = {
	type = country_event
	placement = root
	title = cafm_uca_route.7.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = cafm_uca_route.7.d_federal
				trigger = { has_variable = ca_bloc_event_bloc var:ca_bloc_event_bloc = flag:federal }
			}
			triggered_desc = {
				desc = cafm_uca_route.7.d_centralist
				trigger = { has_variable = ca_bloc_event_bloc var:ca_bloc_event_bloc = flag:centralist }
			}
			triggered_desc = {
				desc = cafm_uca_route.7.d_separatist
				trigger = { always = yes }
			}
		}
	}
	flavor = cafm_uca_route.7.f
	event_image = { video = "unspecific_politicians_arguing" }
	icon = "gfx/interface/icons/event_icons/event_election.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3

	option = {
		name = cafm_uca_route.7.a
		default_option = yes
		if = {
			limit = { var:ca_bloc_event_bloc = flag:federal }
			ca_uca_bar_add = { BAR = frca_support_bar VALUE = 10 }
		}
		else_if = {
			limit = { var:ca_bloc_event_bloc = flag:centralist }
			ca_uca_bar_add = { BAR = ca_frca_fvc_bar VALUE = ca_fvc_shift_c10 }
		}
		else = {
			ca_uca_bar_add = { BAR = frca_support_bar VALUE = -10 }
		}
		ai_chance = { base = 100 }
	}
}
