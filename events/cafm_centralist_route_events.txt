cafm_cen.3 = {
	type = country_event
	placement = root
	title = cafm_cen.3.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = cafm_cen.3.d_success
				trigger = { has_variable = ca_cen_kill_result var:ca_cen_kill_result = flag:success }
			}
			triggered_desc = {
				desc = cafm_cen.3.d_failure
				trigger = { always = yes }
			}
		}
	}
	flavor = cafm_cen.3.f
	event_image = { video = "unspecific_vandalized_storefront" }
	icon = "gfx/interface/icons/event_icons/event_skull.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3

	trigger = {
		c:UCA ?= this
		ca_frca_route_centralist = yes
		ca_cen_journal_active = yes
		has_variable = ca_cen_kill_result
	}

	# La conspiracion triunfa: Morazan muere y un general unitario toma el poder.
	option = {
		name = cafm_cen.3.a
		default_option = yes
		trigger = { var:ca_cen_kill_result = flag:success }
		ca_cen_install_general_ruler = yes
		ca_cen_morazan_out = { KILL = yes }
		ca_cen_morazan_removed = { CAUSE = kill }
		ca_cen_unity_add = { VALUE = ca_cen_unity_morazan_kill }
		add_modifier = { name = ca_cen_morazan_martyr months = 60 }
		ca_frca_loyalty_down = { TAG = els }
		remove_variable = ca_cen_kill_result
		ai_chance = { base = 100 }
	}
	# La conspiracion fracasa: escandalo.
	option = {
		name = cafm_cen.3.b
		trigger = { var:ca_cen_kill_result = flag:failure }
		ca_cen_army_add = { VALUE = ca_cen_army_scandal }
		ca_uca_rival_add = { VALUE = ca_fvc_shift_f10 }
		ig:ig_intelligentsia ?= { add_modifier = { name = ca_ig_approval_down_massive months = 36 } }
		remove_variable = ca_cen_kill_result
		ai_chance = { base = 100 }
	}
}

	trigger = {
		c:UCA ?= this
		ca_frca_route_centralist = yes
		ca_cen_journal_active = yes
		NOT = { has_variable = frca_morazan_removed }
	}

	immediate = {
		if = {
			limit = { ca_cen_election_centralist > ca_cen_election_federal }

		name = cafm_cen.8.b
