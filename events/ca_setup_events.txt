ca_setup_events.3 = {
	type = country_event
	hidden = yes

	trigger = {
		this = c:UCA
		# Solo se ejecuta una vez al inicio
		NOT = { has_variable = ca_setup_executed }
	}

	immediate = {
		set_variable = ca_setup_executed

		# 1. SOLUCIÓN A VARIABLES HUÉRFANAS (Logs de error)
		# Inicializamos variables de eventos históricos futuros
		set_variable = elsalvador_independence_event_var
		set_variable = losaltos_independence_event_var
		set_variable = guatemala_1843_event
		set_variable = guatemala_1858_event
		set_variable = guatemala_1871_event
		set_variable = nicaragua_1854_event
		set_variable = el_salvador_1865_event
		set_variable = honduras_1866_event
		
		# Inicialización de mecánicas de juego
		set_variable = ca_soil_survey_complete
		set_variable = ca_can_build_mines_globally
		set_variable = { name = frca_support value = 55 } # Soporte inicial moderado-bajo
		set_variable = { name = federalists_vs_separatists_var value = 0 }

		# 2. CONFIGURACIÓN GEOPOLÍTICA HISTÓRICA
		# Guatemala: Foco de la oposición conservadora (Cuna de Carrera)
		s:STATE_GUATEMALA = {
			add_modifier = {
				name = ca_separatist_agitation_modifier
				months = 48
			}
			# Guardamos targets para evitar errores de "Event target used but not set"
			save_global_event_target_as = disaster_state
		}

		# Costa Rica: Bajo Braulio Carrillo, buscando orden y café
		s:STATE_COSTA_RICA = {
			add_modifier = {
				name = ca_infrastructure_boom_modifier
				months = 60
			}
			save_global_event_target_as = success_state
		}

		# 3. SETUP DE LÍDERES
		# Francisco Morazán (Líder Federalista)
		if = {
			limit = {
				any_scope_character = { has_variable = is_morazan }
			}
			random_scope_character = {
				limit = { has_variable = is_morazan }
				# Aseguramos su ideología federalista
				set_role = ruler
			}
		}

		# 4. DISPARAR EVENTO DE LOBBIES
		trigger_event = { id = ca_setup_events.2 days = 2 }
	}
}


ca_setup_events.4 = {
	type = country_event
	hidden = yes

	immediate = {
		# Forzamos la creación del Lobby Pro-Federalista en los sujetos leales
		every_subject = {
			limit = {
				is_subject_type = subject_type_ca_dependency
			}
			# El efecto de lobby depende de tu archivo 00_ca_political_lobbies.txt
			# Aquí simulamos el inicio de la presión política
			add_modifier = {
				name = ca_direct_administration_modifier
				months = 24
			}
		}

		# Aplicamos el modificador de "Era de los Caudillos" a nivel nacional
		add_modifier = {
			name = age_of_ca_caudillos_modifier
			months = -1 # Permanente hasta que se remueva por evento
		}
		every_political_movement = {
            limit = {
                OR = {
                    is_political_movement_type = movement_liberal
                    is_political_movement_type = movement_cultural_majority
                    is_political_movement_type = movement_religious_majority
                }
            }
            # Aplicamos un modificador que destruya su radicalismo y soporte
            add_modifier = {
                name = ca_modifier_total_debug
                days = 2
            }
        }
	}
}
