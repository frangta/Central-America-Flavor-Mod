ca_talamanca_events.1 = {
	type = country_event
	placement = root
	title = ca_talamanca_events.1.t
	desc = ca_talamanca_events.1.d
	flavor = ca_talamanca_events.1.f
	event_image = { video = "middleeast_jungle_expedition" }
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3

	trigger = {
		has_journal_entry = je_ca_historic_situation_in_talamanca
		NOT = { has_global_variable = ca_tal_1_fired }
		game_date >= 1862.1.1

	}
	option = {
		name = ca_talamanca_events.1.a
		default_option = yes
		set_global_variable = ca_tal_1_fired
		ca_talamanca_change = { VALUE = 10 }
		add_treasury = { value = ca_money_small multiply = -1 }
		c:RDT ?= { change_relations = { country = root value = 20 } }
		ai_chance = { base = 60 }
	}
	option = {
		name = ca_talamanca_events.1.b
		set_global_variable = ca_tal_1_fired
		ca_talamanca_change = { VALUE = -5 }
		c:RDT ?= {
			if = {
				limit = { is_ai = no }
				trigger_event = { id = ca_talamanca_events.11 days = 3 }
			}
		}
		ai_chance = { base = 40 }
	}
}
