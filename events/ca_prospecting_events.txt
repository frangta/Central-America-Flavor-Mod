namespace = ca_prospecting 

ca_prospecting.1 = {
	type = country_event
	placement = scope:target_state
	title = ca_prospecting.1.t
	desc = ca_prospecting.1.d
	flavor = ca_prospecting.1.f
	
	trigger = {
		has_journal_entry = je_ca_national_geological_survey
		any_owned_state = {
			state_is_in_ca_central_america = yes
		}
	}
	
	immediate = {
		random_owned_state = {
			limit = {
				state_is_in_ca_central_america = yes
			}
			save_scope_as = target_state
		}
	}
	
	option = {
		name = ca_prospecting.1.a
		scope:target_state = {
			change_resource_potential = {
				type = building_gold_mine
				amount = 2
			}
			every_state_pop = {
				limit = { is_pop_type = peasants }
				add_radicals = { value = 0.05 }
			}
		}
	}
	
	option = {
		name = ca_prospecting.1.b
		add_modifier = {
			name = ca_add_power
			months = 36
		}
		change_variable = {
			name = ca_survey_progress
			subtract = 5
		}
		ig:ig_industrialists = {
			add_modifier = {
				name = ca_ig_approval_down
				months = 36
			}
		}
		ig:ig_landowners = {
			add_modifier = {
				name = ca_ig_approval_down
				months = 36
			}
		}
	}
}
