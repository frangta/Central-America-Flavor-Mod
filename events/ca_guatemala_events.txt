ca_guatemala_events.1 = {
	type = country_event
	placement = scope:secession_capital
	title = ca_guatemala_events.01.t
	desc = ca_guatemala_events.01.d
	flavor =  ca_guatemala_events.01.f

	duration = 3

	event_image = {
		video = "gfx/event_pictures/southamerica_aristocrats.bk2"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/southamerica/aristocrats"

	icon = "gfx/interface/icons/event_icons/event_fire.dds"

	trigger = {
               # triggered by je_historic_central_america_falls_apart when the player selected Guatemala play
	}
	
	immediate = {
		set_variable = {
			name = frca_recent_separatism_event_var
			days = 40
		}
		random_scope_state = {
			limit = {
				is_capital = no
				state_region = s:STATE_GUATEMALA
				turmoil > 0.2
			}
			save_scope_as = secession_state
		}
		create_country = {
			tag = GUA
			origin = this
			state = scope:secession_state
		}
		create_diplomatic_pact = {
            country = c:GUA
            type = ca_dependency
        }		
		c:GUA = {
			save_scope_as = scope_GUA
			save_scope_as = secession_country
		add_journal_entry = {
			type = je_ca_historic_situation_in_guatemala
		}			
			capital = {
				save_scope_as = secession_capital
			}
					set_relations = { country = c:UCA value = -50 }
		        create_character = {
				first_name = "jose"
				last_name = "nunez"
				historical = yes
				ig_leader = yes
				interest_group = ig_landowners
				ruler = yes
				birth_date = 1800.02.02
				female = no
				culture = cu:central_american
				ideology = ideology:ideology_meritocratic
				religion = rel:catholic
				traits = {
					basic_diplomat
				}
		    }
		ig:ig_landowners = {
			remove_ideology = ideology_paternalistic
			add_ideology = ideology_central_american_reactionary
			set_interest_group_name = ig_hacendados
		}
		
		ig:ig_armed_forces = {
			remove_ideology = ideology_paternalistic
		}
		ig_intelligentsia = {
		remove_ideology = ideology_liberal
        remove_ideology = ideology_republican		
		add_ideology = ideology_ca_constitutionalist

		}
		ig:ig_devout = {
			add_ideology = ideology_central_american_reactionary
			set_interest_group_name = ig_catholic_church
		}			
			activate_law = law_type:law_freedom_of_conscience
			#activate_law = law_type:law_unitary
			activate_law = law_type:law_mercantilism
			activate_law = law_type:law_wealth_voting
			activate_law = law_type:law_censorship	
		it = {
		limit = { has_variable = historic_central_american }
		#trigger_event = { 
		#	id = central_america.5 months = 1
		#}
		}
		it = {
		limit = { has_variable = random_central_american }
        random_events = {
			50 = ca_guatemala_events.13
			50 = ca_guatemala_events.14
			}
		}
		set_variable = frca_crisis_var
		trigger_event = { 
			id = central_america.5
			months = 1
		}
	 }
	}

	# Accept secession
	option = {
		name = ca_guatemala_events.01.a
		default_option = yes
		play_as = c:GUA
	}

	# Reject Secession
	option = {
		name = ca_guatemala_events.01.b
		
	}
	option = { #Switch tags
		name = ca_guatemala_events.01.c
		
		
		
		ai_chance = {
			base = 0
		}
	}
