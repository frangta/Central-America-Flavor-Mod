        }

		name = ca_guatemala_events.1.a
		default_option = yes

	}

		name = ca_guatemala_events.2.a
		trigger = {
			c:UCA ?= {
			has_variable = ca_state_selection_to_playing
			}
		}
       ca_create_guatemala_effect = yes
	ca_play_guatemala_effect = yes 
    play_as = c:GUA
	}

# 1. 1837-1838: La Crisis de Gálvez y la Rebelión de la Montaña
ca_guatemala_events.5 = {
    type = country_event
    event_image = { video = "southamerica_election" } # CAFM fix: sin imagen el evento muestra el cuadro magenta
    placement = ROOT
    title = ca_guatemala_events.5.t
    desc = ca_guatemala_events.5.d
    icon = "gfx/interface/icons/event_icons/event_election.dds"
    duration = 3

    trigger = {
        c:GUA ?= this
        in_election_campaign = yes
        NOT = { has_variable = gua_elec_1837_fired }
        game_date >= 1837.1.1
        game_date <= 1838.12.31
    }
    
    immediate = { 
        set_variable = { name = gua_elec_1837_fired value = yes } 
    }

    # Opción A (Histórica): Los conservadores designan a Pedro José Valenzuela / Rivera Paz
    option = {
        name = ca_guatemala_events.5.a
        default_option = yes
        ig:ig_landowners ?= { party = { add_momentum = momentum_large } }
        ig:ig_devout ?= { party = { add_momentum = momentum_medium } }
        ig:ig_intelligentsia ?= { party = { add_momentum = momentum_large_decrease } }
    }

    # Opción B (Alternativa): Mariano Gálvez resiste en el poder (Liberales)
    option = {
        name = ca_guatemala_events.5.b
        ig:ig_intelligentsia ?= { party = { add_momentum = momentum_large } }
        # Retener a Gálvez causa furia en el campo
        add_radicals = { value = large_radicals pop_type = peasants }
    }

    # Opción C (Alternativa): Rafael Carrera toma el poder tempranamente
    option = {
        name = ca_guatemala_events.5.c
        ig:ig_rural_folk ?= { party = { add_momentum = momentum_large } }
        ig:ig_armed_forces ?= { party = { add_momentum = momentum_medium } }
    }
}
