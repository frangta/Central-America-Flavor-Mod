# 1. 1837-1838: La Crisis de Gálvez y la Rebelión de la Montaña
# CAFM F15 (E4): solo "La caida de Galvez" (1838); la rebelion esta en ca_central_america.121/.110
ca_guatemala_events.5 = {
	type = country_event
	placement = root
	title = ca_guatemala_events.5.t
	desc = ca_guatemala_events.5.d
	flavor = ca_guatemala_events.5.f
	event_image = { video = "southamerica_war_civilians" }
	icon = "gfx/interface/icons/event_icons/event_fire.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3

	trigger = {
		c:GUA ?= this
		NOT = { has_variable = gua_elec_1837_fired }
		game_date >= 1838.2.1
		game_date <= 1838.12.31
	}
	immediate = {
		set_variable = { name = gua_elec_1837_fired value = yes }
	}
	option = {
		name = ca_guatemala_events.5.a
		default_option = yes
		ig:ig_landowners ?= { party ?= { add_momentum = momentum_large } }
		ig:ig_devout ?= { party ?= { add_momentum = momentum_medium } }
		ig:ig_intelligentsia ?= { party ?= { add_momentum = momentum_large_decrease } }
		set_variable = gua_galvez_fallen
		ai_chance = { base = 70
		modifier = { trigger = { c:UCA ?= { has_variable = ca_gua_reforms_pushed } } add = 20 }
		modifier = { trigger = { has_global_variable = ca_world_historic } add = 300 } }
	}
	option = {
		name = ca_guatemala_events.5.b
		trigger = {
			NOT = { has_global_variable = ca_world_historic }
		}
		ig:ig_intelligentsia ?= { party ?= { add_momentum = momentum_large } }
		add_radicals = { value = large_radicals pop_type = peasants }
		ai_chance = { base = 20 }
	}
	option = {
		name = ca_guatemala_events.5.c
		trigger = {
			ig:ig_rural_folk ?= { is_powerful = yes }
		}
		ig:ig_rural_folk ?= { party ?= { add_momentum = momentum_large } }
		ig:ig_armed_forces ?= { party ?= { add_momentum = momentum_medium } }
		set_variable = gua_galvez_fallen
		c:UCA ?= { ca_frca_front_change = { FRONT = conservative VALUE = 1 } }
		ai_chance = { base = 10 }
	}
}
