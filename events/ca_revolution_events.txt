namespace = ca_revolution_events

# CAFM P11: guerra civil y revolucion. .1-.5 revolucion federalista, .6-.9 unitaria,
# .10-.12 cualquier revolucion de esos dos movimientos o secesion. Pulso: ca_p11_revolution_monthly.

ca_revolution_events.1 = {
	type = country_event
	placement = scope:ca_p11_state
	title = ca_revolution_events.1.t
	desc = ca_revolution_events.1.d
	flavor = ca_revolution_events.1.f
	event_image = { video = "unspecific_devastation" }
	icon = "gfx/interface/icons/event_icons/event_fire.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3
	cooldown = { days = 1825 }

	trigger = {
		ca_p11_country = yes
		NOT = { has_variable = ca_p11_rev_cd }
		any_civil_war = {
			is_civil_war_type = revolution
			political_movement ?= { is_political_movement_type = movement_ca_federalist }
		}
		any_scope_state = { is_in_revolt = yes }
	}
	cancellation_trigger = {
		NOT = { exists = scope:ongoing_revolution }
	}

	immediate = {
		random_civil_war = {
			limit = {
				is_civil_war_type = revolution
				political_movement ?= { is_political_movement_type = movement_ca_federalist }
			}
			save_scope_as = ongoing_revolution
		}
		random_scope_state = {
			limit = { is_in_revolt = yes }
			save_scope_as = ca_p11_state
		}
		set_variable = { name = ca_p11_rev_cd days = 30 }
	}
	option = {
		name = ca_revolution_events.1.a
		default_option = yes
		scope:ongoing_revolution = { add_civil_war_progress = revolution_retreat_medium }
		scope:ca_p11_state = { add_radicals_in_state = { value = medium_radicals strata = middle } }
		ai_chance = { base = 50 }
	}
	option = {
		name = ca_revolution_events.1.b
		scope:ca_p11_state = { add_modifier = { name = ca_p11_state_barricades months = 6 } }
		random_list = {
			1 = { scope:ongoing_revolution = { add_civil_war_progress = revolution_retreat_small } }
			1 = { scope:ongoing_revolution = { add_civil_war_progress = revolution_advance_small } }
		}
		ai_chance = { base = 50 }
	}
}
