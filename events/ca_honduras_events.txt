ca_honduras_events.1 = {
	type = country_event
	placement = s:STATE_HONDURAS
	title = ca_honduras_events.1.t
	desc = ca_honduras_events.1.d
	flavor = ca_honduras_events.1.f
	duration = 3
	event_image = {
		video = "gfx/event_pictures/southamerica_aristocrats.bk2"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/southamerica/aristocrats"
	icon = "gfx/interface/icons/event_icons/event_fire.dds"
	trigger = {
        exists = c:UCA
        this = c:UCA 
        game_date <= 1836.1.15
	}
	
	immediate = {
        s:STATE_HONDURAS = {
        add_state_trait = state_trait_mining_hon
		add_modifier = { name = ca_uca_federalist_heartland }
        add_modifier = { name = waning_federal_stronghold }
        add_modifier = { name = conservative_stronghold_small_nation }
        }
	}

	option = {
		name = ca_honduras_events.1.a
		default_option = yes

	}
	option = {
		name = ca_honduras_events.2.a
		trigger = {
			c:UCA ?= {
			has_variable = ca_state_selection_to_playing
			}
		}
        ca_create_honduras_effect = yes 
		ca_play_honduras_effect = yes
        play_as = c:HON
	}
}
# 1. 1837: La elección de Justo José de Herrera
ca_honduras_events.14 = {
    type = country_event
    placement = ROOT
    title = ca_honduras_events.14.t
    desc = ca_honduras_events.14.d
    icon = "gfx/interface/icons/event_icons/event_election.dds"
    duration = 3

    trigger = {
        c:HON ?= this
        in_election_campaign = yes
        NOT = { has_variable = hon_elec_1837_fired }
        game_date >= 1837.1.1
        game_date <= 1838.1.1
    }
    
    immediate = { 
        set_variable = { name = hon_elec_1837_fired value = yes } 
    }

    # Opción A (Histórica): Justo José de Herrera asume como civil moderado
    option = {
        name = ca_honduras_events.14.a
        default_option = yes
        ig:ig_landowners ?= { party = { add_momentum = momentum_medium } }
        ig:ig_intelligentsia ?= { party = { add_momentum = momentum_medium } }
    }

    # Opción B (Alternativa): Los militares exigen un líder de armas
    option = {
        name = ca_honduras_events.14.b
        ig:ig_armed_forces ?= { party = { add_momentum = momentum_large } }
        add_radicals = { value = small_radicals pop_type = academics }
    }

    # Opción C (Alternativa): Los liberales radicales intentan tomar la Asamblea
    option = {
        name = ca_honduras_events.14.c
        ig:ig_petty_bourgeoisie ?= { party = { add_momentum = momentum_large } }
        ig:ig_landowners ?= { party = { add_momentum = momentum_small_decrease } }
    }
}
