## .2 — Dos ciudades, un Estado
ca_honduras_events.2 = {
	type = country_event
	placement = ROOT
	title = ca_honduras_events.2.t
	desc = ca_honduras_events.2.d
	flavor = ca_honduras_events.2.f
	duration = 3
	event_image = { video = "southamerica_aristocrats" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/southamerica/aristocrats"
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	trigger = {
		c:HON ?= this
		game_date < 1836.1.1
		NOT = { has_variable = hon_e2_fired }
	}

	immediate = {
		set_variable = hon_e2_fired
		if = {
			limit = { c:UCA ?= { has_variable = historic_central_american } }
			set_variable = hon_historic_mode
		}
		# Marca a Rivera en tiempo de juego (no depende de qué plantilla duplicada cargó)
		ruler ?= { set_variable = is_rivera }
		set_variable = { name = hon_british value = 0 }
		# Diario del caudillo: la variable va ANTES del diario para que el objetivo quede en 100
		set_variable = { name = hon_caudillo value = 15 }
		add_journal_entry = { type = je_ca_hon_caudillo }
		# Arranque: Estado inacabado, milicias de comarca y moneda de 1832
		add_modifier = { name = hon_estado_inacabado months = 60 }
		add_modifier = { name = hon_milicias_comarca months = 60 }
		add_modifier = { name = hon_media_leche }
	}

	# Histórica: Comayagua, capital desde 1830
	option = {
		name = ca_honduras_events.2.a
		default_option = yes
		set_variable = hon_seat_comayagua
		# F15 (E2): terratenientes y clero contentos; Tegucigalpa agraviada 2 anos
		ca_fvs_bar_change = { VALUE = 5 }
		s:STATE_HONDURAS.region_state:HON ?= { add_modifier = { name = hon_ciudad_agraviada months = 24 } }
		ig:ig_devout ?= { add_modifier = { name = ca_ig_approval_up months = 24 } }
		ig:ig_landowners ?= { add_modifier = { name = ca_ig_approval_up months = 24 } }
		ig:ig_intelligentsia ?= { add_modifier = { name = ca_ig_approval_down months = 24 } }
		ai_chance = { base = 80 }
	}

	# El dinero está en Tegucigalpa
	option = {
		name = ca_honduras_events.2.b
		set_variable = hon_seat_tegucigalpa
		# F15 (E2): mineros y comerciantes contentos; Comayagua agraviada 2 anos; gancho de la capital (1880)
		ig:ig_industrialists ?= { add_modifier = { name = ca_ig_approval_up months = 24 } }
		ig:ig_petty_bourgeoisie ?= { add_modifier = { name = ca_ig_approval_up months = 24 } }
		s:STATE_HONDURAS.region_state:HON ?= { add_modifier = { name = hon_ciudad_agraviada months = 24 } }
		set_variable = hon_capital_question_1880
		add_modifier = { name = hon_sede_tegucigalpa months = 60 }
		ig:ig_devout ?= { add_modifier = { name = ca_ig_approval_down months = 24 } }
		ig:ig_landowners ?= { add_modifier = { name = ca_ig_approval_down months = 24 } }
		custom_tooltip = {
			text = hon_caudillo_up_5_tt
			change_variable = { name = hon_caudillo add = 5 }
		}
		ai_chance = { base = 15 }
	}

	# Capital alterna entre las dos ciudades
	option = {
		name = ca_honduras_events.2.c
		set_variable = hon_seat_alterna
		set_variable = { name = hon_sede_alterna_push days = 730 } # F15 (E2): barra del caudillo +1 al mes, 2 anos
		add_modifier = { name = hon_sede_alterna months = 60 }
		ai_chance = { base = 5 }
	}
}

ca_honduras_events.42 = {
	type = country_event
	placement = root
	title = ca_honduras_events.42.t
	desc = ca_honduras_events.42.d
	flavor = ca_honduras_events.42.f
	event_image = { video = "unspecific_signed_contract" }
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3

	trigger = {
		c:HON ?= this
		has_journal_entry = je_ca_hon_caudillo
		NOT = { has_variable = hon_pespire }
		OR = {
			has_variable = hon_pespire_talks
			AND = {
				has_variable = hon_historic_mode
				game_date >= 1850.1.1
			}
			AND = {
				NOT = { has_variable = hon_historic_mode }
				game_date >= 1847.1.1
				has_variable = hon_low_caudillo_months
				var:hon_low_caudillo_months >= 12
				NOT = { ruler ?= { interest_group = { is_interest_group_type = ig_armed_forces } } }
			}
		}
		any_scope_character = { has_role = character_role_general }
	}
	immediate = {
		if = {
			limit = {
				has_variable = hon_historic_mode
				has_global_variable = cafm_hc_guardiola
			}
			global_var:cafm_hc_guardiola = { save_scope_as = hon_exile }
		}
		else = {
			ordered_scope_character = {
				limit = { has_role = character_role_general }
				order_by = popularity
				save_scope_as = hon_exile
			}
		}
		if = {
			limit = { has_variable = hon_historic_mode exists = c:ELS }
			c:ELS = { save_scope_as = hon_mediator }
		}
		else = {
			ordered_country = {
				limit = {
					ca_is_ca_national_country = yes
					NOT = { this = root }
					"relations(root)" >= relations_threshold:cordial
				}
				order_by = "relations(root)"
				save_scope_as = hon_mediator
			}
		}
	}
	option = {
		name = ca_honduras_events.42.a
		default_option = yes
		set_variable = hon_pespire
		remove_variable = hon_pespire_talks
		add_modifier = { name = hon_gobierno_civil months = 60 }
		ig:ig_armed_forces ?= { add_modifier = { name = ca_ig_approval_down months = 36 } }
		scope:hon_exile = { exile_character = yes }
		set_variable = { name = hon_exile_may_return days = 1825 }
		if = {
			limit = { exists = scope:hon_mediator }
			change_relations = { country = scope:hon_mediator value = 30 }
		}
		ai_chance = { base = 1 }
	}
}
