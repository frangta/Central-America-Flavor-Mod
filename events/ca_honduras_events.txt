## .2 — Dos ciudades, un Estado
ca_honduras_events.2 = {
	type = country_event
	placement = ROOT
	title = ca_honduras_events.2.t
	desc = ca_honduras_events.2.d
	flavor = ca_honduras_events.2.f
	duration = 3
	event_image = { video = "southamerica_aristocrats" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/southamerica/aristocrats"
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	trigger = {
		c:HON ?= this
		game_date < 1837.1.1
		NOT = { has_variable = hon_e2_fired }
	}

	immediate = {
		set_variable = hon_e2_fired
		if = {
			limit = { c:UCA ?= { has_variable = historic_central_american } }
			set_variable = hon_historic_mode
		}
		# Marca a Rivera en tiempo de juego (no depende de qué plantilla duplicada cargó)
		ruler ?= { set_variable = is_rivera }
		set_variable = { name = hon_british value = 0 }
		# Diario del caudillo: la variable va ANTES del diario para que el objetivo quede en 100
		set_variable = { name = hon_caudillo value = 15 }
		add_journal_entry = { type = je_ca_hon_caudillo }
		# Arranque: Estado inacabado, milicias de comarca y moneda de 1832
		add_modifier = { name = hon_estado_inacabado months = 60 }
		add_modifier = { name = hon_milicias_comarca months = 60 }
		add_modifier = { name = hon_media_leche }
	}

	# Histórica: Comayagua, capital desde 1830
	option = {
		name = ca_honduras_events.2.a
		default_option = yes
		set_variable = hon_seat_comayagua
		ig:ig_devout ?= { add_modifier = { name = ca_ig_approval_up months = 24 } }
		ig:ig_landowners ?= { add_modifier = { name = ca_ig_approval_up months = 24 } }
		ig:ig_intelligentsia ?= { add_modifier = { name = ca_ig_approval_down months = 24 } }
		ai_chance = { base = 80 }
	}

	# El dinero está en Tegucigalpa
	option = {
		name = ca_honduras_events.2.b
		set_variable = hon_seat_tegucigalpa
		add_modifier = { name = hon_sede_tegucigalpa months = 60 }
		ig:ig_devout ?= { add_modifier = { name = ca_ig_approval_down months = 24 } }
		ig:ig_landowners ?= { add_modifier = { name = ca_ig_approval_down months = 24 } }
		custom_tooltip = {
			text = hon_caudillo_up_5_tt
			change_variable = { name = hon_caudillo add = 5 }
		}
		ai_chance = { base = 15 }
	}

	# Capital alterna entre las dos ciudades
	option = {
		name = ca_honduras_events.2.c
		set_variable = hon_seat_alterna
		add_modifier = { name = hon_sede_alterna months = 60 }
		ai_chance = { base = 5 }
	}
}

# 1. 1837: La elección de Justo José de Herrera
ca_honduras_events.14 = {
    type = country_event
    event_image = { video = "southamerica_election" } # CAFM fix: sin imagen el evento muestra el cuadro magenta
    placement = ROOT
    title = ca_honduras_events.14.t
    desc = ca_honduras_events.14.d
    icon = "gfx/interface/icons/event_icons/event_election.dds"
    duration = 3

    trigger = {
        c:HON ?= this
        in_election_campaign = yes
        NOT = { has_variable = hon_elec_1837_fired }
        game_date >= 1837.1.1
        game_date <= 1838.1.1
    }
    
    immediate = { 
        set_variable = { name = hon_elec_1837_fired value = yes } 
    }

    # Opción A (Histórica): Justo José de Herrera asume como civil moderado
    option = {
        name = ca_honduras_events.14.a
        default_option = yes
        # CAFM Fase 1: Herrera gobierna desde el 28 may 1837
        if = {
            limit = {
                has_global_variable = cafm_hc_justo_herrera
                any_scope_character = { this = global_var:cafm_hc_justo_herrera }
            }
            random_scope_character = {
                limit = { this = global_var:cafm_hc_justo_herrera }
                set_character_as_ruler = yes
            }
        }
        else = {
            create_character = {
                template = hon_justo_jose_herrera_character_template
                ruler = yes
            }
        }
        ig:ig_landowners ?= { party = { add_momentum = momentum_medium } }
        ig:ig_intelligentsia ?= { party = { add_momentum = momentum_medium } }
    }

    # Opción B (Alternativa): Los militares exigen un líder de armas
    option = {
        name = ca_honduras_events.14.b
        ig:ig_armed_forces ?= { party = { add_momentum = momentum_large } }
        add_radicals = { value = small_radicals pop_type = academics }
    }

    # Opción C (Alternativa): Los liberales radicales intentan tomar la Asamblea
    option = {
        name = ca_honduras_events.14.c
        ig:ig_petty_bourgeoisie ?= { party = { add_momentum = momentum_large } }
        ig:ig_landowners ?= { party = { add_momentum = momentum_small_decrease } }
    }
}

    event_image = { video = "southamerica_election" } # CAFM fix: sin imagen el evento muestra el cuadro magenta
    placement = ROOT
    title = ca_honduras_events.17.t
    desc = ca_honduras_events.17.d
    icon = "gfx/interface/icons/event_icons/event_election.dds"
    duration = 3

    trigger = {
        c:HON ?= this
        in_election_campaign = yes
        NOT = { has_variable = hon_elec_1847_fired }
        game_date >= 1847.1.1
        game_date <= 1848.1.1
    }
    
    immediate = { 
        set_variable = { name = hon_elec_1847_fired value = yes } 
    }

    # Opción A (Histórica): El estadista Juan Lindo promete educación e institucionalidad
    option = {
        name = ca_honduras_events.17.a
        default_option = yes
        ig:ig_intelligentsia ?= { party = { add_momentum = momentum_very_large } }
        ig:ig_landowners ?= { party = { add_momentum = momentum_medium } }
    }

    # Opción B (Alternativa): Ferrera intenta sabotear a Lindo desde las sombras
    option = {
        name = ca_honduras_events.17.b
        ig:ig_armed_forces ?= { party = { add_momentum = momentum_large } }
        add_radicals = { value = medium_radicals pop_type = academics }
    }
}

# 5. 1848: La Constitución de 1848 y la Ratificación
ca_honduras_events.18 = {
    type = country_event
    event_image = { video = "southamerica_election" } # CAFM fix: sin imagen el evento muestra el cuadro magenta
    placement = ROOT
    title = ca_honduras_events.18.t
    desc = ca_honduras_events.18.d
    icon = "gfx/interface/icons/event_icons/event_election.dds"
    duration = 3

    trigger = {
        c:HON ?= this
        in_election_campaign = yes
        NOT = { has_variable = hon_elec_1848_fired }
        game_date >= 1848.6.1
        game_date <= 1849.6.1
    }
    
    immediate = { 
        set_variable = { name = hon_elec_1848_fired value = yes } 
    }

    # Opción A (Histórica): Juan Lindo ratificado bajo la nueva constitución civil
    option = {

        ig:ig_intelligentsia ?= { party = { add_momentum = momentum_large } }

    option = {
        name = ca_honduras_events.18.b
        ig:ig_armed_forces ?= { party = { add_momentum = momentum_very_large } }

        add_radicals = { value = medium_radicals pop_type = officers }
    }
}
